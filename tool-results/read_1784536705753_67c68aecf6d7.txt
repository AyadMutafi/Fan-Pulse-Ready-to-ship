    22→    22→ *      already attached to that player in the source pool — we copy it
    23→    23→ *      verbatim and pick the player's BEST fact (for Elite) or WORST fact
    24→    24→ *      (for Crisis) when a player appears in multiple rounds.
    25→    25→ *   3. tournamentScore is an APP-INTERNAL composite PER ROUND
    26→    26→ *      (pulseScore × 0.8 + trendBonus × 0.2). For Elite we keep the HIGHEST

    28→    28→ *      Crisis we keep the LOWEST round score across all rounds where they

    30→    30→ *      labelled as "verified". It exists only to rank players relative to
    31→    31→ *      each other.

    33→    33→ *      next-best available verified player rather than inventing. If the entire

   256→   256→type PosGroup = 'GK' | 'DEF' | 'MID' | 'FWD'
   257→   257→function posGroup(p: { position: string }): PosGroup {
   258→   258→  if (p.position === 'GK') return 'GK'
   259→   259→  if (['CB', 'LB', 'RB'].includes(p.position)) return 'DEF'
   260→   260→  if (['CM', 'CAM', 'CDM'].includes(p.position)) return 'MID'

   264→   264→/**
   265→   265→ * A player merged across all rounds where they appeared.
   266→   266→ *
   267→   267→ * `eliteScore` is the HIGHEST round score across every round in which the
   268→   268→ * player appeared on the elite path (group stage, R32 advanced, R16, QF, SF,
   269→   269→ * Final). `crisisScore` is the LOWEST round score across every round in which
   270→   270→ * the player appeared on the crisis path (group stage, R32 eliminated, R16,
   271→   271→ * QF, SF, Final). A player can appear in BOTH paths (e.g. Achraf Hakimi was
   272→   272→ * Elite in R16 but Crisis in QF) — the two scores are tracked independently so
   273→   273→ * the same verified player can anchor either XI. `trend` is the trend from
   274→   274→ * their most recent (latest round) appearance regardless of path.

   276→   276→interface MergedPlayer {
   277→   277→  name: string
   278→   278→  nationCode: string
   279→   279→  position: GroupStageEntry['position']
   280→   280→  /** Hero-path composite score, or null if the player never appeared as a hero. */
   281→   281→  eliteScore: number | null
   282→   282→  /** Villain-path composite score, or null if the player never appeared as a villain. */
   283→   283→  crisisScore: number | null
   284→   284→  /** Best hero fact (for Elite XI matchInfo). */
   285→   285→  eliteMatchInfo: string | null
   286→   286→  /** Worst villain fact (for Crisis XI matchInfo). */
   287→   287→  crisisMatchInfo: string | null
   288→   288→  /** Most recent trend across stages. */
   289→   289→  trend: Trend
   290→   290→}

   292→   292→// Trend bonus: rising = 80, stable = 50, falling = 20.
   293→   293→function trendBonus(trend: Trend): number {
   294→   294→  if (trend === 'rising') return 80
   295→   295→  if (trend === 'stable') return 50
   296→   296→  return 20
   297→   297→}

   299→   299→/**
   300→   300→ * Merge ALL round pools (group stage, R32, R16, QF, SF, Final) into one map of
   301→   301→ * MergedPlayer records, keyed by player name.
   302→   302→ *
   303→   303→ * Per-round scoring:
   304→   304→ *   roundScore = pulseScore × 0.8 + trendBonus × 0.2
   305→   305→ *
   306→   306→ * Per-path aggregation:
   307→   307→ *   - Elite: eliteScore = HIGHEST roundScore across all elite-path appearances;
   308→   308→ *     eliteMatchInfo cites the fact from that best round.
   309→   309→ *   - Crisis: crisisScore = LOWEST roundScore across all crisis-path appearances;
   310→   310→ *     crisisMatchInfo cites the fact from that worst round.
   311→   311→ *
   312→   312→ * A player can appear in BOTH paths (e.g. Achraf Hakimi was Elite in R16 but
   313→   313→ * Crisis in QF) — the two scores are tracked independently. The `trend` on the
   314→   314→ * merged player reflects the trend from their MOST RECENT (latest round)
   315→   315→ * appearance, regardless of path. Rounds are processed in chronological order
   316→   316→ * (group → R32 → R16 → QF → SF → Final) so the last write wins for `trend`.
   317→   317→ *
   318→   318→ * R32 VERIFIED_POOL is wrapped into R32_ELITE / R32_CRISIS (see above) so it
   319→   319→ * can be processed uniformly: advanced → elite path (rising trend),
   320→   320→ * eliminated → crisis path (falling trend). 'upcoming' R32 entries are dropped
   321→   321→ * (no result yet).
   322→   322→ */
   323→   323→function mergeAllPlayers(): Map<string, MergedPlayer> {
   324→   324→  const map = new Map<string, MergedPlayer>()
   325→   325→
   326→   326→  const ensure = (
   327→   327→    name: string,
   328→   328→    nationCode: string,
   329→   329→    position: GroupStageEntry['position'],
   330→   330→  ): MergedPlayer => {
   331→   331→    let p = map.get(name)
   332→   332→    if (!p) {
   333→   333→      p = {
   334→   334→        name, nationCode, position,
   335→   335→        eliteScore: null, crisisScore: null,
   336→   336→        eliteMatchInfo: null, crisisMatchInfo: null,
   337→   337→        trend: 'stable' as Trend,
   338→   338→      }
   339→   339→      map.set(name, p)
   340→   340→    }
   341→   341→    return p
   342→   342→  }
   343→   343→
   344→   344→  // Compute a per-round score: 80% weight on the round's pulse/buzz score,
   345→   345→  // 20% weight on the trend bonus (rising 80, stable 50, falling 20).
   346→   346→  const roundScore = (pulse: number, trend: Trend): number =>
   347→   347→    Math.round(pulse * 0.8 + trendBonus(trend) * 0.2)
   348→   348→
   349→   349→  // Process rounds in chronological order so the merged `trend` reflects the
   350→   350→  // most recent (latest) round each player appeared in.
   351→   351→  for (const round of ALL_ROUNDS) {
   352→   352→    // ── Elite path: keep the HIGHEST round score + the fact from that round ──
   353→   353→    for (const g of round.elitePool) {
   354→   354→      const p = ensure(g.name, g.nationCode, g.position)
   355→   355→      const score = roundScore(g.pulseScore, g.trend)
   356→   356→      if (p.eliteScore === null || score > p.eliteScore) {
   357→   357→        p.eliteScore = score
   358→   358→        p.eliteMatchInfo = g.matchInfo
   359→   359→      }
   360→   360→      p.trend = g.trend
   361→   361→    }
   362→   362→    // ── Crisis path: keep the LOWEST round score + the fact from that round ──
   363→   363→    for (const g of round.crisisPool) {
   364→   364→      const p = ensure(g.name, g.nationCode, g.position)
   365→   365→      const score = roundScore(g.pulseScore, g.trend)
   366→   366→      if (p.crisisScore === null || score < p.crisisScore) {
   367→   367→        p.crisisScore = score
   368→   368→        p.crisisMatchInfo = g.matchInfo
   369→   369→      }
   370→   370→      p.trend = g.trend
   371→   371→    }
   372→   372→  }
   373→   373→
   374→   374→  return map
   375→   375→}

   380→   380→  /** Stable id derived from name + position group. */
   381→   381→  id: string
   382→   382→  name: string
   383→   383→  nationCode: string
   384→   384→  position: string
   385→   385→  /** App-internal composite 0-100. NOT a verified statistic. */
   386→   386→  tournamentScore: number
   387→   387→  /** Verified match fact (copied verbatim from the source pool). */
   388→   388→  matchInfo: string | null
   389→   389→  trend: Trend
   390→   390→}

   392→   392→export interface RetroSide {
   393→   393→  formation: string

   397→   397→export interface TournamentRetroResult {
   398→   398→  elite: RetroSide
   399→   399→  crisis: RetroSide
   400→   400→  generatedAt: string
   401→   401→}
   402→   402→
   403→   403→// ── Formation slot config (4-3-3) ────────────────────────────────────────────
   404→   404→const FORMATION = '4-3-3'
   405→   405→const SLOTS: { group: PosGroup; count: number }[] = [
   406→   406→  { group: 'GK', count: 1 },
   407→   407→  { group: 'DEF', count: 4 },
   408→   408→  { group: 'MID', count: 3 },
   409→   409→  { group: 'FWD', count: 3 },
   410→   410→]
   411→   411→const XI_SIZE = SLOTS.reduce((n, s) => n + s.count, 0) // 11
   412→   412→
   413→   413→/**
   414→   414→ * Pick an XI (4-3-3) from a candidate pool.
   415→   415→ *
   416→   416→ * @param candidates  players with a non-null score for this side
   417→   417→ * @param descending  true for Elite (highest score first), false for Crisis (lowest first)
   418→   418→ *
   419→   419→ * ANTI-HALLUCINATION: if a position group is short, we fill from the next-best
   420→   420→ * available verified player. If the entire pool is exhausted, the slot is
   421→   421→ * returned as a null placeholder (name: 'N/A') — we never fabricate.
   422→   422→ */
   423→   423→function pickXI(
   424→   424→  candidates: MergedPlayer[],
   425→   425→  scoreOf: (p: MergedPlayer) => number | null,
   426→   426→  descending: boolean,
   427→   427→): RetroPick[] {
   428→   428→  // Filter to players that actually have a score for this side, then sort.
   429→   429→  const eligible = candidates
   430→   430→    .filter(p => scoreOf(p) !== null)
   431→   431→    .sort((a, b) => {
   432→   432→      const sa = scoreOf(a)!
   433→   433→      const sb = scoreOf(b)!
   434→   434→      return descending ? sb - sa : sa - sb
   435→   435→    })
   436→   436→
   437→   437→  const used = new Set<string>()
   438→   438→  const picks: (MergedPlayer | null)[] = []
   439→   439→
   440→   440→  // Pass 1: fill each position-group slot from players of that group.
   441→   441→  for (const slot of SLOTS) {
   442→   442→    let filled = 0
   443→   443→    for (const p of eligible) {
   444→   444→      if (filled >= slot.count) break
   445→   445→      if (used.has(p.name)) continue
   446→   446→      if (posGroup(p) === slot.group) {
   447→   447→        picks.push(p)
   448→   448→        used.add(p.name)
   449→   449→        filled++
   450→   450→      }
   451→   451→    }
   452→   452→    // Pad with nulls if this group was under-filled.
   453→   453→    while (filled < slot.count) {
   454→   454→      picks.push(null)
   455→   455→      filled++
   456→   456→    }
   457→   457→  }
   458→   458→
   459→   459→  // Pass 2: replace any null placeholders with the next-best unused verified
   460→   460→  // player (regardless of position group) so the XI fills 11 real names.
   461→   461→  for (let i = 0; i < picks.length; i++) {
   462→   462→    if (picks[i] !== null) continue
   463→   463→    const next = eligible.find(p => !used.has(p.name))
   464→   464→    if (next) {
   465→   465→      picks[i] = next
   466→   466→      used.add(next.name)
   467→   467→    }
   468→   468→    // If no unused player remains, the slot stays null → 'N/A'.
   469→   469→  }
   470→   470→
   471→   471→  // Map to RetroPick, assigning an order within the formation.
   472→   472→  return picks.map((p, i) => {
   473→   473→    const slotIndex = i
   474→   474→    const group = SLOTS.reduce<{ group: PosGroup; count: number } | null>((acc, s) => {
   475→   475→      if (acc) return acc
   476→   476→      return s
   477→   477→    }, null)
   478→   478→    void group // (slot metadata derived from index instead)
   479→   479→    if (!p) {
   480→   480→      return {
   481→   481→        id: `na-${slotIndex}`,
   482→   482→        name: 'N/A',
   483→   483→        nationCode: '',
   484→   484→        position: '',
   485→   485→        tournamentScore: 0,
   486→   486→        matchInfo: null,
   487→   487→        trend: 'stable' as Trend,
   488→   488→      }
   489→   489→    }
   490→   490→    const score = scoreOf(p)!
   491→   491→    const matchInfo = descending ? p.eliteMatchInfo : p.crisisMatchInfo
   492→   492→    return {
   493→   493→      id: `${p.name.toLowerCase().replace(/[^a-z0-9]+/g, '-')}-${slotIndex}`,
   494→   494→      name: p.name,
   495→   495→      nationCode: p.nationCode,
   496→   496→      position: p.position,
   497→   497→      tournamentScore: score,
   498→   498→      matchInfo,
   499→   499→      trend: p.trend,
   500→   500→    }
   501→   501→  })
   502→   502→}
   503→   503→
   504→   504→// ── Main entry point ─────────────────────────────────────────────────────────
   505→   505→
   506→   506→/**
   507→   507→ * Compute the Team of the Tournament Retro (Elite XI + Crisis XI).
   508→   508→ *
   509→   509→ * Pure function — no DB, no SDK. The result is deterministic for a given
   510→   510→ * VERIFIED_POOL + group-stage pool, so the API route caches it for 1 hour.
   511→   511→ */
   512→   512→export function computeTournamentRetro(): TournamentRetroResult {
   513→   513→  const merged = mergeAllPlayers()
   514→   514→  const all = Array.from(merged.values())
   515→   515→
   516→   516→  const elite = pickXI(all, p => p.eliteScore, true)
   517→   517→  const crisis = pickXI(all, p => p.crisisScore, false)
   518→   518→
   519→   519→  return {
   520→   520→    elite: { formation: FORMATION, players: elite },
   521→   521→    crisis: { formation: FORMATION, players: crisis },
   522→   522→    generatedAt: new Date().toISOString(),
   523→   523→  }
   524→   524→}

   526→   526→// ── Anti-hallucination self-check (exported for tests / dev scripts) ─────────
   527→   527→
   528→   528→/**
   529→   529→ * Returns the set of every player name that is eligible to appear in the retro.
   530→   530→ * Used by the API route's self-check to guarantee no fabricated name slipped in.
   531→   531→ *
   532→   532→ * Includes every player from every round pool: group stage, R32 (VERIFIED_POOL),
   533→   533→ * R16, QF, SF, and Final. (3rd Place is intentionally skipped — every player
   534→   534→ * who appeared in the 3rd Place match already appears in an earlier round.)
   535→   535→ */
   536→   536→export function getAllVerifiedNames(): Set<string> {
   537→   537→  const names = new Set<string>()
   538→   538→  for (const round of ALL_ROUNDS) {
   539→   539→    for (const g of round.elitePool) names.add(g.name)
   540→   540→    for (const g of round.crisisPool) names.add(g.name)
   541→   541→  }
   542→   542→  // VERIFIED_POOL is already covered by R32_ELITE + R32_CRISIS, but we iterate
   543→   543→  // it directly as a belt-and-braces safety net (in case a future 'upcoming'
   544→   544→  // entry is added to the pool — those aren't routed into R32_ELITE/CRISIS).
   545→   545→  for (const r of VERIFIED_POOL) names.add(r.name)
   546→   546→  return names
   547→   547→}
