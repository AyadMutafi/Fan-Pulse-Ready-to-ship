   278→// ── Formation card (Elite gold / Crisis red) ─────────────────────────────────

   280→function RetroFormationCard({
   281→  side,
   282→  variant,
   283→  flagMode,
   284→}: {
   285→  side: RetroSide
   286→  variant: 'elite' | 'crisis'
   287→  flagMode: 'emoji' | 'flag'
   288→}) {
   289→  const isElite = variant === 'elite'
   290→  const accent = isElite ? '#F59E0B' : '#EF4444' // gold for Elite, red for Crisis
   291→  const accentSoft = isElite ? 'rgba(245,158,11,0.15)' : 'rgba(239,68,68,0.15)'
   292→  const badgeBg = isElite ? 'bg-[#F59E0B]' : 'bg-[#EF4444]'
   293→  const label = isElite ? 'PULSE ELITE' : 'CRISIS RADAR'
   294→  const emoji = isElite ? '🌟' : '⚠️'
   295→  const subtitle = isElite
   296→    ? 'The tournament heroes — top performers across all 64 matches'
   297→    : 'The tournament villains — worst performers across all 64 matches'
   298→
   299→  const realPlayers = side.players.filter(p => p.name !== 'N/A')
   300→  const avgScore =
   301→    realPlayers.length > 0
   302→      ? Math.round(realPlayers.reduce((s, p) => s + p.tournamentScore, 0) / realPlayers.length)
   303→      : 0
   304→
   305→  const { markSeen: markCardSeen } = useCardCollection()
   306→
   307→  // Convert RetroPick (API response) → PlayerCardData using verified tier logic.
   308→  // Uses pulseScore if present (verified Elite XI), falls back to tournamentScore.
   309→  const retroToCardData = (p: RetroPick): PlayerCardData => {
   310→    const score = p.pulseScore ?? p.tournamentScore
   311→    const isAwardWinner = !!p.isAwardWinner
   312→    const isYoungBreakout = VERIFIED_YOUNG_BREAKOUT_NAMES.has(p.name)
   313→    return {
   314→      id: `retro:${variant}:${p.name}`,
   315→      name: p.name,
   316→      nationCode: p.nationCode,
   317→      position: p.position || '—',
   318→      pulseScore: score,
   319→      scoreLabel: 'Pulse Score',
   320→      trend: p.trend,
   321→      isAwardWinner,
   322→      awardName: p.awardName,
   323→      isYoungBreakout,
   324→      tier: getCardTier(score, p.trend, isAwardWinner, isYoungBreakout),
   325→      verifiedNote: p.matchInfo ?? `Tournament score ${p.tournamentScore}`,
   326→      source: isElite ? 'Team of Tournament' : 'Crisis XI',
   327→      fanSentiment: p.sentiment,
   328→    }
   329→  }
   330→
   331→  // Organize into formation columns: GK | DEF | MID | FWD (landscape, like WC tab)
   332→  const gk = side.players.filter(p => p.position === 'GK' || p.name === 'N/A' && side.players.indexOf(p) === 0)
   333→  const def = side.players.filter(p => ['CB', 'LB', 'RB'].includes(p.position))
   334→  const mid = side.players.filter(p => ['CM', 'CAM', 'CDM'].includes(p.position))
   335→  const fwd = side.players.filter(p => ['LW', 'RW', 'ST', 'CF'].includes(p.position))
   336→  // Fallback: if a player has an empty position (N/A slot), drop them into the
   337→  // shortest column so the pitch still shows 11 slots.
   338→  const nas = side.players.filter(p => p.name === 'N/A' && p.position === '')
   339→  const columns = [gk, def, mid, fwd]
   340→  for (const na of nas) {
   341→    const shortest = columns.reduce((a, b) => (a.length <= b.length ? a : b))
   342→    shortest.push(na)
   343→  }
   344→
   345→  return (
   346→    <motion.div
   347→      initial={{ opacity: 0, y: 16 }}
   348→      animate={{ opacity: 1, y: 0 }}
   349→      transition={{ duration: 0.4 }}
   350→      className="rounded-2xl overflow-hidden border-2 shadow-sm"
   351→      style={{ borderColor: accent, boxShadow: `0 4px 12px ${accentSoft}` }}
   352→    >
   353→      {/* Top accent bar */}
   354→      <div
   355→        className="h-1 w-full"
   356→        style={{
   357→          background: isElite
   358→            ? 'linear-gradient(to right, #F59E0B, #FBBF24, #F59E0B)'
   359→            : 'linear-gradient(to right, #EF4444, #DC2626, #EF4444)',
   360→        }}
   361→      />
   362→      {/* Header */}
   363→      <div className="flex items-center gap-2 px-4 py-2.5" style={{ background: accentSoft }}>
   364→        <span className="text-lg">{emoji}</span>
   365→        <div className="flex-1">
   366→          <div className="flex items-center gap-2">
   367→            <h3 className="text-sm font-black" style={{ color: accent }}>{label}</h3>
   368→            <Badge
   369→              className={`${badgeBg} text-white border-0 text-[11px] font-bold px-1.5 py-0 gap-1`}
   370→            >
   371→              <Lock className="size-2.5" />
   372→              {side.formation}
   373→            </Badge>
   374→          </div>
   375→          <p className="text-[11px] text-[#666] dark:text-[#CCCCCC] mt-0.5">{subtitle}</p>
   376→        </div>
   377→        <div className="text-right">
   378→          <p className="text-[11px] font-bold text-[#666] dark:text-[#CCCCCC]">AVG</p>
   379→          <p className="brutalist-number text-sm font-black" style={{ color: accent }}>{(avgScore / 10).toFixed(1)}</p>
   380→        </div>
   381→      </div>
   382→
   383→      {/* Pitch */}
   384→      <div className="px-3 py-3 bg-[#F8F9FA] dark:bg-[#2D2D2D]">
   385→        <div className="mx-auto max-w-[520px]">
   386→          <div className={`pitch-bg relative ${isElite ? '' : 'crisis-pitch'}`}>
   387→            <RetroPitchMarkings crisis={!isElite} />
   388→            <div className="relative z-10 px-1.5 py-1 flex justify-between items-center h-full">
   389→              {columns.map((col, ci) => (
   390→                <div key={ci} className="flex flex-col items-center gap-0.5">
   391→                  {col.map((player) => (
   392→                    <RetroPlayerChip key={player.id} player={player} accent={accent} flagMode={flagMode} isElite={isElite} />
   393→                  ))}
   394→                </div>
   395→              ))}
   396→            </div>
   397→          </div>
   398→        </div>
   399→      </div>
   400→
   401→      {/* Player Cards — FUT-style collectible cards (tap to flip) */}
   402→      <div className="px-4 py-3 bg-white dark:bg-[#1A1A1A] border-t border-[#E0E0E0]/50 dark:border-white/5">
   403→        <p className="text-[10px] font-bold text-[#666] dark:text-[#CCCCCC] mb-2 uppercase tracking-wide">
   404→          {isElite ? 'Elite Player Cards' : 'Crisis Player Cards'} · tap to flip
   405→        </p>
   406→        <div className="flex gap-3 overflow-x-auto pb-2 scrollbar-none -mx-1 px-1">
   407→          {realPlayers.map((player) => (
   408→            <div key={`card-${player.id}`} className="shrink-0">
   409→              <PlayerCard data={retroToCardData(player)} size="compact" onView={markCardSeen} />
   410→            </div>
   411→          ))}
   412→        </div>
   413→      </div>
   414→
   415→      {/* Match facts list (scrollable if long) */}
   416→      <div className="px-4 py-3 bg-white dark:bg-[#1A1A1A] border-t border-[#E0E0E0]/50 dark:border-white/5">
   417→        <p className="text-[10px] font-bold text-[#666] dark:text-[#CCCCCC] mb-2 uppercase tracking-wide">
   418→          {isElite ? 'Tournament-defining moments' : 'Where it went wrong'}
   419→        </p>
   420→        <div className="max-h-40 overflow-y-auto space-y-1.5 pr-1 retro-scroll">
   421→          {realPlayers.map((p) => (
   422→            <div key={p.id} className="flex items-start gap-2 text-xs">
   423→              <span
   424→                className="shrink-0 mt-0.5 size-1.5 rounded-full"
   425→                style={{ background: accent }}
   426→              />
   427→              <span className="font-bold text-[#1A1A1A] dark:text-white min-w-[90px] shrink-0 flex items-center gap-1">
   428→                {p.name}
   429→                {p.isAwardWinner && (
   430→                  <span
   431→                    title={p.awardName ?? 'Award winner'}
   432→                    className="inline-flex items-center gap-0.5 rounded-full bg-[#F59E0B] text-white px-1 py-0 text-[7px] font-black leading-none"
   433→                  >
   434→                    <Trophy className="size-2" />{p.awardName ?? 'AWARD'}
   435→                  </span>
   436→                )}
   437→              </span>
   438→              <span className="text-[#666] dark:text-[#CCCCCC] text-[11px]">{p.matchInfo ?? '—'}</span>
   439→            </div>
   440→          ))}
   441→        </div>
   442→      </div>
   443→    </motion.div>
   444→  )
   445→}

   447→// ── Player chip on the pitch ─────────────────────────────────────────────────

   449→function RetroPlayerChip({
   450→  player,
   451→  accent,
   452→  flagMode,
   453→  isElite,
   454→}: {
   455→  player: RetroPick
   456→  accent: string
   457→  flagMode: 'emoji' | 'flag'
   458→  isElite: boolean
   459→}) {
   460→  const team = findNationalTeam(player.nationCode)
   461→  const flagEmoji = team?.flag ?? '🏳️'
   462→  const isNA = player.name === 'N/A'
   463→  const faceEmoji = isNA ? '❓' : getPulseFaceEmoji(player.tournamentScore)
   464→  const ratingValue = isNA ? '—' : (player.tournamentScore / 10).toFixed(1)
   465→  // Trend shown as a colored arrow on the avatar's top-right corner — kept
   466→  // visually distinct from the position badge and rating pill below.

   572→// ── Share-as-image button ────────────────────────────────────────────────────
   573→// The Web Share API can share files (canvas screenshots) on supporting browsers.
   574→// As a lightweight fallback we generate a text blob the user can save/screenshot.
   575→
   576→function ShareAsImageButton({ text }: { text: string }) {
   577→  const [shared, setShared] = useState(false)
   578→
   579→  const handleShareImage = async () => {
   580→    // Build a canvas "share card" with the retro summary.
   581→    try {
   582→      const canvas = document.createElement('canvas')
   583→      canvas.width = 1080
   584→      canvas.height = 1080
   585→      const ctx = canvas.getContext('2d')
   586→      if (!ctx) throw new Error('no ctx')
   587→
   588→      // Background gradient
   589→      const grad = ctx.createLinearGradient(0, 0, 1080, 1080)
   590→      grad.addColorStop(0, '#1A1A1A')
   591→      grad.addColorStop(1, '#2D2D2D')
   592→      ctx.fillStyle = grad
   593→      ctx.fillRect(0, 0, 1080, 1080)
   594→
   595→      // Gold border
   596→      ctx.strokeStyle = '#F59E0B'
   597→      ctx.lineWidth = 8
   598→      ctx.strokeRect(24, 24, 1032, 1032)
   599→
   600→      // Title
   601→      ctx.fillStyle = '#F59E0B'
   602→      ctx.font = 'bold 56px sans-serif'
   603→      ctx.fillText('🏆 Team of the Tournament', 60, 110)
   604→      ctx.fillStyle = '#FFFFFF'
   605→      ctx.font = 'bold 32px sans-serif'
   606→      ctx.fillText('2026 FIFA World Cup', 60, 160)
   607→
   608→      // Body text (wrapped)
   609→      ctx.font = '28px sans-serif'
   610→      ctx.fillStyle = '#FFFFFF'
   611→      const lines = wrapText(ctx, text.replace(/^🏆.*$/m, '').trim(), 960)
   612→      let y = 240
   613→      for (const line of lines) {
   614→        ctx.fillText(line, 60, y)
   615→        y += 40
   616→      }
   617→
   618→      // Footer
   619→      ctx.fillStyle = '#999999'
   620→      ctx.font = '24px sans-serif'
   621→      ctx.fillText('Fan Pulse — real fan sentiment', 60, 1020)
   622→
   623→      // Try Web Share with the canvas as a file (mobile), else download.
   624→      canvas.toBlob(async (blob) => {
   625→        if (!blob) {
   626→          toast.error('Could not generate share card')
   627→          return
   628→        }
   629→        const file = new File([blob], 'team-of-tournament-2026.png', { type: 'image/png' })
   630→        const nav = navigator as Navigator & {
   631→          canShare?: (data: { files: File[] }) => boolean
   632→          share?: (data: { files?: File[]; title?: string; text?: string }) => Promise<void>
   633→        }
   634→        if (nav.canShare && nav.canShare({ files: [file] }) && nav.share) {
   635→          try {
   636→            await nav.share({ files: [file], title: 'Team of the Tournament 2026', text: text.slice(0, 200) })
   637→            setShared(true)
   638→            setTimeout(() => setShared(false), 2000)
   639→          } catch (err) {
   640→            if (err instanceof Error && err.name !== 'AbortError') {
   641→              toast.error('Share cancelled')
   642→            }
   643→          }
   644→        } else {
   645→          // Download fallback
   646→          const url = URL.createObjectURL(blob)
   647→          const a = document.createElement('a')
   648→          a.href = url
   649→          a.download = 'team-of-tournament-2026.png'
   650→          a.click()
   651→          URL.revokeObjectURL(url)
   652→          toast.success('Share card saved — post it anywhere!')
   653→          setShared(true)
   654→          setTimeout(() => setShared(false), 2000)
   655→        }
   656→      }, 'image/png')
   657→    } catch {
   658→      toast.error('Could not generate share card')
   659→    }
   660→  }
   661→
   662→  return (
   663→    <Button
   664→      size="sm"
   665→      onClick={handleShareImage}
   666→      variant="outline"
   667→      className="gap-1.5 text-[11px] font-bold h-8 rounded-lg border-[#F59E0B]/40 text-[#F59E0B] hover:bg-[#F59E0B]/10"
   668→    >
   669→      {shared ? <Check className="size-3" /> : <Share2 className="size-3" />}
   670→      {shared ? 'Shared' : 'Share Image'}
   671→    </Button>
   672→  )
   673→}

   675→// Canvas text wrapper helper
   676→function wrapText(ctx: CanvasRenderingContext2D, text: string, maxWidth: number): string[] {
   677→  const out: string[] = []
   678→  const paragraphs = text.split('\n')
   679→  for (const para of paragraphs) {
   680→    if (!para) { out.push(''); continue }
   681→    const words = para.split(' ')
   682→    let line = ''
   683→    for (const word of words) {
   684→      const test = line ? `${line} ${word}` : word
   685→      if (ctx.measureText(test).width > maxWidth && line) {
   686→        out.push(line)
   687→        line = word
   688→      } else {
   689→        line = test
   690→      }
   691→    }
   692→    if (line) out.push(line)
   693→  }
   694→  return out
   695→}
