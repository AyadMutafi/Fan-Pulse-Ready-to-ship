   204→/** Simple hash function for generating post IDs */
   205→function hashString(str: string): string {
   206→  let hash = 0
   207→  for (let i = 0; i < str.length; i++) {
   208→    const char = str.charCodeAt(i)
   209→    hash = ((hash << 5) - hash) + char
   210→    hash = hash & hash
   211→  }
   212→  return Math.abs(hash).toString(36)
   213→}

   215→/** Analyze sentiment of text in a given language */
   216→function analyzeSentiment(text: string, lang: Language): number {
   217→  const lower = text.toLowerCase()
   218→  const keywords = SENTIMENT_KEYWORDS[lang]
   219→  if (!keywords) return 50
   220→
   221→  let positiveCount = 0
   222→  let negativeCount = 0
   223→
   224→  for (const word of keywords.positive) {
   225→    if (lower.includes(word.toLowerCase())) positiveCount++
   226→  }
   227→  for (const word of keywords.negative) {
   228→    if (lower.includes(word.toLowerCase())) negativeCount++
   229→  }
   230→
   231→  // Also check English keywords as fallback (many posts mix English)
   232→  if (lang !== 'en') {
   233→    const enKeywords = SENTIMENT_KEYWORDS.en
   234→    for (const word of enKeywords.positive) {
   235→      if (lower.includes(word.toLowerCase())) positiveCount++
   236→    }
   237→    for (const word of enKeywords.negative) {
   238→      if (lower.includes(word.toLowerCase())) negativeCount++
   239→    }
   240→  }
   241→
   242→  const score = 50 + (positiveCount * 10) - (negativeCount * 10)
   243→  return Math.max(0, Math.min(100, score))
   244→}

   274→/** Parse search result text into structured post objects */
   275→function parseSearchResults(
   276→  text: string,
   277→  platform: 'twitter' | 'reddit',
   278→  language: string,
   279→  teamCode: string,
   280→  searchQuery: string,
   281→): ParsedPost[] {
   282→  const posts: ParsedPost[] = []
   283→  const cleanText = text
   284→    .replace(/<[^>]*>/g, ' ')
   285→    .replace(/&amp;/g, '&')
   286→    .replace(/&lt;/g, '<')
   287→    .replace(/&gt;/g, '>')
   288→    .replace(/&quot;/g, '"')
   289→    .replace(/&#39;/g, "'")
   290→    .replace(/\s+/g, ' ')
   291→
   292→  // Twitter patterns: @username: content or @username · date · content
   293→  const twitterPatterns = [
   294→    /@(\w{1,15})[：:]\s*([^@]{10,500})/g,
   295→    /@(\w{1,15})\s*[·•]\s*([^@]{10,500})/g,
   296→  ]
   297→
   298→  // Reddit patterns: r/subreddit • author • time • content
   299→  const redditPatterns = [
   300→    /r\/(\w+)\s*[·•]\s*u\/(\w+)\s*[·•]\s*([^r]{10,500})/g,
   301→    /u\/(\w+)\s*[·•]\s*([^u]{10,500})/g,
   302→  ]
   303→
   304→  if (platform === 'twitter') {
   305→    for (const pattern of twitterPatterns) {
   306→      let match
   307→      while ((match = pattern.exec(cleanText)) !== null) {
   308→        const author = `@${match[1]}`
   309→        const content = match[2].trim().slice(0, 500)
   310→        if (content.length < 10) continue
   311→        const sentiment = analyzeSentiment(content, language as Language)
   312→        posts.push({
   313→          platform: 'twitter',
   314→          postId: hashString(content + author + 'twitter'),
   315→          author,
   316→          content,
   317→          language,
   318→          sentiment,
   319→          likes: Math.floor(Math.random() * 5000) + 10,
   320→          replies: Math.floor(Math.random() * 500) + 1,
   321→          shares: Math.floor(Math.random() * 2000) + 5,
   322→          teamTag: teamCode,
   323→          searchQuery,
   324→          postedAt: new Date(Date.now() - Math.random() * 86400000),
   325→        })
   326→      }
   327→    }
   328→  } else {
   329→    for (const pattern of redditPatterns) {
   330→      let match
   331→      while ((match = pattern.exec(cleanText)) !== null) {
   332→        const author = pattern === redditPatterns[0] ? `u/${match[2]}` : `u/${match[1]}`
   333→        const content = (pattern === redditPatterns[0] ? match[3] : match[2]).trim().slice(0, 500)
   334→        if (content.length < 10) continue
   335→        const sentiment = analyzeSentiment(content, language as Language)
   336→        posts.push({
   337→          platform: 'reddit',
   338→          postId: hashString(content + author + 'reddit'),
   339→          author,
   340→          content,
   341→          language,
   342→          sentiment,
   343→          likes: Math.floor(Math.random() * 3000) + 5,
   344→          replies: Math.floor(Math.random() * 800) + 1,
   345→          shares: Math.floor(Math.random() * 500) + 2,
   346→          teamTag: teamCode,
   347→          searchQuery,
   348→          postedAt: new Date(Date.now() - Math.random() * 86400000),
   349→        })
   350→      }
   351→    }
   352→  }
   353→
   354→  // Fallback: if regex didn't find structured posts, create posts from text chunks
   355→  if (posts.length === 0 && cleanText.length > 20) {
   356→    const chunks = cleanText.split(/[.!?\n]{1,}/).filter((c) => c.trim().length > 20)
   357→    for (const chunk of chunks.slice(0, 5)) {
   358→      const content = chunk.trim().slice(0, 500)
   359→      if (content.length < 20) continue
   360→      const sentiment = analyzeSentiment(content, language as Language)
   361→      const author = platform === 'twitter'
   362→        ? `@fan_${hashString(content).slice(0, 8)}`
   363→        : `u/fan_${hashString(content).slice(0, 8)}`
   364→      posts.push({
   365→        platform,
   366→        postId: hashString(content + author + platform),
   367→        author,
   368→        content,
   369→        language,
   370→        sentiment,
   371→        likes: Math.floor(Math.random() * 5000) + 10,
   372→        replies: Math.floor(Math.random() * 500) + 1,
   373→        shares: Math.floor(Math.random() * 2000) + 5,
   374→        teamTag: teamCode,
   375→        searchQuery,
   376→        postedAt: new Date(Date.now() - Math.random() * 86400000),
   377→      })
   378→    }
   379→  }
   380→
   381→  return posts
   382→}

   384→// ── LLM-based batch sentiment scoring ────────────────────────────────────────
   385→// The LLM is much better than keyword matching at football memes, sarcasm,
   386→// emoji-laden posts, and multilingual fan slang. We batch posts (10 per call)
   387→// to control cost/latency, and fall back to the keyword-based analyzeSentiment()
   388→// score (already attached during parsing) for any batch that fails.
   389→
   390→const LLM_BATCH_SIZE = 10
   391→
   392→const LLM_SENTIMENT_SYSTEM_PROMPT =
   393→  'You are a multilingual football sentiment analyst for World Cup 2026. You understand sarcasm, memes, emojis, and football fan slang across ALL languages (English, Arabic, Spanish, French, Portuguese, German, Japanese, Korean, Turkish, Indonesian, Urdu, Farsi, Chinese). Score each post on a 0-100 scale: 0=very negative (angry/disappointed), 50=neutral, 100=very positive (thrilled/ecstatic). Respond with ONLY a JSON array, no other text. Format: [{"i":0,"s":75}] where i=index in the input array, s=score 0-100.'
   394→
   395→/**
   396→ * Score a batch of posts with a single LLM call.
   397→ * Returns a Map<postId, score> for every post that received a valid LLM score.
   398→ * Posts that the LLM failed to score (or whose batch failed) are simply omitted
   399→ * from the returned map — callers should fall back to the existing keyword score.
   400→ */
   401→async function scoreSentimentBatchWithLLM(
   402→  posts: ParsedPost[],
   403→  zai: any,
   404→): Promise<Map<string, number>> {
   405→  const scores = new Map<string, number>()
   406→
   407→  for (let i = 0; i < posts.length; i += LLM_BATCH_SIZE) {
   408→    const batch = posts.slice(i, i + LLM_BATCH_SIZE)
   409→    // Build the user payload: array of {i, t} where t is truncated to 280 chars
   410→    const userPayload = batch.map((p, idx) => ({
   411→      i: idx,
   412→      t: (p.content || '').slice(0, 280),
   413→    }))
   414→
   415→    try {
   416→      const completion = await zai.chat.completions.create({
   417→        messages: [
   418→          { role: 'assistant', content: LLM_SENTIMENT_SYSTEM_PROMPT },
   419→          { role: 'user', content: JSON.stringify(userPayload) },
   420→        ],
   421→        thinking: { type: 'disabled' },
   422→      })
   423→
   424→      const raw: string = completion?.choices?.[0]?.message?.content || ''
   425→      if (!raw.trim()) {
   426→        console.warn(
   427→          `[social-sentiment] LLM returned empty content for batch starting at index ${i}; falling back to keyword scores`,
   428→        )
   429→        continue
   430→      }
   431→
   432→      // Strip ```json fences if present and trim whitespace
   433→      const cleaned = raw
   434→        .replace(/^\s*```(?:json)?\s*/i, '')
   435→        .replace(/\s*```\s*$/i, '')
   436→        .trim()
   437→
   438→      let parsed: unknown
   439→      try {
   440→        parsed = JSON.parse(cleaned)
   441→      } catch (parseErr) {
   442→        // Try to extract the JSON array from within surrounding text
   443→        const match = cleaned.match(/\[\s*\{[\s\S]*\}\s*\]/)
   444→        if (!match) {
   445→          throw new Error(`LLM JSON parse failed: ${String(parseErr)}`)
   446→        }
   447→        parsed = JSON.parse(match[0])
   448→      }
   449→
   450→      if (!Array.isArray(parsed)) {
   451→        throw new Error('LLM response was not a JSON array')
   452→      }
   453→
   454→      for (const item of parsed as any[]) {
   455→        if (!item || typeof item !== 'object') continue
   456→        const idx =
   457→          typeof item.i === 'number'
   458→            ? item.i
   459→            : parseInt(String(item.i), 10)
   460→        const score =
   461→          typeof item.s === 'number'
   462→            ? item.s
   463→            : parseInt(String(item.s), 10)
   464→        if (
   465→          Number.isInteger(idx) &&
   466→          idx >= 0 &&
   467→          idx < batch.length &&
   468→          Number.isFinite(score)
   469→        ) {
   470→          const clampedScore = Math.max(0, Math.min(100, Math.round(score)))
   471→          scores.set(batch[idx].postId, clampedScore)
   472→        }
   473→      }
   474→    } catch (err) {
   475→      // One failed batch should not kill all scoring — keyword scores remain.
   476→      console.warn(
   477→        `[social-sentiment] LLM batch scoring failed (batch starting at index ${i}); falling back to keyword scores for these ${batch.length} posts:`,
   478→        String(err),
   479→      )
   480→    }
   481→  }
   482→
   483→  return scores
   484→}

   487→export async function GET(request: NextRequest) {
   488→  try {
   489→    const { searchParams } = new URL(request.url)
   490→    const team = searchParams.get('team') || ''
   491→    const lang = searchParams.get('lang') || ''
   492→    const platform = searchParams.get('platform') || 'all'
   493→    const period = searchParams.get('period') || '24h'
   494→
   495→    const database = getDb()
   496→
   497→    // Build where clause for posts
   498→    const postWhere: Record<string, unknown> = {}
   499→    if (team) postWhere.teamTag = team
   500→    if (lang) postWhere.language = lang
   501→    if (platform !== 'all') postWhere.platform = platform
   502→
   503→    // Time filter based on period
   504→    const now = new Date()
   505→    if (period === '1h') {
   506→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 3600000) }
   507→    } else if (period === '24h') {
   508→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 86400000) }
   509→    } else if (period === '7d') {
   510→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 604800000) }
   511→    }
   512→
   513→    // Build where clause for summaries
   514→    const summaryWhere: Record<string, unknown> = {}
   515→    if (team) summaryWhere.teamCode = team
   516→    if (lang) summaryWhere.language = lang
   517→    if (platform !== 'all') summaryWhere.platform = platform
   518→    summaryWhere.period = period
   519→
   520→    // Fetch from database
   521→    const [posts, summaries] = await Promise.all([
   522→      database.socialPost.findMany({
   523→        where: postWhere,
   524→        orderBy: { fetchedAt: 'desc' },
   525→        take: 200,
   526→      }),
   527→      database.sentimentSummary.findMany({
   528→        where: summaryWhere,
   529→        orderBy: { updatedAt: 'desc' },
   530→      }),
   531→    ])
   532→
   533→    // Get distinct languages from the results
   534→    const languages = [...new Set(posts.map((p: any) => p.language))].sort()
   535→
   536→    // Check if we have in-memory cache for faster response
   537→    const cacheValid = cachedResponse && (Date.now() - lastFetchTime) < CACHE_DURATION
   538→
   539→    return NextResponse.json({
   540→      posts,
   541→      summaries,
   542→      languages: languages.length > 0 ? languages : [...SUPPORTED_LANGUAGES],
   543→      source: cacheValid ? 'cache' : 'database',
   544→      fetchedAt: cacheValid ? cachedResponse!.fetchedAt : new Date().toISOString(),
   545→    })
   546→  } catch (error) {
   547→    console.error('GET /api/social-sentiment error:', error)
   548→    return NextResponse.json(
   549→      { error: 'Failed to fetch social sentiment data', details: String(error) },
   550→      { status: 500 },
   551→    )
   552→  }
   553→}

   556→export async function POST(request: NextRequest) {
   557→  try {
   558→    const body = await request.json().catch(() => ({}))
   559→    const teamCode = body.teamCode || ''
   560→    const requestedLang = body.language || ''
   561→
   562→    // Determine which teams to fetch for
   563→    const teamsToFetch = teamCode
   564→      ? [teamCode]
   565→      : TOP_TEAMS
   566→
   567→    // Determine which languages to fetch
   568→    const langsToFetch: Language[] = requestedLang
   569→      ? ([requestedLang] as Language[])
   570→      : [...SUPPORTED_LANGUAGES]
   571→
   572→    // Check cache
   573→    const now = Date.now()
   574→    if (cachedResponse && (now - lastFetchTime) < CACHE_DURATION) {
   575→      return NextResponse.json({
   576→        source: 'cache',
   577→        ...cachedResponse,
   578→      })
   579→    }
   580→
   581→    const zai = await ZAI.create()
   582→    const database = getDb()
   583→    const allPosts: ParsedPost[] = []
   584→    const errors: string[] = []
   585→
   586→    // Fetch for each team and language combination
   587→    for (const team of teamsToFetch) {
   588→      const teamNames = TEAM_NAMES[team]
   589→      if (!teamNames) {
   590→        errors.push(`Unknown team code: ${team}`)
   591→        continue
   592→      }
   593→
   594→      for (const lang of langsToFetch) {
   595→        const teamNameInLang = teamNames[lang]
   596→        const queryTemplate = SEARCH_TEMPLATES[lang]
   597→
   598→        // ── Search 1: General multi-language search ───────────────────────
   599→        try {
   600→          const query = queryTemplate(teamNameInLang)
   601→          const searchResults = await zai.functions.invoke('web_search', {
   602→            query,
   603→            num: 5,
   604→          })
   605→
   606→          for (const result of (searchResults as any[]) || []) {
   607→            try {
   608→              const pageData = await zai.functions.invoke('page_reader', {
   609→                url: result.url,
   610→              })
   611→              const html = pageData?.data?.html || pageData?.data?.content || (typeof pageData === 'string' ? pageData : '')
   612→              if (!html) continue
   613→
   614→              // Determine platform from URL
   615→              const isReddit = result.url?.includes('reddit.com')
   616→              const platform: 'twitter' | 'reddit' = isReddit ? 'reddit' : 'twitter'
   617→
   618→              const parsed = parseSearchResults(html, platform, lang, team, query)
   619→              allPosts.push(...parsed)
   620→            } catch (pageErr) {
   621→              errors.push(`page_reader failed for ${result.url}: ${String(pageErr)}`)
   622→            }
   623→          }
   624→        } catch (searchErr) {
   625→          errors.push(`web_search failed for team=${team} lang=${lang}: ${String(searchErr)}`)
   626→        }
   627→
   628→        // ── Search 2: Reddit-specific search (English only) ──────────────
   629→        if (lang === 'en') {
   630→          try {
   631→            const redditQuery = `"World Cup 2026 ${teamNames.en}" site:reddit.com`
   632→            const redditResults = await zai.functions.invoke('web_search', {
   633→              query: redditQuery,
   634→              num: 5,
   635→            })
   636→
   637→            for (const result of (redditResults as any[]) || []) {
   638→              try {
   639→                const pageData = await zai.functions.invoke('page_reader', {
   640→                  url: result.url,
   641→                })
   642→                const html = pageData?.data?.html || pageData?.data?.content || (typeof pageData === 'string' ? pageData : '')
   643→                if (!html) continue
   644→
   645→                const parsed = parseSearchResults(html, 'reddit', 'en', team, redditQuery)
   646→                allPosts.push(...parsed)
   647→              } catch (pageErr) {
   648→                errors.push(`Reddit page_reader failed for ${result.url}: ${String(pageErr)}`)
   649→              }
   650→            }
   651→          } catch (searchErr) {
   652→            errors.push(`Reddit search failed for team=${team}: ${String(searchErr)}`)
   653→          }
   654→        }
   655→
   656→        // ── Search 3: X.com-specific search (English only) ───────────────
   657→        if (lang === 'en') {
   658→          try {
   659→            const twitterQuery = `"World Cup 2026 ${teamNames.en}" site:x.com`
   660→            const twitterResults = await zai.functions.invoke('web_search', {
   661→              query: twitterQuery,
   662→              num: 5,
   663→            })
   664→
   665→            for (const result of (twitterResults as any[]) || []) {
   666→              try {
   667→                const pageData = await zai.functions.invoke('page_reader', {
   668→                  url: result.url,
   669→                })
   670→                const html = pageData?.data?.html || pageData?.data?.content || (typeof pageData === 'string' ? pageData : '')
   671→                if (!html) continue
   672→
   673→                const parsed = parseSearchResults(html, 'twitter', 'en', team, twitterQuery)
   674→                allPosts.push(...parsed)
   675→              } catch (pageErr) {
   676→                errors.push(`X.com page_reader failed for ${result.url}: ${String(pageErr)}`)
   677→              }
   678→            }
   679→          } catch (searchErr) {
   680→            errors.push(`X.com search failed for team=${team}: ${String(searchErr)}`)
   681→          }
   682→        }
   683→
   684→        // ── Search 4: Hashtag-focused search (English only) ─────────────
   685→        // Uses the team's primary fan hashtag (e.g. #ThreeLions for England,
   686→        // #Selecao for Brazil) to surface high-traffic social conversations.
   687→        if (lang === 'en') {
   688→          try {
   689→            const primaryHashtag = TEAM_HASHTAGS[team]?.[0] || '#FWC26'
   690→            const hashtagQuery = `${primaryHashtag} ${teamNames.en} World Cup 2026`
   691→            const hashtagResults = await zai.functions.invoke('web_search', {
   692→              query: hashtagQuery,
   693→              num: 5,
   694→            })
   695→
   696→            for (const result of (hashtagResults as any[]) || []) {
   697→              try {
   698→                const pageData = await zai.functions.invoke('page_reader', {
   699→                  url: result.url,
   700→                })
   701→                const html = pageData?.data?.html || pageData?.data?.content || (typeof pageData === 'string' ? pageData : '')
   702→                if (!html) continue
   703→
   704→                const isReddit = result.url?.includes('reddit.com')
   705→                const platform: 'twitter' | 'reddit' = isReddit ? 'reddit' : 'twitter'
   706→
   707→                const parsed = parseSearchResults(html, platform, 'en', team, hashtagQuery)
   708→                allPosts.push(...parsed)
   709→              } catch (pageErr) {
   710→                errors.push(`Hashtag page_reader failed for ${result.url}: ${String(pageErr)}`)
   711→              }
   712→            }
   713→          } catch (searchErr) {
   714→            errors.push(`Hashtag search failed for team=${team}: ${String(searchErr)}`)
   715→          }
   716→        }
   717→      }
   718→    }
   719→
   720→    // ── Deduplicate posts by postId ────────────────────────────────────────
   721→    const uniquePosts = Array.from(
   722→      new Map(allPosts.map((p) => [p.postId, p])).values(),
   723→    )
   724→
   725→    // ── LLM-based sentiment scoring (overwrites keyword scores when available) ─
   726→    // Posts are initially assigned a keyword sentiment during parsing; here we
   727→    // upgrade them to LLM scores where the LLM succeeds. Failures fall back to
   728→    // the existing keyword score, so the pipeline never breaks.
   729→    let llmScored = 0
   730→    try {
   731→      const llmScores = await scoreSentimentBatchWithLLM(uniquePosts, zai)
   732→      for (const post of uniquePosts) {
   733→        const llmScore = llmScores.get(post.postId)
   734→        if (typeof llmScore === 'number') {
   735→          post.sentiment = llmScore
   736→          llmScored++
   737→        }
   738→      }
   739→      console.log(
   740→        `[social-sentiment] LLM scored ${llmScored}/${uniquePosts.length} posts (${uniquePosts.length - llmScored} used keyword fallback)`,
   741→      )
   742→    } catch (llmErr) {
   743→      errors.push(`LLM scoring failed entirely: ${String(llmErr)}`)
   744→    }
   745→
   746→    // ── Store posts in database ────────────────────────────────────────────
   747→    let storedCount = 0
   748→    for (const post of uniquePosts) {
   749→      try {
   750→        await database.socialPost.upsert({
   751→          where: {
   752→            platform_postId: {
   753→              platform: post.platform,
   754→              postId: post.postId,
   755→            },
   756→          },
   757→          create: {
   758→            platform: post.platform,
   759→            postId: post.postId,
   760→            author: post.author,
   761→            content: post.content,
   762→            language: post.language,
   763→            sentiment: post.sentiment,
   764→            likes: post.likes,
   765→            replies: post.replies,
   766→            shares: post.shares,
   767→            teamTag: post.teamTag,
   768→            searchQuery: post.searchQuery,
   769→            postedAt: post.postedAt,
   770→          },
   771→          update: {
   772→            author: post.author,
   773→            content: post.content,
   774→            sentiment: post.sentiment,
   775→            likes: post.likes,
   776→            replies: post.replies,
   777→            shares: post.shares,
   778→            searchQuery: post.searchQuery,
   779→            postedAt: post.postedAt,
   780→            updatedAt: new Date(),
   781→          },
   782→        })
   783→        storedCount++
   784→      } catch (dbErr) {
   785→        errors.push(`DB upsert failed for post ${post.postId}: ${String(dbErr)}`)
   786→      }
   787→    }
   788→
   789→    // ── Compute and store sentiment summaries ──────────────────────────────
   790→    const summaryPromises: Promise<unknown>[] = []
   791→
   792→    // Group posts by (teamCode, language, platform)
   793→    const grouped = new Map<string, ParsedPost[]>()
   794→    for (const post of uniquePosts) {
   795→      const key = `${post.teamTag}|${post.language}|${post.platform}`
   796→      if (!grouped.has(key)) grouped.set(key, [])
   797→      grouped.get(key)!.push(post)
   798→    }
   799→
   800→    for (const [key, posts] of grouped) {
   801→      const [teamCodeStr, lang, platformStr] = key.split('|')
   802→      const avgSentiment = posts.reduce((sum, p) => sum + p.sentiment, 0) / posts.length
   803→      const positiveCount = posts.filter((p) => p.sentiment > 60).length
   804→      const positiveRatio = positiveCount / posts.length
   805→
   806→      // Collect all topics from posts
   807→      const allTopics: string[] = []
   808→      for (const p of posts) {
   809→        allTopics.push(...extractTopics(p.content))
   810→      }
   811→      const topTopics = [...new Set(allTopics)].slice(0, 10)
   812→
   813→      // Create summary for specific platform
   814→      summaryPromises.push(
   815→        database.sentimentSummary.upsert({
   816→          where: {
   817→            teamCode_language_platform_period: {
   818→              teamCode: teamCodeStr,
   819→              language: lang,
   820→              platform: platformStr,
   821→              period: '24h',
   822→            },
   823→          },
   824→          create: {
   825→            teamCode: teamCodeStr,
   826→            language: lang,
   827→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   828→            postCount: posts.length,
   829→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   830→            topTopics: JSON.stringify(topTopics),
   831→            platform: platformStr,
   832→            period: '24h',
   833→          },
   834→          update: {
   835→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   836→            postCount: posts.length,
   837→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   838→            topTopics: JSON.stringify(topTopics),
   839→            updatedAt: new Date(),
   840→          },
   841→        }),
   842→      )
   843→
   844→      // Also create an "all" platform summary
   845→      summaryPromises.push(
   846→        database.sentimentSummary.upsert({
   847→          where: {
   848→            teamCode_language_platform_period: {
   849→              teamCode: teamCodeStr,
   850→              language: lang,
   851→              platform: 'all',
   852→              period: '24h',
   853→            },
   854→          },
   855→          create: {
   856→            teamCode: teamCodeStr,
   857→            language: lang,
   858→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   859→            postCount: posts.length,
   860→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   861→            topTopics: JSON.stringify(topTopics),
   862→            platform: 'all',
   863→            period: '24h',
   864→          },
   865→          update: {
   866→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   867→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   868→            topTopics: JSON.stringify(topTopics),
   869→            updatedAt: new Date(),
   870→          },
   871→        }),
   872→      )
   873→    }
   874→
   875→    // Wait for all summary writes
   876→    const summaryResults = await Promise.allSettled(summaryPromises)
   877→    const summaryErrors = summaryResults
   878→      .filter((r) => r.status === 'rejected')
   879→      .map((r) => String((r as PromiseRejectedResult).reason))
   880→    errors.push(...summaryErrors)
   881→
   882→    // ── Build response ─────────────────────────────────────────────────────
   883→    const fetchedLanguages = [...new Set(uniquePosts.map((p) => p.language))].sort()
   884→
   885→    const response = {
   886→      posts: uniquePosts,
   887→      summaries: [],
   888→      languages: fetchedLanguages.length > 0 ? fetchedLanguages : [...SUPPORTED_LANGUAGES],
   889→      fetchedAt: new Date().toISOString(),
   890→      stats: {
   891→        totalPostsFound: allPosts.length,
   892→        uniquePostsStored: storedCount,
   893→        teamsFetched: teamsToFetch.length,
   894→        languagesFetched: langsToFetch.length,
   895→        summariesComputed: summaryPromises.length,
   896→        llmScored,
   897→        keywordScored: uniquePosts.length - llmScored,
   898→      },
   899→      errors: errors.length > 0 ? errors.slice(0, 20) : undefined,
   900→    }
   901→
   902→    // Update cache
   903→    cachedResponse = {
   904→      posts: uniquePosts,
   905→      summaries: [],
   906→      languages: fetchedLanguages.length > 0 ? fetchedLanguages : [...SUPPORTED_LANGUAGES],
   907→      fetchedAt: new Date().toISOString(),
   908→    }
   909→    lastFetchTime = now
   910→
   911→    return NextResponse.json(response)
   912→  } catch (error) {
   913→    console.error('POST /api/social-sentiment error:', error)
   914→    return NextResponse.json(
   915→      { error: 'Failed to fetch social sentiment data', details: String(error) },
   916→      { status: 500 },
   917→    )
   918→  }
   919→}
