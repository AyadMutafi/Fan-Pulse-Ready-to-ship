   112→const labelColor: Record<string, string> = {
   113→  positive: 'bg-emerald-500/15 text-emerald-300 border-emerald-500/30',
   114→  negative: 'bg-rose-500/15 text-rose-300 border-rose-500/30',
   115→  neutral: 'bg-zinc-500/15 text-zinc-300 border-zinc-500/30',
   116→  mixed: 'bg-amber-500/15 text-amber-300 border-amber-500/30',
   117→}
   118→
   119→function scoreColor(s: number): string {
   120→  if (s >= 75) return 'text-emerald-400'
   121→  if (s >= 60) return 'text-lime-400'
   122→  if (s >= 40) return 'text-amber-400'
   123→  if (s >= 20) return 'text-orange-400'
   124→  return 'text-rose-400'
   125→}
   126→
   127→function trendIcon(t: string) {
   128→  if (t === 'rising') return <TrendingUp className="h-3 w-3 text-emerald-400" />
   129→  if (t === 'falling') return <TrendingDown className="h-3 w-3 text-rose-400" />
   130→  return <Minus className="h-3 w-3 text-zinc-400" />
   131→}

   135→export default function AdminPage() {
   136→  const router = useRouter()
   137→  const [authed, setAuthed] = useState<boolean | null>(null)
   138→
   139→  // Session check on mount.
   140→  useEffect(() => {

   143→      .then((d) => setAuthed(Boolean(d.authed)))
   144→      .catch(() => setAuthed(false))
   145→  }, [])
   146→
   147→  if (authed === null) {
   148→    return (
   149→      <div className="flex min-h-screen items-center justify-center bg-zinc-950 text-zinc-300">
   150→        <Loader2 className="h-6 w-6 animate-spin" />
   151→      </div>
   152→    )
   153→  }
   154→
   155→  if (!authed) {
   156→    return <LoginGate onAuthed={() => setAuthed(true)} />
   157→  }
   158→
   159→  return <Dashboard onLogout={() => setAuthed(false)} router={router} />
   160→}

   162→// ── Login Gate ───────────────────────────────────────────────────────────────
   163→
   164→function LoginGate({ onAuthed }: { onAuthed: () => void }) {
   165→  const [password, setPassword] = useState('')
   166→  const [loading, setLoading] = useState(false)
   167→  const [error, setError] = useState<string | null>(null)
   168→
   169→  async function submit(e: React.FormEvent) {
   170→    e.preventDefault()
   171→    setError(null)
   172→    setLoading(true)
   173→    try {
   174→      const res = await fetch('/api/admin/login', {
   175→        method: 'POST',
   176→        headers: { 'content-type': 'application/json' },
   177→        body: JSON.stringify({ password }),
   178→      })
   179→      const data = await res.json()
   180→      if (!res.ok) {
   181→        setError(data.error ?? 'Login failed')
   182→        return
   183→      }
   184→      toast.success('Welcome back, admin')
   185→      onAuthed()
   186→    } catch {
   187→      setError('Network error — please retry')
   188→    } finally {
   189→      setLoading(false)
   190→    }
   191→  }
   192→
   193→  return (
   194→    <div className="flex min-h-screen items-center justify-center bg-zinc-950 px-4 py-10">
   195→      <Card className="w-full max-w-md border-zinc-800 bg-zinc-900/60 backdrop-blur">
   196→        <CardHeader className="space-y-2 text-center">
   197→          <div className="mx-auto flex h-12 w-12 items-center justify-center rounded-full bg-cyan-500/10 ring-1 ring-cyan-500/30">
   198→            <Lock className="h-6 w-6 text-cyan-400" />
   199→          </div>
   200→          <CardTitle className="text-zinc-100">Admin Access</CardTitle>
   201→          <CardDescription className="text-zinc-400">
   202→            Fan Pulse · Semi-auto rating pipeline
   203→          </CardDescription>
   204→        </CardHeader>
   205→        <CardContent>
   206→          <form onSubmit={submit} className="space-y-4">
   207→            <div className="space-y-2">
   208→              <Label htmlFor="pw" className="text-zinc-300">
   209→                Admin password
   210→              </Label>
   211→              <Input
   212→                id="pw"
   213→                type="password"
   214→                value={password}
   215→                onChange={(e) => setPassword(e.target.value)}
   216→                placeholder="••••••••••••"
   217→                autoFocus
   218→                disabled={loading}
   219→                className="border-zinc-700 bg-zinc-950 text-zinc-100 placeholder:text-zinc-600"
   220→              />
   221→            </div>
   222→            {error && (
   223→              <Alert variant="destructive">
   224→                <AlertTriangle className="h-4 w-4" />
   225→                <AlertDescription>{error}</AlertDescription>
   226→              </Alert>
   227→            )}
   228→            <Button
   229→              type="submit"
   230→              disabled={loading || !password}
   231→              className="w-full bg-cyan-600 hover:bg-cyan-500 text-white"
   232→            >
   233→              {loading ? (
   234→                <>
   235→                  <Loader2 className="h-4 w-4 animate-spin" /> Authenticating…
   236→                </>
   237→              ) : (
   238→                <>
   239→                  <ShieldCheck className="h-4 w-4" /> Enter dashboard
   240→                </>
   241→              )}
   242→            </Button>
   243→          </form>
   244→          <p className="mt-4 text-center text-xs text-zinc-500">
   245→            This page is intentionally unlinked from the public site.
   246→          </p>
   247→        </CardContent>
   248→      </Card>
   249→    </div>
   250→  )
   251→}

   253→// ── Dashboard ────────────────────────────────────────────────────────────────
   254→
   255→function Dashboard({
   256→  onLogout,
   257→  router,
   258→}: {
   259→  onLogout: () => void
   260→  router: ReturnType<typeof useRouter>
   261→}) {
   262→  const [players, setPlayers] = useState<PlayerLite[]>([])
   263→  const [selectedPlayerId, setSelectedPlayerId] = useState<string>('')
   264→  const [posts, setPosts] = useState<CuratedPost[]>([])
   265→  const [postsLoading, setPostsLoading] = useState(false)
   266→  const [aiResult, setAiResult] = useState<AiResult | null>(null)
   267→  const [aiLoading, setAiLoading] = useState(false)
   268→  const [aiError, setAiError] = useState<string | null>(null)
   269→
   270→  // Form state for pasting a tweet.
   271→  const [form, setForm] = useState({
   272→    text: '',
   273→    author: '',
   274→    sourceUrl: '',
   275→    postedAt: '',
   276→    likes: '',
   277→    replies: '',
   278→    shares: '',
   279→  })
   280→  const [adding, setAdding] = useState(false)
   281→
   282→  // Load the player list once.
   283→  useEffect(() => {
   284→    fetch('/api/social-posts?include=players', { cache: 'no-store' })
   285→      .then((r) => r.json())
   286→      .then((d) => {
   287→        setPlayers(d.players ?? [])
   288→      })
   289→      .catch(() => toast.error('Failed to load players'))
   290→  }, [])
   291→
   292→  // Load curated posts whenever the selected player changes.
   293→  const refreshPosts = useCallback(async (playerId: string) => {
   294→    if (!playerId) {
   295→      setPosts([])
   296→      return
   297→    }
   298→    setPostsLoading(true)
   299→    try {
   300→      const res = await fetch(
   301→        `/api/social-posts?playerId=${encodeURIComponent(playerId)}`,
   302→        { cache: 'no-store' },
   303→      )
   304→      const d = await res.json()
   305→      setPosts(d.posts ?? [])
   306→    } catch {
   307→      toast.error('Failed to load curated tweets')
   308→    } finally {
   309→      setPostsLoading(false)
   310→    }
   311→  }, [])
   312→
   313→  useEffect(() => {
   314→    setAiResult(null)
   315→    setAiError(null)
   316→    refreshPosts(selectedPlayerId)
   317→  }, [selectedPlayerId, refreshPosts])
   318→
   319→  async function handleAddTweet(e: React.FormEvent) {
   320→    e.preventDefault()
   321→    if (!selectedPlayerId) {
   322→      toast.error('Pick a player first')
   323→      return
   324→    }
   325→    if (!form.text.trim()) {
   326→      toast.error('Tweet text is required')
   327→      return
   328→    }
   329→    const player = players.find((p) => p.id === selectedPlayerId)
   330→    if (!player) return
   331→    setAdding(true)
   332→    try {
   333→      const res = await fetch('/api/social-posts', {
   334→        method: 'POST',
   335→        headers: { 'content-type': 'application/json' },
   336→        body: JSON.stringify({
   337→          text: form.text.trim(),
   338→          author: form.author.trim(),
   339→          sourceUrl: form.sourceUrl.trim(),
   340→          playerId: selectedPlayerId,
   341→          playerName: player.playerName,
   342→          nationCode: player.nationCode,
   343→          postedAt: form.postedAt || undefined,
   344→          likes: Number(form.likes) || 0,
   345→          replies: Number(form.replies) || 0,
   346→          shares: Number(form.shares) || 0,
   347→        }),
   348→      })
   349→      const d = await res.json()
   350→      if (!res.ok) {
   351→        toast.error(d.error ?? 'Failed to add tweet')
   352→        return
   353→      }
   354→      toast.success('Tweet added as evidence')
   355→      setForm({
   356→        text: '',
   357→        author: '',
   358→        sourceUrl: '',
   359→        postedAt: '',
   360→        likes: '',
   361→        replies: '',
   362→        shares: '',
   363→      })
   364→      refreshPosts(selectedPlayerId)
   365→    } catch {
   366→      toast.error('Network error')
   367→    } finally {
   368→      setAdding(false)
   369→    }
   370→  }
   371→
   372→  async function handleDelete(id: string) {
   373→    try {
   374→      const res = await fetch(`/api/social-posts/${id}`, { method: 'DELETE' })
   375→      if (!res.ok) {
   376→        const d = await res.json()
   377→        toast.error(d.error ?? 'Delete failed')
   378→        return
   379→      }
   380→      toast.success('Tweet removed')
   381→      setPosts((prev) => prev.filter((p) => p.id !== id))
   382→    } catch {
   383→      toast.error('Network error')
   384→    }
   385→  }
   386→
   387→  async function handleRate() {
   388→    if (!selectedPlayerId) return
   389→    setAiLoading(true)
   390→    setAiError(null)
   391→    setAiResult(null)
   392→    try {
   393→      const res = await fetch('/api/ai-rate-player', {
   394→        method: 'POST',
   395→        headers: { 'content-type': 'application/json' },
   396→        body: JSON.stringify({ playerId: selectedPlayerId }),
   397→      })
   398→      const d = await res.json()
   399→      if (!res.ok) {
   400→        setAiError(d.error ?? 'AI rating failed')
   401→        return
   402→      }
   403→      setAiResult(d as AiResult)
   404→      toast.success(`AI rating: ${d.ai.score} (${d.ai.label})`)
   405→      // Refresh the posts so per-post labels show up.
   406→      refreshPosts(selectedPlayerId)
   407→    } catch {
   408→      setAiError('Network error')
   409→    } finally {
   410→      setAiLoading(false)
   411→    }
   412→  }
   413→
   414→  async function handleLogout() {
   415→    await fetch('/api/admin/logout', { method: 'POST' })
   416→    toast.success('Signed out')
   417→    onLogout()
   418→    router.refresh()
   419→  }
   420→
   421→  const selectedPlayer = players.find((p) => p.id === selectedPlayerId)
   422→
   423→  return (
   424→    <div className="min-h-screen bg-zinc-950 text-zinc-100">
   425→      {/* Header */}
   426→      <header className="sticky top-0 z-10 border-b border-zinc-800 bg-zinc-950/80 backdrop-blur">
   427→        <div className="mx-auto flex max-w-6xl items-center justify-between px-4 py-3">
   428→          <div className="flex items-center gap-2">
   429→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-cyan-500/15 ring-1 ring-cyan-500/30">
   430→              <Sparkles className="h-4 w-4 text-cyan-400" />
   431→            </div>
   432→            <div>
   433→              <h1 className="text-sm font-semibold">Fan Pulse · Admin</h1>
   434→              <p className="text-[11px] text-zinc-500">
   435→                Semi-auto rating pipeline
   436→              </p>
   437→            </div>
   438→          </div>
   439→          <Button
   440→            variant="ghost"
   441→            size="sm"
   442→            onClick={handleLogout}
   443→            className="text-zinc-400 hover:text-zinc-200"
   444→          >
   445→            <LogOut className="mr-1 h-4 w-4" /> Sign out
   446→          </Button>
   447→        </div>
   448→      </header>
   449→
   450→      <main className="mx-auto max-w-6xl space-y-6 px-4 py-6">
   451→        {/* Pipeline explainer */}
   452→        <Alert className="border-cyan-500/30 bg-cyan-500/5 text-cyan-100">
   453→          <Sparkles className="h-4 w-4 text-cyan-400" />
   454→          <AlertTitle className="text-cyan-200">How the pipeline works</AlertTitle>
   455→          <AlertDescription className="text-cyan-100/80">
   456→            1. Pick a player → 2. Paste tweets as evidence → 3. Click{' '}
   457→            <strong>Rate with AI</strong> → 4. AI derives a social score (0-100)
   458→            from the tweets → 5. Score flows into the player&apos;s Pulse
   459→            Breakdown (fan sentiment component) → 6. Front-end Sentiments &amp;
   460→            Pulse Elite update live. AI suggests, admin approves.
   461→          </AlertDescription>
   462→        </Alert>
   463→
   464→        {/* Player selector */}
   465→        <Card className="border-zinc-800 bg-zinc-900/40">
   466→          <CardHeader>
   467→            <CardTitle className="text-base">1 · Select a player</CardTitle>
   468→            <CardDescription className="text-zinc-400">
   469→              Choose which player to curate social evidence for.
   470→            </CardDescription>
   471→          </CardHeader>
   472→          <CardContent>
   473→            <Select value={selectedPlayerId} onValueChange={setSelectedPlayerId}>
   474→              <SelectTrigger className="w-full border-zinc-700 bg-zinc-950">
   475→                <SelectValue placeholder="Search a player…" />
   476→              </SelectTrigger>
   477→              <SelectContent className="max-h-80 border-zinc-700 bg-zinc-900">
   478→                {players.map((p) => (
   479→                  <SelectItem key={p.id} value={p.id}>
   480→                    <span className="font-medium">{p.playerName}</span>
   481→                    <span className="ml-2 text-xs text-zinc-500">
   482→                      {p.nationCode} · {p.position} · {p.pulseScore}
   483→                    </span>
   484→                    <span className="ml-2 inline-flex">{trendIcon(p.trend)}</span>
   485→                  </SelectItem>
   486→                ))}
   487→              </SelectContent>
   488→            </Select>
   489→            {selectedPlayer && (
   490→              <div className="mt-3 flex flex-wrap items-center gap-2 text-xs text-zinc-400">
   491→                <Badge variant="outline" className="border-zinc-700 text-zinc-300">
   492→                  {selectedPlayer.nationCode}
   493→                </Badge>
   494→                <Badge variant="outline" className="border-zinc-700 text-zinc-300">
   495→                  {selectedPlayer.position}
   496→                </Badge>
   497→                <Badge variant="outline" className="border-zinc-700 text-zinc-300">
   498→                  Pulse {selectedPlayer.pulseScore}
   499→                </Badge>
   500→                {selectedPlayer.matchInfo && (
   501→                  <span className="text-zinc-500">
   502→                    Last match: {selectedPlayer.matchInfo}
   503→                  </span>
   504→                )}
   505→              </div>
   506→            )}
   507→          </CardContent>
   508→        </Card>
   509→
   510→        {/* Tweet paste form */}
   511→        <Card className="border-zinc-800 bg-zinc-900/40">
   512→          <CardHeader>
   513→            <CardTitle className="text-base">2 · Paste tweet as evidence</CardTitle>
   514→            <CardDescription className="text-zinc-400">
   515→              Copy a tweet&apos;s text, author handle, and URL. Engagement metrics
   516→              (likes/replies/shares) help the AI weight viral tweets.
   517→            </CardDescription>
   518→          </CardHeader>
   519→          <CardContent>
   520→            <form onSubmit={handleAddTweet} className="space-y-4">
   521→              <div className="space-y-2">
   522→                <Label htmlFor="text" className="text-zinc-300">
   523→                  Tweet text <span className="text-rose-400">*</span>
   524→                </Label>
   525→                <Textarea
   526→                  id="text"
   527→                  value={form.text}
   528→                  onChange={(e) => setForm({ ...form, text: e.target.value })}
   529→                  placeholder="Paste the full tweet text here…"
   530→                  rows={4}
   531→                  disabled={!selectedPlayerId || adding}
   532→                  className="border-zinc-700 bg-zinc-950 text-zinc-100 placeholder:text-zinc-600"
   533→                />
   534→              </div>
   535→              <div className="grid gap-3 sm:grid-cols-2">
   536→                <div className="space-y-2">
   537→                  <Label htmlFor="author" className="text-zinc-300">
   538→                    Author handle
   539→                  </Label>
   540→                  <Input
   541→                    id="author"
   542→                    value={form.author}
   543→                    onChange={(e) => setForm({ ...form, author: e.target.value })}
   544→                    placeholder="FabrizioRomano"
   545→                    disabled={!selectedPlayerId || adding}
   546→                    className="border-zinc-700 bg-zinc-950 text-zinc-100 placeholder:text-zinc-600"
   547→                  />
   548→                </div>
   549→                <div className="space-y-2">
   550→                  <Label htmlFor="url" className="text-zinc-300">
   551→                    Tweet URL
   552→                  </Label>
   553→                  <Input

   557→                      setForm({ ...form, sourceUrl: e.target.value })
