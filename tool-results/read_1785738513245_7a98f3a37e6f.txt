    22→// ── Admin Curation Page ──────────────────────────────────────────────────────
    23→// Founder-facing UI for the "70% manual curation" approach. The founder
    24→// pastes real tweet/post URLs for a specific match; the AI reads + scores
    25→// ONLY those URLs (never invents content).
    26→//
    27→// SECURITY: Cookie-based admin auth (same pattern as /admin/feed-monitor).
    28→// The password is POSTed to /api/admin/login, which sets an HttpOnly +
    29→// Secure + SameSite=Strict cookie. JS cannot read it → XSS-proof.
    30→
    31→interface Match {
    32→  id: string
    33→  homeTeam: { code: string; name: string; flag: string }
    34→  awayTeam: { code: string; name: string; flag: string }
    35→  score: string
    36→  status: string
    37→  league: string
    38→  group: string
    39→  matchDate: string
    40→}
    41→

    43→  url: string
    44→  status: 'added' | 'skipped' | 'error'
    45→  reason?: string
    46→  linkId?: string
    47→  author?: string
    48→  sentimentScore?: number
    49→  sentimentLabel?: string
    50→}
    51→

    82→export default function CurateAdminPage() {
    83→  const [password, setPassword] = useState('')
    84→  const [authed, setAuthed] = useState(false)
    85→  const [checkingAuth, setCheckingAuth] = useState(true)
    86→  const [authError, setAuthError] = useState('')
    87→
    88→  // On mount, check if a valid cookie session already exists.
    89→  // /api/admin/session returns { authed: boolean } with HTTP 200 either way
    90→  // (it doesn't leak whether the password is set). We check the body.
    91→  useEffect(() => {
    92→    let cancelled = false
    93→    fetch('/api/admin/session')
    94→      .then((res) => res.json())
    95→      .then((data) => {
    96→        if (cancelled) return
    97→        setAuthed(!!data?.authed)
    98→      })
    99→      .catch(() => {
   100→        if (!cancelled) setAuthed(false)
   101→      })
   102→      .finally(() => {
   103→        if (!cancelled) setCheckingAuth(false)
   104→      })
   105→    return () => { cancelled = true }
   106→  }, [])
   107→
   108→  const handleAuth = () => {
   109→    if (!password.trim()) {
   110→      setAuthError('Password is required')
   111→      return
   112→    }
   113→    fetch('/api/admin/login', {
   114→      method: 'POST',
   115→      headers: { 'Content-Type': 'application/json' },
   116→      body: JSON.stringify({ password }),
   117→    })
   118→      .then((res) => {
   119→        if (res.ok) {
   120→          setAuthed(true)
   121→          setAuthError('')
   122→          setPassword('')
   123→        } else {
   124→          setAuthError('Invalid password')
   125→          setAuthed(false)
   126→        }
   127→      })
   128→      .catch(() => {
   129→        setAuthError('Network error — try again')
   130→      })
   131→  }
   132→
   133→  const handleLogout = () => {
   134→    fetch('/api/admin/logout', { method: 'POST' })
   135→      .catch(() => {})
   136→      .finally(() => {
   137→        setAuthed(false)
   138→        setPassword('')
   139→      })
   140→  }
   141→
   142→  if (checkingAuth) {
   143→    return (
   144→      <div className="min-h-screen flex items-center justify-center bg-[#0A0A0A]">
   145→        <div className="animate-spin w-8 h-8 rounded-full border-2 border-white/10 border-t-[#6C2BD9]" />
   146→      </div>
   147→    )
   148→  }
   149→
   150→  if (!authed) {
   151→    return (
   152→      <div className="min-h-screen flex items-center justify-center bg-[#0A0A0A] p-4">
   153→        <motion.div
   154→          initial={{ opacity: 0, y: 20 }}
   155→          animate={{ opacity: 1, y: 0 }}
   156→          className="w-full max-w-md bg-[#1A1A1A] border border-white/10 rounded-2xl p-8 shadow-2xl"
   157→        >
   158→          <div className="flex items-center gap-3 mb-6">
   159→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center text-white font-black text-xl">
   160→              F
   161→            </div>
   162→            <div>
   163→              <h1 className="text-white font-bold text-lg">Fan Pulse Admin</h1>
   164→              <p className="text-white/50 text-xs">Curation Studio</p>
   165→            </div>
   166→          </div>
   167→          <label className="block text-xs font-bold text-white/70 mb-2 uppercase tracking-wider">
   168→            Admin Password
   169→          </label>
   170→          <input
   171→            type="password"
   172→            value={password}
   173→            onChange={(e) => setPassword(e.target.value)}
   174→            onKeyDown={(e) => e.key === 'Enter' && handleAuth()}
   175→            placeholder="Enter admin password"
   176→            className="w-full bg-[#0A0A0A] border border-white/10 rounded-lg px-4 py-3 text-white text-sm focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30 transition-all"
   177→            autoFocus
   178→          />
   179→          {authError && (
   180→            <p className="mt-2 text-xs text-[#EF4444] flex items-center gap-1.5">
   181→              <AlertCircle className="size-3.5" /> {authError}
   182→            </p>
   183→          )}
   184→          <button
   185→            onClick={handleAuth}
   186→            className="mt-4 w-full bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white font-bold py-3 rounded-lg hover:opacity-90 transition-opacity flex items-center justify-center gap-2 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   187→          >
   188→            <Lock className="size-4" /> Unlock Admin
   189→          </button>
   190→          <p className="mt-4 text-xs text-white/40 text-center">
   191→            Admin password required. Set ADMIN_PASSWORD in your environment.
   192→          </p>
   193→        </motion.div>
   194→      </div>
   195→    )
   196→  }
   197→
   198→  return <CurationDashboard onLogout={handleLogout} />
   199→}

   201→// ── Curation Dashboard (shown when authed) ───────────────────────────────────
   202→
   203→function CurationDashboard({ onLogout }: { onLogout: () => void }) {
   204→  const [matches, setMatches] = useState<Match[]>([])
   205→  const [selectedMatchId, setSelectedMatchId] = useState<string>('')
   206→  const [useManualMatch, setUseManualMatch] = useState(false)
   207→  const [manualHome, setManualHome] = useState('')
   208→  const [manualAway, setManualAway] = useState('')
   209→  const [manualDate, setManualDate] = useState('')
   210→  const [manualLabel, setManualLabel] = useState('')
   211→  const [urls, setUrls] = useState('')
   212→  const [hashtags, setHashtags] = useState('')
   213→  const [submitting, setSubmitting] = useState(false)
   214→  const [result, setResult] = useState<{
   215→    added: number
   216→    skipped: number
   217→    total: number
   218→    results: CurationResult[]
   219→    errors: string[]
   220→  } | null>(null)
   221→  const [recentLinks, setRecentLinks] = useState<CuratedLinkRow[]>([])
   222→  const [loadingRecent, setLoadingRecent] = useState(true)
   223→  const [toast, setToast] = useState<{ msg: string; type: 'success' | 'error' } | null>(null)
   224→
   225→  // Fetch recent matches for the dropdown
   226→  useEffect(() => {
   227→    fetch('/api/matches?limit=20')
   228→      .then((res) => res.json())
   229→      .then((data) => {
   230→        if (data.matches) setMatches(data.matches)
   231→      })
   232→      .catch((err) => console.error('Failed to fetch matches:', err))
   233→  }, [])
   234→
   235→  const fetchRecentLinks = useCallback(async () => {
   236→    setLoadingRecent(true)
   237→    try {
   238→      // Fetch all recent curated links (admin view — no matchId filter)
   239→      const res = await fetch('/api/curate/recent?limit=20')
   240→      if (res.ok) {
   241→        const data = await res.json()
   242→        setRecentLinks(data.links || [])
   243→      } else {
   244→        setRecentLinks([])
   245→      }
   246→    } catch (err) {
   247→      console.error('Failed to fetch recent curated links:', err)
   248→      setRecentLinks([])
   249→    } finally {
   250→      setLoadingRecent(false)
   251→    }
   252→  }, [])
   253→
   254→  useEffect(() => {
   255→    fetchRecentLinks()
   256→  }, [fetchRecentLinks])
   257→
   258→  const showToast = (msg: string, type: 'success' | 'error') => {
   259→    setToast({ msg, type })
   260→    setTimeout(() => setToast(null), 4000)
   261→  }
   262→
   263→  const handleSubmit = async () => {
   264→    // Validate inputs
   265→    const urlList = urls
   266→      .split('\n')
   267→      .map((u) => u.trim())
   268→      .filter(Boolean)
   269→
   270→    if (urlList.length === 0) {
   271→      showToast('Paste at least one URL', 'error')
   272→      return
   273→    }
   274→
   275→    let matchId: string | null = null
   276→    let matchLabel: string
   277→
   278→    if (useManualMatch) {
   279→      if (!manualHome.trim() || !manualAway.trim()) {
   280→        showToast('Enter both team names for manual match', 'error')
   281→        return
   282→      }
   283→      matchLabel = manualLabel.trim() ||
   284→        `${manualHome.trim()} vs ${manualAway.trim()}${manualDate ? ` — ${manualDate}` : ''}`
   285→    } else {
   286→      if (!selectedMatchId) {
   287→        showToast('Select a match or switch to manual entry', 'error')
   288→        return
   289→      }
   290→      const m = matches.find((x) => x.id === selectedMatchId)
   291→      if (!m) {
   292→        showToast('Selected match not found', 'error')
   293→        return
   294→      }
   295→      matchId = m.id
   296→      const dateStr = m.matchDate ? new Date(m.matchDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) : ''
   297→      matchLabel = `${m.homeTeam.name} vs ${m.awayTeam.name}${dateStr ? ` — ${dateStr}` : ''} (${m.league}${m.group ? ` ${m.group}` : ''})`
   298→    }
   299→
   300→    const hashtagList = hashtags
   301→      .split(',')
   302→      .map((h) => h.trim())
   303→      .filter(Boolean)
   304→
   305→    setSubmitting(true)
   306→    setResult(null)
   307→    try {
   308→      const res = await fetch('/api/curate', {
   309→        method: 'POST',
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify({
   312→          matchId,
   313→          matchLabel,
   314→          urls: urlList,
   315→          hashtags: hashtagList,
   316→        }),
   317→      })
   318→
   319→      const data = await res.json()
   320→
   321→      if (res.ok) {
   322→        setResult(data)
   323→        showToast(`Curated ${data.added} links, skipped ${data.skipped}`, 'success')
   324→        // Clear the URL field on success
   325→        if (data.added > 0) {
   326→          setUrls('')
   327→          fetchRecentLinks()
   328→        }
   329→      } else if (res.status === 429) {
   330→        showToast(`Rate limited — wait ${data.retryAfter || 60}s`, 'error')
   331→      } else {
   332→        showToast(data.error || 'Curation failed', 'error')
   333→      }
   334→    } catch (err) {
   335→      showToast('Network error — try again', 'error')
   336→      console.error(err)
   337→    } finally {
   338→      setSubmitting(false)
   339→    }
   340→  }
   341→
   342→  const handleDelete = async (id: string) => {
   343→    if (!confirm('Remove this curated link? It will be hidden from Fan Talk.')) return
   344→    try {
   345→      const res = await fetch(`/api/curate/recent?id=${encodeURIComponent(id)}`, {
   346→        method: 'DELETE',
   347→      })
   348→      if (res.ok) {
   349→        setRecentLinks((prev) => prev.filter((l) => l.id !== id))
   350→        showToast('Link removed', 'success')
   351→      } else {
   352→        showToast('Failed to remove link', 'error')
   353→      }
   354→    } catch {
   355→      showToast('Network error', 'error')
   356→    }
   357→  }
   358→
   359→  return (
   360→    <div className="min-h-screen bg-[#0A0A0A] text-white">
   361→      {/* Header */}
   362→      <header className="sticky top-0 z-20 border-b border-white/10 bg-[#0A0A0A]/95 backdrop-blur">
   363→        <div className="max-w-5xl mx-auto px-4 py-3 flex items-center justify-between">
   364→          <div className="flex items-center gap-3">
   365→            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center text-white font-black text-lg">
   366→              F
   367→            </div>
   368→            <div>
   369→              <h1 className="font-bold text-sm sm:text-base">Curation Studio</h1>
   370→              <p className="text-white/40 text-xs">Paste real URLs → AI scores them</p>
   371→            </div>
   372→          </div>
   373→          <button
   374→            onClick={onLogout}
   375→            className="flex items-center gap-1.5 text-xs text-white/60 hover:text-white px-3 py-1.5 rounded-lg hover:bg-white/5 transition-colors"
   376→          >
   377→            <LogOut className="size-3.5" /> Logout
   378→          </button>
   379→        </div>
   380→      </header>
   381→
   382→      <main className="max-w-5xl mx-auto px-4 py-6 pb-24 space-y-6">
   383→        {/* ── Curation Form ── */}
   384→        <section className="bg-[#1A1A1A] border border-white/10 rounded-2xl p-5 sm:p-6">
   385→          <div className="flex items-center gap-2 mb-4">
   386→            <Sparkles className="size-5 text-[#8B5CF6]" />
   387→            <h2 className="font-bold text-base">Curate & Analyze</h2>
   388→          </div>
   389→
   390→          {/* Match selector */}
   391→          <div className="space-y-3 mb-4">
   392→            <label className="block text-xs font-bold text-white/60 uppercase tracking-wider">
   393→              Match
   394→            </label>
   395→            {!useManualMatch ? (
   396→              <div className="flex gap-2">
   397→                <select
   398→                  value={selectedMatchId}
   399→                  onChange={(e) => setSelectedMatchId(e.target.value)}
   400→                  className="flex-1 bg-[#0A0A0A] border border-white/10 rounded-lg px-3 py-2.5 text-sm focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30"
   401→                >
   402→                  <option value="">Select a recent match…</option>
   403→                  {matches.map((m) => (
   404→                    <option key={m.id} value={m.id}>
   405→                      {m.homeTeam.name} vs {m.awayTeam.name} — {m.score} ({m.league}
   406→                      {m.group ? ` ${m.group}` : ''})
   407→                    </option>
   408→                  ))}
   409→                </select>
   410→                <button
   411→                  onClick={() => setUseManualMatch(true)}
   412→                  className="flex items-center gap-1.5 px-3 py-2.5 rounded-lg bg-white/5 border border-white/10 hover:bg-white/10 transition-colors text-xs font-bold whitespace-nowrap"
   413→                >
   414→                  <Plus className="size-3.5" /> New

   416→              </div>
   417→            ) : (
   418→              <div className="space-y-3 p-4 bg-[#0A0A0A] border border-white/10 rounded-lg">
   419→                <div className="grid grid-cols-2 gap-3">
   420→                  <div>
   421→                    <label className="block text-xs text-white/50 mb-1">Home team</label>
   422→                    <input
   423→                      value={manualHome}
   424→                      onChange={(e) => setManualHome(e.target.value)}
   425→                      placeholder="Arsenal"
   426→                      className="w-full bg-[#1A1A1A] border border-white/10 rounded-lg px-3 py-2 text-sm focus:outline-none focus:border-[#6C2BD9]"
   427→                    />
   428→                  </div>
   429→                  <div>
   430→                    <label className="block text-xs text-white/50 mb-1">Away team</label>
   431→                    <input
   432→                      value={manualAway}
   433→                      onChange={(e) => setManualAway(e.target.value)}
   434→                      placeholder="Chelsea"
   435→                      className="w-full bg-[#1A1A1A] border border-white/10 rounded-lg px-3 py-2 text-sm focus:outline-none focus:border-[#6C2BD9]"
   436→                    />
   437→                  </div>
   438→                </div>
   439→                <div className="grid grid-cols-2 gap-3">
   440→                  <div>
   441→                    <label className="block text-xs text-white/50 mb-1">Date (optional)</label>
   442→                    <input
   443→                      value={manualDate}
   444→                      onChange={(e) => setManualDate(e.target.value)}
   445→                      placeholder="Jul 28"
   446→                      className="w-full bg-[#1A1A1A] border border-white/10 rounded-lg px-3 py-2 text-sm focus:outline-none focus:border-[#6C2BD9]"
   447→                    />
   448→                  </div>
   449→                  <div>
   450→                    <label className="block text-xs text-white/50 mb-1">Label (optional)</label>
   451→                    <input
   452→                      value={manualLabel}
   453→                      onChange={(e) => setManualLabel(e.target.value)}
   454→                      placeholder="Arsenal vs Chelsea — Friendly"
   455→                      className="w-full bg-[#1A1A1A] border border-white/10 rounded-lg px-3 py-2 text-sm focus:outline-none focus:border-[#6C2BD9]"
   456→                    />
   457→                  </div>
   458→                </div>
   459→                <button
   460→                  onClick={() => setUseManualMatch(false)}
   461→                  className="text-xs text-[#8B5CF6] hover:text-[#a78bfa] font-bold"

   463→                  ← Use existing match instead

   465→              </div>
   466→            )}
   467→          </div>
   468→
   469→          {/* URL paste area */}
   470→          <div className="space-y-2 mb-4">
   471→            <label className="block text-xs font-bold text-white/60 uppercase tracking-wider flex items-center gap-1.5">
   472→              <LinkIcon className="size-3.5" /> URLs (one per line)
   473→            </label>
   474→            <textarea
   475→              value={urls}
   476→              onChange={(e) => setUrls(e.target.value)}
   477→              placeholder={'https://x.com/FabrizioRomano/status/...\nhttps://www.reddit.com/r/soccer/comments/...\nhttps://www.bbc.com/sport/football/...'}
   478→              rows={6}
   479→              className="w-full bg-[#0A0A0A] border border-white/10 rounded-lg px-3 py-2.5 text-sm font-mono focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30 resize-y"
   480→            />

   482→              Accepted: x.com, twitter.com, reddit.com, instagram.com, facebook.com, tiktok.com, ESPN, BBC, Sky Sports, The Athletic, Guardian, Goal, Al Jazeera, Reuters

   486→          {/* Hashtags */}

   489→              <Hash className="size-3.5" /> Hashtags (comma-separated, optional)
   490→            </label>

   495→              className="w-full bg-[#0A0A0A] border border-white/10 rounded-lg px-3 py-2.5 text-sm focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30"
   496→            />
   497→          </div>
   498→
   499→          {/* Submit */}
   500→          <button
   501→            onClick={handleSubmit}
   502→            disabled={submitting}
   503→            className="w-full bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white font-bold py-3 rounded-lg hover:opacity-90 transition-opacity flex items-center justify-center gap-2 disabled:opacity-50 disabled:cursor-not-allowed focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   504→          >
   505→            {submitting ? (
   506→              <>
   507→                <Loader2 className="size-4 animate-spin" /> Analyzing…
   508→              </>
   509→            ) : (
   510→              <>
   511→                <Sparkles className="size-4" /> Curate & Analyze
   512→              </>
   513→            )}
   514→          </button>
   515→        </section>
   516→
   517→        {/* ── Results ── */}
   518→        <AnimatePresence>
   519→          {result && (
   520→            <motion.section
   521→              initial={{ opacity: 0, height: 0 }}
