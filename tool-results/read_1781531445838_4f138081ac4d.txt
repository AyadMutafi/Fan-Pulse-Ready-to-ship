   153→// ── Helpers ──────────────────────────────────────────────────

   155→function getFlag(nationCode: string): string {
   156→  const team = findNationalTeam(nationCode)
   157→  return team?.flag ?? '🏳️'
   158→}
   159→
   160→function getTrendIcon(trend: string) {
   161→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   162→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   163→  return <Minus className="size-3 text-[#FF6B35]" />
   164→}
   165→
   166→function getSentimentColor(score: number) {
   167→  if (score >= 80) return 'text-[#10B981]'
   168→  if (score >= 50) return 'text-[#FF6B35]'
   169→  return 'text-[#EF4444]'
   170→}
   171→
   172→function getSentimentBg(score: number) {
   173→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   174→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   175→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   176→}
   177→
   178→function getProgressClass(score: number) {
   179→  if (score >= 80) return 'progress-emerald'
   180→  if (score >= 50) return 'progress-amber'
   181→  return 'progress-red'
   182→}
   183→
   184→// Face emoji mapping for /10 rating system
   185→function getPulseFaceEmoji(pulseScore: number): string {
   186→  if (pulseScore >= 90) return '🤩'
   187→  if (pulseScore >= 70) return '😊'
   188→  if (pulseScore >= 50) return '😐'
   189→  if (pulseScore >= 30) return '😟'
   190→  return '😵'
   191→}
   192→
   193→function getRatingColor(rating: number): string {
   194→  if (rating >= 9) return '#10B981'
   195→  if (rating >= 7) return '#6C2BD9'
   196→  if (rating >= 5) return '#FF6B35'
   197→  if (rating >= 3) return '#EF4444'
   198→  return '#DC2626'
   199→}
