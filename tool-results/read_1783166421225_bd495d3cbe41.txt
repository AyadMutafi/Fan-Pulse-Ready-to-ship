   241→// ── Seed handler ─────────────────────────────────────────────────────────────
   242→// AUTH REQUIRED: this route wipes and re-creates most tables. Only the admin
   243→// may call it. The frontend no longer calls this route; it runs once at deploy.
   244→export async function POST(request: Request) {
   245→  // ── Auth gate ──
   246→  if (!isAdminAuthorized(request)) {
   247→    return unauthorizedResponse()
   248→  }
   249→
   250→  try {
   251→    // ── Check if already seeded (skip if ?force=true) ──
   252→    const { searchParams } = new URL(request.url)
   253→    const force = searchParams.get('force') === 'true'
   254→    if (!force) {
   255→      const existingStages = await db.wCStage.count()
   256→      if (existingStages > 0) {
   257→        return NextResponse.json({ message: 'Already seeded', stages: existingStages })
   258→      }
   259→    }
   260→
   261→    // ── 1. Clean up existing data ──
   262→    // NOTE: fanVote is intentionally NOT wiped — user votes are the product and
   263→    // must survive re-seeds. pulseBreakdown + sentimentSummary are safe to wipe
   264→    // because they are fully recomputed by the pulse engine below.
   265→    await db.wCSelectionPlayer.deleteMany()
   266→    await db.wCSelection.deleteMany()
   267→    await db.wCStage.deleteMany()
   268→    await db.match.deleteMany()
   269→    await db.nationalTeam.deleteMany()
   270→    await db.pulseBreakdown.deleteMany().catch((e) => {
   271→      console.warn('pulseBreakdown deleteMany failed (continuing):', e)
   272→    })
   273→    await db.sentimentSummary.deleteMany().catch((e) => {
   274→      console.warn('sentimentSummary deleteMany failed (continuing):', e)
   275→    })
   276→    // Intentionally NOT deleting fanVote — user votes are preserved.
   277→
   278→    // ── 2. Create 7 stages — Group Stage + R32 completed, R16 upcoming ──
   279→    // NOTE: R32 was finalized 2026-07-03 after all 16 matches were played +
   280→    // synced via /api/world-cup/r32-results-sync + /api/world-cup/r32-finalize.
   281→    // For a fresh seed, R32 is marked completed with the verified scores below;
   282→    // the R32 Elite/Crisis XI is re-seeded (locked=true) by the seed step 7b.
   283→    const stagesData = [
   284→      { name: 'Group Stage', nameAr: 'دور المجموعات', order: 1, status: 'completed', key: 'group-stage', startedAt: new Date('2026-06-11'), completedAt: new Date('2026-06-27') },
   285→      { name: 'Round of 32', nameAr: 'دور الـ 32', order: 2, status: 'completed', key: 'round-of-32', startedAt: new Date('2026-06-28'), completedAt: new Date('2026-07-03') },
   286→      { name: 'Round of 16', nameAr: 'دور الـ 16', order: 3, status: 'live', key: 'round-of-16', startedAt: new Date('2026-07-04'), completedAt: undefined as Date | undefined },
   287→      { name: 'Quarter Finals', nameAr: 'ربع النهائي', order: 4, status: 'upcoming', key: 'quarter-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   288→      { name: 'Semi Finals', nameAr: 'نصف النهائي', order: 5, status: 'upcoming', key: 'semi-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   289→      { name: 'Third Place', nameAr: 'مركز الثالث', order: 6, status: 'upcoming', key: 'third-place', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   290→      { name: 'Final', nameAr: 'النهائي', order: 7, status: 'upcoming', key: 'final', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   291→    ]
   292→
   293→    const stages: Array<{ id: string; name: string; order: number; status: string; key: string }> = []
   294→    for (const sd of stagesData) {
   295→      const stage = await db.wCStage.create({
   296→        data: {
   297→          name: sd.name,
   298→          nameAr: sd.nameAr,
   299→          order: sd.order,
   300→          status: sd.status,
   301→          startedAt: sd.startedAt,

   348→          status: m.status,

   435→    // (finalized 2026-07-03) — selections are locked=true (historical, like Group
   436→    // Stage). No live cron refresh for R32 anymore; that system moves to R16.
   437→    // See src/lib/r32-buzz-ranker.ts for the anti-hallucination contract.

   439→    let r32Seeded: { elite: number; crisis: number; buzzSource: string } | null = null

   452→      console.error('R32 seed failed (continuing with group stage only):', err)
   453→    }
   454→
   455→    // ── 8. Seed R16 Elite XI + Crisis XI from the VERIFIED_POOL baseline ──────
   456→    // R16 is the LIVE stage (matches Jul 4-7, 2026). Seeded from the R16
   457→    // VERIFIED_POOL baseline (45 web-verified players covering all 8 R16
   458→    // matchups, captured 2026-07-04). locked=false so the live cron refresh

   478→    return NextResponse.json({

   488→    })
   489→  } catch (error) {
   490→    console.error('Seed failed:', error)
   491→    return NextResponse.json({ error: 'Seed failed', details: String(error) }, { status: 500 })
   492→  }
   493→}
