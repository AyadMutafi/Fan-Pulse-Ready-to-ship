    28→// ── World Cup stage label helper ─────────────────────────────
    29→// Renders an accurate label for a WC match based on its `group` value.
    30→// Group-stage matches use letters A-L; knockout rounds use R32/R16/QF/SF/Final.
    31→// (Previously everything was labelled "WC Group {x}", which produced the
    32→// nonsensical "WC Group R32" for knockout matches.)
    33→function wcStageLabel(group: string | null | undefined): string {
    34→  if (!group) return 'World Cup 2026'
    35→  if (/^[A-L]$/.test(group)) return `WC Group ${group}`
    36→  switch (group) {
    37→    case 'R32': return 'WC Round of 32'
    38→    case 'R16': return 'WC Round of 16'
    39→    case 'QF': return 'WC Quarter Finals'
    40→    case 'SF': return 'WC Semi Finals'
    41→    case 'Final': return 'WC Final'
    42→    default: return 'World Cup 2026'
    43→  }
    44→}

   236→function HomeTab() {
   237→  const { t } = useLanguage()
   238→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   239→  const [apiMatches, setApiMatches] = useState<Array<{
   240→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   241→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   242→    status: string; group: string; matchDate: string
   243→  }>>([])
   244→
   245→  // Fan vote state
   246→  const [sessionId, setSessionId] = useState<string>('')
   247→  const [fanVotes, setFanVotes] = useState<FanVoteAgg[]>([])
   248→  const [myVotes, setMyVotes] = useState<Array<{ teamCode: string; score: number }>>([])
   249→  const [votesLoading, setVotesLoading] = useState(true)
   250→  const [selectedVoteTeam, setSelectedVoteTeam] = useState<string | null>(null)
   251→  const [submitting, setSubmitting] = useState(false)
   252→  const [toast, setToast] = useState<{ msg: string; emoji: string } | null>(null)
   253→  const [fanCardOffer, setFanCardOffer] = useState<{ teamCode: string; score: number } | null>(null)
   254→
   255→  useEffect(() => {
   256→    async function fetchMatches() {
   257→      try {
   258→        const res = await fetch('/api/matches?league=WC')
   259→        if (res.ok) {
   260→          const data = await res.json()
   261→          const mapped = (data.matches || []).map((m: any) => ({
   262→            id: m.id,
   263→            home: m.homeTeam.code,
   264→            away: m.awayTeam.code,
   265→            homeFlag: m.homeTeam.flag || '',
   266→            awayFlag: m.awayTeam.flag || '',
   267→            score: m.score,
   268→            homeSentiment: Math.round(m.homeTeam.sentiment),
   269→            awaySentiment: Math.round(m.awayTeam.sentiment),
   270→            live: m.status === 'live',
   271→            league: m.league === 'WC' ? wcStageLabel(m.group) : m.league,
   272→            status: m.status || 'upcoming',
   273→            group: m.group || '',
   274→            matchDate: m.matchDate || '',
   275→          }))
   276→          setApiMatches(mapped)
   277→        }
   278→      } catch (err) {
   279→        console.error('Failed to fetch matches:', err)
   280→      }
   281→    }
   282→    fetchMatches()
   283→  }, [])
   284→
   285→  // SSR-safe: read/create sessionId inside useEffect, never during render.
   286→  useEffect(() => {
   287→    try {
   288→      const existing = typeof window !== 'undefined' ? window.localStorage.getItem('fan_session_id') : null
   289→      if (existing && existing.length > 0) {
   290→        setSessionId(existing)
   291→        return
   292→      }
   293→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
   294→        ? crypto.randomUUID()
   295→        : `anon-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`
   296→      if (typeof window !== 'undefined') {
   297→        window.localStorage.setItem('fan_session_id', newId)
   298→      }
   299→      setSessionId(newId)
   300→    } catch {
   301→      setSessionId(`anon-${Date.now()}`)
   302→    }
   303→  }, [])
   304→
   305→  // Fetch fan votes whenever sessionId changes.
   306→  useEffect(() => {
   307→    if (!sessionId) return
   308→    let cancelled = false
   309→    async function loadVotes() {
   310→      setVotesLoading(true)
   311→      try {
   312→        const res = await fetch(`/api/fan-vote?session=${encodeURIComponent(sessionId)}`)
   313→        if (!res.ok) return
   314→        const data = await res.json()
   315→        if (cancelled) return
   316→        setFanVotes(Array.isArray(data.votes) ? data.votes : [])
   317→        setMyVotes(Array.isArray(data.myVotes) ? data.myVotes : [])
   318→      } catch (err) {
   319→        console.error('Failed to fetch fan votes:', err)
   320→      } finally {
   321→        if (!cancelled) setVotesLoading(false)
   322→      }
   323→    }
   324→    loadVotes()
   325→    return () => { cancelled = true }
   326→  }, [sessionId])
   327→
   328→  // Close voting modal on Escape key (accessibility — modal was only closable
   329→  // via the explicit Close button or selecting a mood)
   330→  useEffect(() => {
   331→    if (!selectedVoteTeam) return
   332→    const onKey = (e: KeyboardEvent) => {
   333→      if (e.key === 'Escape' && !submitting) setSelectedVoteTeam(null)
   334→    }
   335→    window.addEventListener('keydown', onKey)
   336→    return () => window.removeEventListener('keydown', onKey)
   337→  }, [selectedVoteTeam, submitting])
   338→
   339→  const totalVoteCount = fanVotes.reduce((sum, v) => sum + (v.count || 0), 0)
   340→
   341→  const handleVote = async (teamCode: string, score: number) => {
   342→    if (!sessionId || submitting) return
   343→    setSubmitting(true)
   344→    // Optimistic update: bump my vote immediately
   345→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   346→    setMyVotes(prev => {
   347→      const without = prev.filter(v => v.teamCode !== teamCode)
   348→      return [...without, { teamCode, score }]
   349→    })
   350→    setFanVotes(prev => {
   351→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   352→      if (idx === -1) {
   353→        return [...prev, { teamCode, score, count: 1 }]
   354→      }
   355→      const next = [...prev]
   356→      const wasMine = prevMyVote !== null
   357→      // Adjust running average: if I had voted, swap my old score for the new.
   358→      const current = next[idx]
   359→      const totalScore = current.score * current.count
   360→      const newScore = wasMine
   361→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   362→        : Math.round((totalScore + score) / (current.count + 1))
   363→      const newCount = wasMine ? current.count : current.count + 1
   364→      next[idx] = { teamCode, score: newScore, count: newCount }
   365→      return next
   366→    })
   367→    setSelectedVoteTeam(null)
   368→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   369→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   370→    setTimeout(() => setToast(null), 2500)
   371→    // Show the Fan Card offer for 8 seconds — long enough to click "Get Fan Card"
   372→    setFanCardOffer({ teamCode, score })
   373→    const offerTimer = setTimeout(() => setFanCardOffer(null), 8000)
   374→    try {
   375→      await fetch('/api/fan-vote', {
   376→        method: 'POST',
   377→        headers: { 'Content-Type': 'application/json' },
   378→        body: JSON.stringify({ teamCode, score, sessionId }),
   379→      })
   380→    } catch (err) {
   381→      console.error('Failed to submit vote:', err)
   382→      // Revert on failure
   383→      setMyVotes(prev => {
   384→        const without = prev.filter(v => v.teamCode !== teamCode)
   385→        if (prevMyVote !== null) return [...without, { teamCode, score: prevMyVote }]
   386→        return without
   387→      })
   388→      setToast({ msg: 'Vote failed — please retry', emoji: '⚠️' })
   389→      setFanCardOffer(null)
   390→      clearTimeout(offerTimer)
   391→    } finally {
   392→      setSubmitting(false)
   393→    }
   394→  }
   395→
   396→  const filteredMatches = (matchFilter === 'ALL'
   397→    ? apiMatches
   398→    : apiMatches.filter(m => m.league.startsWith('WC'))
   399→  ).slice(0, 24) // Limit to 24 cards max for performance
   400→
   401→  const fanVoteIntelText = totalVoteCount === 0
   402→    ? 'Be the first to vote in the Fan Mood section below'
   403→    : `${totalVoteCount.toLocaleString()} fan votes tallied for World Cup 2026 Group Stage`
   404→
   405→  // ── Arena Intelligence: VERIFIED insights only ──────────────────────────
   406→  // Every insight below is a historical fact sourced from VERIFIED_DATA.md
   407→  // (the project's single source of truth, cross-checked against Wikipedia,
   408→  // ESPN, Olympics.com, FIFA.com on 2026-07-02). We deliberately do NOT
   409→  // auto-generate insights from apiMatches because that approach produced two
   410→  // verifiable hallucinations in production:
   411→  //   (1) "World Cup 2026 kicked off with ESP 0-0 AUT in the opener" — FALSE.
   412→  //       The old code used parsed[0] (first array element) as "the opener",
   413→  //       but the first row happened to be an UPCOMING R32 match (ESP vs AUT,
   414→  //       scheduled Jul 3, not yet played). The real opener was MEX 2-0 RSA
   415→  //       on Jun 11 (the earliest matchDate among completed matches).
   416→  //   (2) "Shock in Group Stage: ESP 0-0 AUT" — FALSE. The old code used
   417→  //       league.includes('WC') to label the stage as "Group Stage", but ALL
   418→  //       WC matches (including R32) match that check. ESP vs AUT is an R32
   419→  //       match, not a group-stage match. The real group-stage shock was
   420→  //       ESP 0-0 CPV (Spain held scoreless by Cape Verde).
   421→  //
   422→  // Fix: hardcode the verified insight set with explicit VERIFIED_DATA.md
   423→  // citations. Only the fan-vote count stays dynamic (it is live data).
   424→  // This guarantees we never describe an upcoming match as played, never
   425→  // mislabel a knockout match as a group-stage match, and never invent stats.
   426→  const arenaIntel = useMemo<Array<{ icon: typeof Sparkles; text: string; color: string; stage: string }>>(() => {
   427→    const items: Array<{ icon: typeof Sparkles; text: string; color: string; stage: string }> = []
   428→
   429→    // 1. Tournament opener — VERIFIED_DATA.md Part 1, Group A, match 1

  1248→function RateTab() {
  1249→  const { t } = useLanguage()
  1250→  const [players, setPlayers] = useState<Array<{
  1251→    id: string
  1252→    playerName: string
  1253→    nationCode: string
  1254→    position: string
  1255→    avgRating: number
  1256→    totalRatings: number
  1257→  }>>([])
  1258→  const [myRatings, setMyRatings] = useState<Record<string, number>>({})
  1259→  const [loading, setLoading] = useState(true)
  1260→  const [submittingId, setSubmittingId] = useState<string | null>(null)
  1261→
  1262→  // ── Load real players from the DB + this session's ratings ──
  1263→  useEffect(() => {
  1264→    let cancelled = false
  1265→    async function load() {
  1266→      setLoading(true)
  1267→      try {
  1268→        const res = await fetch('/api/ratings')
  1269→        if (!res.ok) return
  1270→        const data = await res.json()
  1271→        if (cancelled) return
  1272→        const list = Array.isArray(data.ratings) ? data.ratings : []
  1273→        setPlayers(list)
  1274→      } catch {
  1275→        /* ignore — toast shown on submit failure */
  1276→      } finally {
  1277→        if (!cancelled) setLoading(false)
  1278→      }
  1279→    }
  1280→    load()
  1281→    return () => { cancelled = true }
  1282→  }, [])
  1283→
  1284→  // ── Per-browser session ID (crypto.randomUUID, persisted in localStorage) ──
  1285→  // Required by the API — the old 'anonymous' default is rejected. A real
  1286→  // session ID makes the (sessionId, playerId) unique constraint meaningful,
  1287→  // enforcing one rating per session per player.
  1288→  const [sessionId, setSessionId] = useState<string>('')
  1289→  useEffect(() => {
  1290→    try {
  1291→      const existing = window.localStorage.getItem('fp_session_id')
  1292→      if (existing && existing.length >= 8) {
  1293→        setSessionId(existing)
  1294→        return
  1295→      }
  1296→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
  1297→        ? crypto.randomUUID()
  1298→        : `fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`
  1299→      window.localStorage.setItem('fp_session_id', newId)
  1300→      setSessionId(newId)
  1301→    } catch {
  1302→      setSessionId(`fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`)
  1303→    }
  1304→  }, [])
  1305→
  1306→  const handleRate = async (playerId: string, rating: number) => {
  1307→    if (!sessionId || submittingId === playerId) return
  1308→    setSubmittingId(playerId)
  1309→    // Optimistically update local state so the stars fill instantly.
  1310→    setMyRatings(prev => ({ ...prev, [playerId]: rating }))
  1311→    try {
  1312→      const res = await fetch('/api/ratings', {
  1313→        method: 'POST',
  1314→        headers: { 'Content-Type': 'application/json' },
  1315→        body: JSON.stringify({ playerId, rating, sessionId }),
  1316→      })
  1317→      const data = await res.json().catch(() => null)
  1318→      if (!res.ok) {
  1319→        // Revert on failure.
  1320→        setMyRatings(prev => {
