   114→function getTrendIcon(trend: string) {

   120→function getSentimentColor(score: number) {

   212→interface FanVoteAgg {
   213→  teamCode: string
   214→  score: number
   215→  count: number

   217→
   218→function HomeTab() {
   219→  const { t } = useLanguage()
   220→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   221→  const [apiMatches, setApiMatches] = useState<Array<{
   222→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   223→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   224→  }>>([])
   225→
   226→  // Fan vote state
   227→  const [sessionId, setSessionId] = useState<string>('')
   228→  const [fanVotes, setFanVotes] = useState<FanVoteAgg[]>([])
   229→  const [myVotes, setMyVotes] = useState<Array<{ teamCode: string; score: number }>>([])
   230→  const [votesLoading, setVotesLoading] = useState(true)
   231→  const [selectedVoteTeam, setSelectedVoteTeam] = useState<string | null>(null)
   232→  const [submitting, setSubmitting] = useState(false)
   233→  const [toast, setToast] = useState<{ msg: string; emoji: string } | null>(null)
   234→
   235→  useEffect(() => {
   236→    async function fetchMatches() {
   237→      try {
   238→        const res = await fetch('/api/matches?league=WC')
   239→        if (res.ok) {
   240→          const data = await res.json()
   241→          const mapped = (data.matches || []).map((m: any) => ({
   242→            id: m.id,
   243→            home: m.homeTeam.code,
   244→            away: m.awayTeam.code,
   245→            homeFlag: m.homeTeam.flag || '',
   246→            awayFlag: m.awayTeam.flag || '',
   247→            score: m.score,
   248→            homeSentiment: Math.round(m.homeTeam.sentiment),
   249→            awaySentiment: Math.round(m.awayTeam.sentiment),
   250→            live: m.status === 'live',
   251→            league: m.league === 'WC' ? `WC Group ${m.group}` : m.league,
   252→          }))
   253→          setApiMatches(mapped)
   254→        }
   255→      } catch (err) {
   256→        console.error('Failed to fetch matches:', err)
   257→      }
   258→    }
   259→    fetchMatches()
   260→  }, [])
   261→
   262→  // SSR-safe: read/create sessionId inside useEffect, never during render.
   263→  useEffect(() => {
   264→    try {
   265→      const existing = typeof window !== 'undefined' ? window.localStorage.getItem('fan_session_id') : null
   266→      if (existing && existing.length > 0) {
   267→        setSessionId(existing)
   268→        return
   269→      }
   270→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
   271→        ? crypto.randomUUID()
   272→        : `anon-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`
   273→      if (typeof window !== 'undefined') {
   274→        window.localStorage.setItem('fan_session_id', newId)
   275→      }
   276→      setSessionId(newId)
   277→    } catch {
   278→      setSessionId(`anon-${Date.now()}`)
   279→    }
   280→  }, [])
   281→
   282→  // Fetch fan votes whenever sessionId changes.
   283→  useEffect(() => {
   284→    if (!sessionId) return
   285→    let cancelled = false
   286→    async function loadVotes() {
   287→      setVotesLoading(true)
   288→      try {
   289→        const res = await fetch(`/api/fan-vote?session=${encodeURIComponent(sessionId)}`)
   290→        if (!res.ok) return
   291→        const data = await res.json()
   292→        if (cancelled) return
   293→        setFanVotes(Array.isArray(data.votes) ? data.votes : [])
   294→        setMyVotes(Array.isArray(data.myVotes) ? data.myVotes : [])
   295→      } catch (err) {
   296→        console.error('Failed to fetch fan votes:', err)
   297→      } finally {
   298→        if (!cancelled) setVotesLoading(false)
   299→      }
   300→    }
   301→    loadVotes()
   302→    return () => { cancelled = true }
   303→  }, [sessionId])
   304→
   305→  const totalVoteCount = fanVotes.reduce((sum, v) => sum + (v.count || 0), 0)
   306→
   307→  const handleVote = async (teamCode: string, score: number) => {
   308→    if (!sessionId || submitting) return
   309→    setSubmitting(true)
   310→    // Optimistic update: bump my vote immediately
   311→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   312→    setMyVotes(prev => {
   313→      const without = prev.filter(v => v.teamCode !== teamCode)
   314→      return [...without, { teamCode, score }]
   315→    })
   316→    setFanVotes(prev => {
   317→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   318→      if (idx === -1) {
   319→        return [...prev, { teamCode, score, count: 1 }]
   320→      }
   321→      const next = [...prev]
   322→      const wasMine = prevMyVote !== null
   323→      // Adjust running average: if I had voted, swap my old score for the new.
   324→      const current = next[idx]
   325→      const totalScore = current.score * current.count
   326→      const newScore = wasMine
   327→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   328→        : Math.round((totalScore + score) / (current.count + 1))
   329→      const newCount = wasMine ? current.count : current.count + 1
   330→      next[idx] = { teamCode, score: newScore, count: newCount }
   331→      return next
   332→    })
   333→    setSelectedVoteTeam(null)
   334→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   335→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   336→    setTimeout(() => setToast(null), 2500)
   337→    try {
   338→      await fetch('/api/fan-vote', {
   339→        method: 'POST',
   340→        headers: { 'Content-Type': 'application/json' },
   341→        body: JSON.stringify({ teamCode, score, sessionId }),
   342→      })
   343→    } catch (err) {
   344→      console.error('Failed to submit vote:', err)
   345→      // Revert on failure
   346→      setMyVotes(prev => {
   347→        const without = prev.filter(v => v.teamCode !== teamCode)
   348→        if (prevMyVote !== null) return [...without, { teamCode, score: prevMyVote }]
   349→        return without
   350→      })
   351→      setToast({ msg: 'Vote failed — please retry', emoji: '⚠️' })
   352→    } finally {
   353→      setSubmitting(false)
   354→    }
   355→  }
   356→
   357→  const filteredMatches = (matchFilter === 'ALL'
   358→    ? apiMatches
   359→    : apiMatches.filter(m => m.league.startsWith('WC'))
   360→  ).slice(0, 24) // Limit to 24 cards max for performance
   361→
   362→  const fanVoteIntelText = totalVoteCount === 0
   363→    ? 'Be the first to vote in the Fan Mood section below'
   364→    : `${totalVoteCount.toLocaleString()} fan votes tallied for World Cup 2026 Group Stage`
   365→
   366→  const moodTeamEntries = FAN_MOOD_TEAM_CODES.map(code => {
   367→    const team = NATIONAL_TEAMS.find(t => t.code === code)
   368→    const vote = fanVotes.find(v => v.teamCode === code)
   369→    const myVote = myVotes.find(v => v.teamCode === code)
   370→    return {
   371→      code,
   372→      flag: team?.flag ?? '🏳️',
   373→      name: team?.name ?? code,
   374→      score: vote?.score ?? 50,
   375→      count: vote?.count ?? 0,
   376→      myVote: myVote?.score ?? null,
   377→    }
   378→  })
   379→
   380→  return (
   381→    <div className="space-y-6">
   382→      {/* Hero */}
   383→      <motion.div
   384→        initial={{ opacity: 0, y: 20 }}
   385→        animate={{ opacity: 1, y: 0 }}
   386→        transition={{ duration: 0.5 }}
   387→        className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-[#6C2BD9]/8 via-white dark:via-[#1A1A1A] to-[#FF6B35]/5 p-6 border border-[#6C2BD9]/10 dark:border-[#6C2BD9]/20"
   388→      >
   389→        <div className="absolute -right-8 -top-8 size-32 rounded-full bg-[#6C2BD9]/5 blur-3xl" />
   390→        <div className="absolute -bottom-8 -left-8 size-32 rounded-full bg-[#FF6B35]/5 blur-3xl" />
   391→        <h2 className="relative text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
   392→          {t('home.your_pulse')} <span className="text-[#6C2BD9]">⚡</span>
   393→        </h2>
   394→        <p className="relative mt-2 text-sm text-[#666] dark:text-[#CCCCCC]">
   395→          {t('home.mood_desc')}
   396→        </p>
   397→        <div className="relative mt-4 flex items-center gap-3">
   398→          <div className="flex items-center gap-1.5 rounded-full bg-[#10B981]/10 px-3 py-1.5 text-xs font-semibold text-[#10B981]">
   399→            <Zap className="size-3.5" />
   400→            78% {t('home.positive')}
   401→          </div>
   402→          <div className="flex items-center gap-1.5 rounded-full bg-[#EF4444]/10 px-3 py-1.5 text-xs font-semibold text-[#EF4444]">
   403→            <Activity className="size-3.5" />
   404→            {apiMatches.filter(m => m.live).length} {t('home.live')}
   405→          </div>
   406→        </div>
   407→      </motion.div>
   408→
   409→      {/* Featured Matches */}
   410→      <div>
   411→        <div className="mb-3 flex items-center justify-between">
   412→          <h3 className="text-xs font-bold uppercase tracking-wider text-[#666] dark:text-[#CCCCCC]">
   413→            {t('home.featured')}
   414→          </h3>
   415→          <div className="flex gap-1.5">
   416→            {(['WC', 'ALL'] as const).map((filter) => (
   417→              <button
   418→                key={filter}
   419→                onClick={() => setMatchFilter(filter)}
   420→                className={`
   421→                  rounded-full px-3 py-1 text-[10px] font-bold transition-all duration-200
   422→                  ${matchFilter === filter
   423→                    ? 'bg-[#6C2BD9] text-white shadow-sm'
   424→                    : 'bg-[#F8F9FA] dark:bg-[#2D2D2D] text-[#666] dark:text-[#CCCCCC] border border-[#E0E0E0] dark:border-white/10'
   425→                  }
   426→                `}
   427→              >
   428→                {filter === 'ALL' ? '⚽ All' : '🏆 World Cup'}
