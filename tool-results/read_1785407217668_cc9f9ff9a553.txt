    33→// ── World Cup stage label helper ─────────────────────────────
    34→// Renders an accurate label for a WC match based on its `group` value.
    35→// Group-stage matches use letters A-L; knockout rounds use R32/R16/QF/SF/Final.
    36→// (Previously everything was labelled "WC Group {x}", which produced the
    37→// nonsensical "WC Group R32" for knockout matches.)
    38→function wcStageLabel(group: string | null | undefined): string {
    39→  if (!group) return 'World Cup 2026'
    40→  if (/^[A-L]$/.test(group)) return `WC Group ${group}`
    41→  switch (group) {
    42→    case 'R32': return 'WC Round of 32'
    43→    case 'R16': return 'WC Round of 16'
    44→    case 'QF': return 'WC Quarter Finals'
    45→    case 'SF': return 'WC Semi Finals'
    46→    case 'Final': return 'WC Final'
    47→    default: return 'World Cup 2026'
    48→  }
    49→}

   627→  function formatRelativeTime(iso: string | null): string {
   628→    if (!iso) return 'recently'
   629→    const then = new Date(iso).getTime()
   630→    if (Number.isNaN(then)) return 'recently'
   631→    const diffMs = Date.now() - then
   632→    const mins = Math.floor(diffMs / 60000)
   633→    if (mins < 1) return 'just now'
   634→    if (mins < 60) return `${mins}m ago`
   635→    const hrs = Math.floor(mins / 60)
   636→    if (hrs < 24) return `${hrs}h ago`
   637→    const days = Math.floor(hrs / 24)
   638→    if (days < 7) return `${days}d ago`
   639→    const weeks = Math.floor(days / 7)
   640→    return `${weeks}w ago`
   641→  }

  1621→function RateTab() {
  1622→  const { t } = useLanguage()
  1623→  const [players, setPlayers] = useState<Array<{
  1624→    id: string
  1625→    playerName: string
  1626→    nationCode: string
  1627→    position: string
  1628→    avgRating: number
  1629→    totalRatings: number
  1630→  }>>([])
  1631→  const [myRatings, setMyRatings] = useState<Record<string, number>>({})
  1632→  const [loading, setLoading] = useState(true)
  1633→  const [submittingId, setSubmittingId] = useState<string | null>(null)
  1634→
  1635→  // ── Load real players from the DB + this session's ratings ──
  1636→  useEffect(() => {
  1637→    let cancelled = false
  1638→    async function load() {
  1639→      setLoading(true)
  1640→      try {
  1641→        const res = await fetch('/api/ratings')
  1642→        if (!res.ok) return
  1643→        const data = await res.json()
  1644→        if (cancelled) return
  1645→        const list = Array.isArray(data.ratings) ? data.ratings : []
  1646→        setPlayers(list)
  1647→      } catch {
  1648→        /* ignore — toast shown on submit failure */
  1649→      } finally {
  1650→        if (!cancelled) setLoading(false)
  1651→      }
  1652→    }
  1653→    load()
  1654→    return () => { cancelled = true }
  1655→  }, [])
  1656→
  1657→  // ── Per-browser session ID (crypto.randomUUID, persisted in localStorage) ──
  1658→  // Required by the API — the old 'anonymous' default is rejected. A real
  1659→  // session ID makes the (sessionId, playerId) unique constraint meaningful,
  1660→  // enforcing one rating per session per player.
  1661→  const [sessionId, setSessionId] = useState<string>('')
  1662→  useEffect(() => {
  1663→    try {
  1664→      const existing = window.localStorage.getItem('fp_session_id')
  1665→      if (existing && existing.length >= 8) {
  1666→        setSessionId(existing)
  1667→        return
  1668→      }
  1669→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
  1670→        ? crypto.randomUUID()
  1671→        : `fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`
  1672→      window.localStorage.setItem('fp_session_id', newId)
  1673→      setSessionId(newId)
  1674→    } catch {
  1675→      setSessionId(`fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`)
  1676→    }
  1677→  }, [])
  1678→
  1679→  const handleRate = async (playerId: string, rating: number) => {
  1680→    if (!sessionId || submittingId === playerId) return
  1681→    setSubmittingId(playerId)
  1682→    // Optimistically update local state so the stars fill instantly.
  1683→    setMyRatings(prev => ({ ...prev, [playerId]: rating }))
  1684→    try {
  1685→      const res = await fetch('/api/ratings', {
  1686→        method: 'POST',
  1687→        headers: { 'Content-Type': 'application/json' },
  1688→        body: JSON.stringify({ playerId, rating, sessionId }),
  1689→      })
  1690→      const data = await res.json().catch(() => null)
  1691→      if (!res.ok) {
  1692→        // Revert on failure.
  1693→        setMyRatings(prev => {
  1694→          const next = { ...prev }
  1695→          delete next[playerId]
  1696→          return next
  1697→        })
  1698→        if (res.status === 429) {
  1699→          toast.error('Too many ratings — please slow down.')
  1700→        } else {
  1701→          toast.error(data?.error || 'Failed to submit rating')
  1702→        }
  1703→        return
  1704→      }
  1705→      toast.success(data?.updated ? 'Rating updated!' : 'Rating submitted!', {
  1706→        description: `${rating}/10`,
  1707→      })
  1708→      // Refresh aggregates so the avg display updates.
  1709→      const fresh = await fetch('/api/ratings')
  1710→      if (fresh.ok) {
  1711→        const fd = await fresh.json().catch(() => null)
  1712→        if (Array.isArray(fd?.ratings)) setPlayers(fd.ratings)
  1713→      }
  1714→    } catch {
  1715→      setMyRatings(prev => {
  1716→        const next = { ...prev }
  1717→        delete next[playerId]
  1718→        return next
  1719→      })
  1720→      toast.error('Network error — please try again')
  1721→    } finally {
  1722→      setSubmittingId(null)
  1723→    }
  1724→  }
  1725→
  1726→  return (
  1727→    <div className="space-y-6">
  1728→      <motion.div
  1729→        initial={{ opacity: 0, y: 20 }}
  1730→        animate={{ opacity: 1, y: 0 }}
  1731→        className="space-y-1"
  1732→      >
  1733→        <h2 className="text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
  1734→          {t('ratings.title')}
  1735→        </h2>
  1736→        <p className="text-sm text-[#666] dark:text-[#CCCCCC]">{t('ratings.desc')}</p>
  1737→      </motion.div>
  1738→
  1739→      <div className="space-y-3">
  1740→        {loading ? (
  1741→          <Card className="border-[#E0E0E0]/50 dark:border-white/5">
  1742→            <CardContent className="p-6 text-center text-sm text-[#666] dark:text-[#CCCCCC]">
  1743→              Loading players…
  1744→            </CardContent>
  1745→          </Card>
  1746→        ) : players.length === 0 ? (
