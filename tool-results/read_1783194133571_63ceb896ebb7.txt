    44→export default function FeedMonitorAdminPage() {
    45→  // Persist auth in localStorage so refresh doesn't lose session.
    46→  // Use lazy initial state to avoid the setState-in-effect anti-pattern.
    47→  const [password, setPassword] = useState(() => {
    48→    if (typeof window === 'undefined') return ''
    49→    return localStorage.getItem('fp_admin_pw') || ''
    50→  })
    51→  const [authed, setAuthed] = useState(() => {
    52→    if (typeof window === 'undefined') return false
    53→    return !!localStorage.getItem('fp_admin_pw')
    54→  })
    55→  const [authError, setAuthError] = useState('')
    56→
    57→  const handleAuth = () => {
    58→    if (!password.trim()) {
    59→      setAuthError('Password is required')
    60→      return
    61→    }
    62→    // Test the password by hitting the GET endpoint
    63→    fetch('/api/admin/feed-monitor', {
    64→      headers: { 'x-admin-password': password },
    65→    })
    66→      .then((res) => {
    67→        if (res.ok) {
    68→          setAuthed(true)
    69→          setAuthError('')
    70→          if (typeof window !== 'undefined') {
    71→            localStorage.setItem('fp_admin_pw', password)
    72→          }
    73→        } else {
    74→          setAuthError('Invalid password')
    75→          setAuthed(false)
    76→        }
    77→      })
    78→      .catch(() => {
    79→        setAuthError('Network error — try again')
    80→      })
    81→  }
    82→
    83→  const handleLogout = () => {
    84→    setAuthed(false)
    85→    setPassword('')
    86→    if (typeof window !== 'undefined') {
    87→      localStorage.removeItem('fp_admin_pw')
    88→    }
    89→  }
    90→
    91→  if (!authed) {
    92→    return (
    93→      <div className="min-h-screen flex items-center justify-center bg-[#0A0A0A] p-4">
    94→        <motion.div
    95→          initial={{ opacity: 0, y: 20 }}
    96→          animate={{ opacity: 1, y: 0 }}
    97→          className="w-full max-w-md bg-[#1A1A1A] border border-white/10 rounded-2xl p-8 shadow-2xl"
    98→        >
    99→          <div className="flex items-center gap-3 mb-6">
   100→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center text-white font-black text-xl">
   101→              F
   102→            </div>
   103→            <div>
   104→              <h1 className="text-white font-bold text-lg">Fan Pulse Admin</h1>
   105→              <p className="text-white/50 text-xs">Feed Monitor Pipeline</p>
   106→            </div>
   107→          </div>
   108→          <label className="block text-xs font-bold text-white/70 mb-2 uppercase tracking-wider">
   109→            Admin Password
   110→          </label>
   111→          <input
   112→            type="password"
   113→            value={password}
   114→            onChange={(e) => setPassword(e.target.value)}
   115→            onKeyDown={(e) => e.key === 'Enter' && handleAuth()}
   116→            placeholder="Enter admin password"
   117→            className="w-full bg-[#0A0A0A] border border-white/10 rounded-lg px-4 py-3 text-white text-sm focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30 transition-all"
   118→            autoFocus
   119→          />
   120→          {authError && (
   121→            <p className="mt-2 text-xs text-[#EF4444] flex items-center gap-1.5">
   122→              <AlertCircle className="size-3.5" /> {authError}
   123→            </p>
   124→          )}
   125→          <button
   126→            onClick={handleAuth}
   127→            className="mt-4 w-full bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white font-bold py-3 rounded-lg hover:opacity-90 transition-opacity flex items-center justify-center gap-2"
   128→          >
   129→            <Lock className="size-4" /> Unlock Admin
   130→          </button>
   131→          <p className="mt-4 text-xs text-white/40 text-center">
   132→            Admin password required. Set ADMIN_PASSWORD in your environment.
   133→          </p>
   134→        </motion.div>
   135→      </div>
   136→    )
   137→  }
   138→
   139→  return <MonitorDashboard password={password} onLogout={handleLogout} />
   140→}

   144→function MonitorDashboard({ password, onLogout }: { password: string; onLogout: () => void }) {

   149→  const [refreshing, setRefreshing] = useState<string | null>(null)
   150→  const [toast, setToast] = useState<{ msg: string; type: 'success' | 'error' } | null>(null)
   151→
   152→  const fetchMonitors = useCallback(async () => {
   153→    try {
   154→      const res = await fetch('/api/admin/feed-monitor', {
   155→        headers: { 'x-admin-password': password },
   156→      })
   157→      if (res.ok) {
   158→        const data = await res.json()
   159→        setMonitors(data.monitors || [])
   160→      }
   161→    } catch (err) {
   162→      console.error('Failed to fetch monitors:', err)

   164→      setLoading(false)
   165→    }
   166→  }, [password])
   167→
   168→  useEffect(() => {
   169→    fetchMonitors()
   170→    // Auto-refresh the monitor list every 30s so admins see live updates
   171→    const interval = setInterval(fetchMonitors, 30000)
   172→    return () => clearInterval(interval)
   173→  }, [fetchMonitors])
   174→
   175→  const handleRefresh = async (id: string) => {
   176→    setRefreshing(id)

   194→      setToast({ msg: 'Network error', type: 'error' })

   228→        headers: { 'x-admin-password': password },

   231→        fetchMonitors()
   232→        setToast({ msg: 'Monitor deleted', type: 'success' })
   233→        setTimeout(() => setToast(null), 3000)

   246→            <div className="w-9 h-9 rounded-lg bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center font-black">
   247→              F
   248→            </div>
   249→            <div>
   250→              <h1 className="font-bold text-sm">Feed Monitor Pipeline</h1>

   370→function MonitorCard({
   371→  monitor,
   372→  password,
   373→  expanded,
   374→  onToggle,
   375→  onRefresh,
   376→  onStatusChange,
   377→  onDelete,
   378→  refreshing,
   379→}: {
   380→  monitor: Monitor
   381→  password: string
   382→  expanded: boolean
   383→  onToggle: () => void
   384→  onRefresh: () => void
   385→  onStatusChange: (status: 'active' | 'paused' | 'ended') => void
   386→  onDelete: () => void
   387→  refreshing: boolean
   388→}) {
   389→  const statusColors = {
   390→    active: 'bg-[#10B981]/10 text-[#10B981] border-[#10B981]/30',
   391→    paused: 'bg-[#FF6B35]/10 text-[#FF6B35] border-[#FF6B35]/30',

   393→  }
   394→

   396→    ? new Date(monitor.lastRefreshedAt).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
   397→    : 'never'
   398→
   399→  const endsAt = new Date(monitor.endsAt)
   400→  const timeLeft = endsAt.getTime() - Date.now()
   401→  const timeLeftLabel = timeLeft > 0 ? `${Math.floor(timeLeft / 3600000)}h ${Math.floor((timeLeft % 3600000) / 60000)}m left` : 'ended'

   405→      <div className="p-4 flex items-center gap-3">

   412→            <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full border uppercase ${statusColors[monitor.status]}`}>
   413→              {monitor.status}

   415→          </div>

   417→            <span className="flex items-center gap-1">

   421→              <MessageCircle className="size-3" /> {monitor.postCount} posts

   425→            </span>
   426→            <span className="flex items-center gap-1">
   427→              <Clock className="size-3" /> refreshed {lastRefreshed}
   428→            </span>
   429→            <span className="text-white/40">· {timeLeftLabel}</span>
   430→          </div>
   431→        </div>
   432→        <div className="flex items-center gap-1">
   433→          <button
   434→            onClick={onRefresh}
   435→            disabled={refreshing || monitor.status === 'ended'}
   436→            title="Manual refresh"
   437→            className="p-2 rounded-lg hover:bg-white/5 text-white/60 hover:text-white disabled:opacity-30 transition-colors"
   438→          >
   439→            <RefreshCw className={`size-4 ${refreshing ? 'animate-spin' : ''}`} />
   440→          </button>

   442→            <button
   443→              onClick={() => onStatusChange('paused')}
   444→              title="Pause"

   447→              <Pause className="size-4" />
   448→            </button>

   451→              onClick={() => onStatusChange('active')}

   453→              className="p-2 rounded-lg hover:bg-white/5 text-[#10B981] transition-colors"

   458→          <button

   461→            title="End monitor"
   462→            className="p-2 rounded-lg hover:bg-white/5 text-white/60 hover:text-white disabled:opacity-30 transition-colors"
   463→          >
   464→            <Square className="size-4" />
   465→          </button>
   466→          <button
   467→            onClick={onDelete}
   468→            title="Delete"
   469→            className="p-2 rounded-lg hover:bg-[#EF4444]/10 text-white/60 hover:text-[#EF4444] transition-colors"
   470→          >
   471→            <Trash2 className="size-4" />
   472→          </button>
   473→        </div>
   474→      </div>
   475→
   476→      <AnimatePresence>
   477→        {expanded && (
   478→          <motion.div
   479→            initial={{ height: 0, opacity: 0 }}
   480→            animate={{ height: 'auto', opacity: 1 }}
   481→            exit={{ height: 0, opacity: 0 }}
   482→            className="border-t border-white/10"
   483→          >

   486→        )}

   488→    </div>

   492→function MonitorDetail({ monitor, password }: { monitor: Monitor; password: string }) {
   493→  const [posts, setPosts] = useState<Post[]>([])
   494→  const [loadingPosts, setLoadingPosts] = useState(true)
   495→  const [postsOffset, setPostsOffset] = useState(0)
   496→
   497→  useEffect(() => {
   498→    let cancelled = false
   499→    // Use a ref-free cancelled flag instead of setState-in-effect.
   500→    // The loading state is set when the fetch starts (in .then() below),
   501→    // not synchronously at the top of the effect body.
   502→    fetch(`/api/admin/feed-monitor/${monitor.id}/posts?limit=20&offset=${postsOffset}`, {
   503→      headers: { 'x-admin-password': password },
   504→    })
   505→      .then((res) => res.json())
   506→      .then((data) => {
   507→        if (cancelled) return
   508→        setPosts(data.posts || [])
   509→        setLoadingPosts(false)
   510→      })
   511→      .catch(() => {
   512→        if (!cancelled) setLoadingPosts(false)
   513→      })
   514→    return () => { cancelled = true }
   515→  }, [monitor.id, password, postsOffset])
   516→
   517→  return (
   518→    <div className="p-4 space-y-4 bg-[#0F0F0F]">
   519→      {/* Hashtags + Seed URLs */}
   520→      <div className="grid md:grid-cols-2 gap-4">
   521→        <div>
   522→          <h4 className="text-xs font-bold text-white/60 uppercase tracking-wider mb-2 flex items-center gap-1.5">
   523→            <Hash className="size-3" /> Hashtags ({monitor.hashtags.length})
   524→          </h4>

   551→                </a>
   552→              ))}
   553→              {monitor.seedUrls.length > 3 && (
   554→                <p className="text-xs text-white/40">+ {monitor.seedUrls.length - 3} more</p>
   555→              )}
   556→            </div>
   557→          )}
   558→        </div>
   559→      </div>
   560→
   561→      {/* Posts */}
   562→      <div>
   563→        <h4 className="text-xs font-bold text-white/60 uppercase tracking-wider mb-2 flex items-center gap-1.5">
   564→          <MessageCircle className="size-3" /> Recent Posts ({monitor.postCount} total)
   565→        </h4>
   566→        {loadingPosts ? (
   567→          <div className="flex justify-center py-6">
   568→            <div className="animate-spin w-5 h-5 rounded-full border-2 border-white/10 border-t-[#6C2BD9]" />
   569→          </div>
   570→        ) : posts.length === 0 ? (
   571→          <div className="text-center py-6 text-xs text-white/40">

   573→          </div>
   574→        ) : (
   575→          <div className="space-y-2 max-h-96 overflow-y-auto pr-2">
   576→            {posts.map((post) => (
   577→              <PostRow key={post.id} post={post} />
   578→            ))}
   579→          </div>
   580→        )}
   581→      </div>
   582→    </div>
   583→  )

   586→function PostRow({ post }: { post: Post }) {
   587→  const [expanded, setExpanded] = useState(false)
   588→
   589→  const platformColors = {
   590→    twitter: 'bg-[#1DA1F2]/10 text-[#1DA1F2] border-[#1DA1F2]/30',
   591→    reddit: 'bg-[#FF4500]/10 text-[#FF4500] border-[#FF4500]/30',
   592→    web: 'bg-white/5 text-white/60 border-white/10',
   593→  }
   594→
   595→  const sentimentColor =
   596→    post.sentimentScore >= 70
   597→      ? 'text-[#10B981]'
   598→      : post.sentimentScore >= 45
   599→        ? 'text-[#FF6B35]'
   600→        : 'text-[#EF4444]'
   601→
   602→  return (
   603→    <div className="bg-[#1A1A1A] border border-white/5 rounded-lg p-3">
   604→      <div className="flex items-start justify-between gap-2 mb-1">
   605→        <div className="flex items-center gap-2 flex-wrap">
   606→          <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full border uppercase ${platformColors[post.platform as keyof typeof platformColors] || platformColors.web}`}>
   607→            {post.platform}
   608→          </span>
   609→          <span className="text-xs font-bold text-white/80">{post.author || 'unknown'}</span>
   610→          <span className="text-xs text-white/40">{new Date(post.postedAt).toLocaleDateString()}</span>
   611→        </div>
   612→        <div className="flex items-center gap-2 shrink-0">
   613→          <span className={`text-sm font-black ${sentimentColor}`}>{Math.round(post.sentimentScore)}</span>
   614→          <a
   615→            href={post.url}
   616→            target="_blank"
   617→            rel="noopener noreferrer"
   618→            className="text-white/40 hover:text-white transition-colors"
   619→          >
   620→            <ExternalLink className="size-3.5" />
   621→          </a>
   622→        </div>
   623→      </div>
   624→      <p className={`text-xs text-white/70 ${expanded ? '' : 'line-clamp-2'}`}>
   625→        {post.content}
   626→      </p>
   627→      {post.content.length > 200 && (
   628→        <button
   629→          onClick={() => setExpanded(!expanded)}
   630→          className="text-xs text-[#8B5CF6] hover:underline mt-1"
   631→        >
   632→          {expanded ? 'show less' : 'show more'}
   633→        </button>
   634→      )}
   635→      {post.topQuote && (
   636→        <div className="mt-2 px-3 py-2 bg-[#6C2BD9]/5 border-l-2 border-[#6C2BD9] rounded-r">
   637→          <p className="text-xs italic text-white/80">&ldquo;{post.topQuote}&rdquo;</p>
   638→        </div>
   639→      )}
   640→      {post.mentionedPlayers.length > 0 && (
   641→        <div className="mt-2 flex items-center gap-1 flex-wrap">
   642→          <span className="text-[10px] text-white/40">mentions:</span>
   643→          {post.mentionedPlayers.slice(0, 3).map((id, i) => (
   644→            <span key={i} className="text-[10px] text-[#FF6B35] font-mono">
   645→              {id.slice(-6)}
   646→            </span>
   647→          ))}
   648→        </div>
   649→      )}
   650→    </div>
   651→  )
   652→}

   656→function CreateMonitorModal({
   657→  password,
   658→  onClose,
   659→  onCreated,
   660→}: {
   661→  password: string
   662→  onClose: () => void
   663→  onCreated: () => void
   664→}) {
   665→  const [form, setForm] = useState({
   666→    matchLabel: '',
   667→    teamCodes: '',
   668→    hashtags: '',
   669→    seedUrls: '',
   670→    playerIds: '',
   671→    refreshInterval: '5',
   672→    durationHours: '6',
   673→  })
   674→  const [submitting, setSubmitting] = useState(false)
   675→  const [error, setError] = useState('')
   676→
   677→  const handleSubmit = async () => {
   678→    setError('')
   679→    if (!form.matchLabel.trim()) {
   680→      setError('Match label is required (e.g. "ESP vs KSA — Matchday 2")')
   681→      return
   682→    }
   683→    const teamCodes = form.teamCodes
   684→      .split(/[,\n\s]+/)
   685→      .map((s) => s.trim().toUpperCase())
   686→      .filter(Boolean)
   687→    const hashtags = form.hashtags
   688→      .split(/[,\n]/)
   689→      .map((s) => s.trim())
   690→      .filter(Boolean)
   691→    if (teamCodes.length === 0) {
   692→      setError('Add at least one team code (e.g. ESP, KSA)')
   693→      return
   694→    }
   695→    if (hashtags.length === 0) {
   696→      setError('Add at least one hashtag (e.g. #LaRoja, #LamineYamal)')
   697→      return
   698→    }
   699→
   700→    const seedUrls = form.seedUrls
   701→      .split(/[,\n\s]+/)
   702→      .map((s) => s.trim())
   703→      .filter((s) => s.startsWith('http'))
   704→    const playerIds = form.playerIds
   705→      .split(/[,\n\s]+/)
   706→      .map((s) => s.trim())
   707→      .filter(Boolean)
   708→
   709→    setSubmitting(true)
   710→    try {
   711→      const res = await fetch('/api/admin/feed-monitor', {
   712→        method: 'POST',
   713→        headers: {
   714→          'x-admin-password': password,
   715→          'Content-Type': 'application/json',
   716→        },
   717→        body: JSON.stringify({
   718→          matchLabel: form.matchLabel.trim(),
   719→          teamCodes,
   720→          hashtags,
   721→          seedUrls,
   722→          playerIds,
   723→          refreshInterval: Number(form.refreshInterval),
   724→          durationHours: Number(form.durationHours),
   725→        }),
   726→      })
   727→      const data = await res.json()
   728→      if (res.ok) {
   729→        onCreated()
   730→      } else {
   731→        setError(data.error || 'Failed to create monitor')
   732→      }
   733→    } catch (err) {
   734→      setError('Network error — try again')
   735→    } finally {
   736→      setSubmitting(false)
   737→    }
   738→  }
   739→
   740→  return (
   741→    <motion.div
   742→      initial={{ opacity: 0 }}
   743→      animate={{ opacity: 1 }}
   744→      exit={{ opacity: 0 }}
   745→      className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4"
   746→      onClick={onClose}
   747→    >
   748→      <motion.div
   749→        initial={{ scale: 0.95, opacity: 0 }}
   750→        animate={{ scale: 1, opacity: 1 }}
   751→        exit={{ scale: 0.95, opacity: 0 }}
   752→        className="bg-[#1A1A1A] border border-white/10 rounded-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto"
   753→        onClick={(e) => e.stopPropagation()}
   754→      >
   755→        <div className="sticky top-0 bg-[#1A1A1A] border-b border-white/10 px-6 py-4 flex items-center justify-between">
   756→          <div>
   757→            <h2 className="font-bold text-base">Create Feed Monitor</h2>
   758→            <p className="text-xs text-white/50">Scrape real fan sentiment for a match</p>
   759→          </div>
   760→          <button onClick={onClose} className="p-1 rounded-lg hover:bg-white/5 text-white/60">
   761→            <X className="size-5" />
   762→          </button>
   763→        </div>
   764→
   765→        <div className="p-6 space-y-4">
   766→          <Field
   767→            label="Match Label"
   768→            placeholder="e.g. ESP vs KSA — Matchday 2"
   769→            value={form.matchLabel}
   770→            onChange={(v) => setForm({ ...form, matchLabel: v })}
   771→            hint="Human-readable label shown in the monitor list"
   772→          />
   773→          <Field
   774→            label="Team Codes"
   775→            placeholder="ESP, KSA"
   776→            value={form.teamCodes}
   777→            onChange={(v) => setForm({ ...form, teamCodes: v })}
   778→            hint="Comma-separated FIFA team codes (ESP, KSA, BRA, etc.)"
   779→          />
   780→          <Field
   781→            label="Hashtags"
   782→            placeholder={"#LaRoja, #LamineYamal, #ESPKSA\n#VamosEspaña"}
   783→            value={form.hashtags}
   784→            onChange={(v) => setForm({ ...form, hashtags: v })}
   785→            hint="One per line or comma-separated. These anchor the search."
   786→            textarea
   787→          />
   788→          <Field
   789→            label="Seed Tweet/Post URLs (optional but recommended)"
   790→            placeholder={"https://x.com/.../status/1234...\nhttps://reddit.com/r/soccer/comments/..."}
   791→            value={form.seedUrls}
   792→            onChange={(v) => setForm({ ...form, seedUrls: v })}
   793→            hint="Viral tweets that anchor the conversation. 3-5 URLs ideal."
   794→            textarea
   795→          />
   796→          <Field
   797→            label="Player IDs to track (optional)"
   798→            placeholder="cmqj36lcp00bordjrdro0geqk, cmqj36lci00b6rdjrsv75dxon"
   799→            value={form.playerIds}
   800→            onChange={(v) => setForm({ ...form, playerIds: v })}
   801→            hint="Comma-separated WCSelectionPlayer IDs. Leave empty to track all players on selected teams."
   802→          />
   803→          <div className="grid grid-cols-2 gap-3">
   804→            <Field
   805→              label="Refresh Interval (min)"
   806→              placeholder="5"
   807→              value={form.refreshInterval}
   808→              onChange={(v) => setForm({ ...form, refreshInterval: v })}
   809→              hint="1-60 min. 5 min recommended for live matches."
   810→            />
   811→            <Field
   812→              label="Duration (hours)"
   813→              placeholder="6"
   814→              value={form.durationHours}
   815→              onChange={(v) => setForm({ ...form, durationHours: v })}
   816→              hint="1-48 hours. Auto-end after this window."
   817→            />
   818→          </div>
   819→
   820→          {error && (
   821→            <div className="bg-[#EF4444]/10 border border-[#EF4444]/30 rounded-lg p-3 text-xs text-[#EF4444] flex items-start gap-2">
   822→              <AlertCircle className="size-4 shrink-0 mt-0.5" />
   823→              <span>{error}</span>
   824→            </div>
   825→          )}
   826→
   827→          <div className="bg-[#6C2BD9]/5 border border-[#6C2BD9]/20 rounded-lg p-3 text-xs text-white/60">
   828→            <p className="font-bold text-[#8B5CF6] mb-1">How it works</p>
   829→            <p>Once created, the system will immediately run a sentiment refresh in the background (30-60s). The cron job then re-searches your hashtags every {form.refreshInterval || 5} minutes for new posts, scores them with the LLM, and updates each tracked player's Pulse Score. Monitors auto-end after {form.durationHours || 6} hours.</p>
   830→          </div>
   831→        </div>
   832→
   833→        <div className="sticky bottom-0 bg-[#1A1A1A] border-t border-white/10 px-6 py-4 flex justify-end gap-2">
   834→          <button
   835→            onClick={onClose}
   836→            className="text-sm font-bold px-4 py-2.5 rounded-lg text-white/60 hover:text-white hover:bg-white/5 transition-colors"
   837→          >
   838→            Cancel
   839→          </button>
   840→          <button
   841→            onClick={handleSubmit}
   842→            disabled={submitting}
   843→            className="bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white text-sm font-bold px-5 py-2.5 rounded-lg hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center gap-2"
   844→          >
   845→            {submitting ? (
   846→              <>
   847→                <RefreshCw className="size-4 animate-spin" /> Creating...
   848→              </>
   849→            ) : (
   850→              <>
   851→                <Plus className="size-4" /> Create Monitor
   852→              </>
   853→            )}
   854→          </button>
   855→        </div>
   856→      </motion.div>
   857→    </motion.div>
   858→  )
   859→}
