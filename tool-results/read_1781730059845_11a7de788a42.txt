   247→   247→function HomeTab() {
   248→   248→  const { t } = useLanguage()
   249→   249→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   250→   250→  const [apiMatches, setApiMatches] = useState<Array<{
   251→   251→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   252→   252→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   253→   253→  }>>([])
   254→   254→
   255→   255→  useEffect(() => {
   256→   256→    async function fetchMatches() {
   257→   257→      try {
   258→   258→        const res = await fetch('/api/matches?league=WC')
   259→   259→        if (res.ok) {
   260→   260→          const data = await res.json()
   261→   261→          const mapped = (data.matches || []).map((m: any) => ({
   262→   262→            id: m.id,
   263→   263→            home: m.homeTeam.code,
   264→   264→            away: m.awayTeam.code,
   265→   265→            homeFlag: m.homeTeam.flag || '',
   266→   266→            awayFlag: m.awayTeam.flag || '',
   267→   267→            score: m.score,
   268→   268→            homeSentiment: Math.round(m.homeTeam.sentiment),
   269→   269→            awaySentiment: Math.round(m.awayTeam.sentiment),
   270→   270→            live: m.status === 'live',
   271→   271→            league: m.league === 'WC' ? `WC Group ${m.group}` : m.league,
   272→   272→          }))
   273→   273→          setApiMatches(mapped)
   274→   274→        }
   275→   275→      } catch (err) {
   276→   276→        console.error('Failed to fetch matches:', err)
   277→   277→      }
   278→   278→    }
   279→   279→    fetchMatches()
   280→   280→  }, [])
   281→   281→
   282→   282→  const filteredMatches = (matchFilter === 'ALL'
