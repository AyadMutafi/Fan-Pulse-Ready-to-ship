    29→// ── World Cup stage label helper ─────────────────────────────
    30→// Renders an accurate label for a WC match based on its `group` value.
    31→// Group-stage matches use letters A-L; knockout rounds use R32/R16/QF/SF/Final.
    32→// (Previously everything was labelled "WC Group {x}", which produced the
    33→// nonsensical "WC Group R32" for knockout matches.)
    34→function wcStageLabel(group: string | null | undefined): string {
    35→  if (!group) return 'World Cup 2026'
    36→  if (/^[A-L]$/.test(group)) return `WC Group ${group}`
    37→  switch (group) {
    38→    case 'R32': return 'WC Round of 32'
    39→    case 'R16': return 'WC Round of 16'
    40→    case 'QF': return 'WC Quarter Finals'
    41→    case 'SF': return 'WC Semi Finals'
    42→    case 'Final': return 'WC Final'
    43→    default: return 'World Cup 2026'
    44→  }
    45→}

   423→  const handleVote = async (teamCode: string, score: number) => {
   424→    if (!sessionId || submitting) return
   425→    setSubmitting(true)
   426→    // Optimistic update: bump my vote immediately
   427→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   428→    setMyVotes(prev => {
   429→      const without = prev.filter(v => v.teamCode !== teamCode)
   430→      return [...without, { teamCode, score }]
   431→    })
   432→    setFanVotes(prev => {
   433→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   434→      if (idx === -1) {
   435→        return [...prev, { teamCode, score, count: 1 }]
   436→      }
   437→      const next = [...prev]
   438→      const wasMine = prevMyVote !== null
   439→      // Adjust running average: if I had voted, swap my old score for the new.
   440→      const current = next[idx]
   441→      const totalScore = current.score * current.count
   442→      const newScore = wasMine
   443→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   444→        : Math.round((totalScore + score) / (current.count + 1))
   445→      const newCount = wasMine ? current.count : current.count + 1
   446→      next[idx] = { teamCode, score: newScore, count: newCount }
   447→      return next
   448→    })
   449→    setSelectedVoteTeam(null)
   450→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   451→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   452→    setTimeout(() => setToast(null), 2500)
   453→    // Show the Fan Card offer for 8 seconds — long enough to click "Get Fan Card"
   454→    setFanCardOffer({ teamCode, score })
   455→    const offerTimer = setTimeout(() => setFanCardOffer(null), 8000)
   456→    try {
   457→      await fetch('/api/fan-vote', {
   458→        method: 'POST',
   459→        headers: { 'Content-Type': 'application/json' },
   460→        body: JSON.stringify({ teamCode, score, sessionId }),
