     7→// ── Admin Feed Monitor Page ──────────────────────────────────────────────────
     8→// Password-protected UI for creating and managing FeedMonitors.
     9→//
    10→// H4 SECURITY FIX: Auth is now cookie-based, NOT localStorage-based.
    11→// The admin enters the password on the login form; the client POSTs it to
    12→// /api/admin/login, which validates it and sets an HttpOnly + Secure +
    13→// SameSite=Strict cookie. The browser sends the cookie automatically on
    14→// every subsequent request — JS cannot read it, so XSS cannot exfiltrate it
    15→// (unlike the old localStorage approach). Logout clears the cookie via
    16→// /api/admin/logout.
    17→
    18→interface Monitor {
    19→  id: string
    20→  matchLabel: string
    21→  stageId: string | null
    22→  teamCodes: string[]
    23→  playerIds: string[]
    24→  hashtags: string[]
    25→  seedUrls: string[]
    26→  status: 'active' | 'paused' | 'ended'
    27→  refreshInterval: number
    28→  lastRefreshedAt: string | null
    29→  endsAt: string
    30→  createdAt: string
    31→  postCount: number
    32→  playerSentimentCount: number
    33→}
    34→
    35→interface Post {
    36→  id: string
    37→  platform: string
    38→  url: string
    39→  author: string
    40→  content: string
    41→  language: string
    42→  sentimentScore: number
    43→  positiveRatio: number
    44→  mentionedPlayers: string[]
    45→  topQuote: string | null
    46→  postedAt: string
    47→  analyzedAt: string
    48→}
    49→
    50→export default function FeedMonitorAdminPage() {
    51→  // H4: No more localStorage. The password is never stored client-side.
    52→  // `authed` starts false; on mount we probe the API (cookie sent automatically)
    53→  // to see if a valid admin session already exists.
    54→  const [password, setPassword] = useState('')
    55→  const [authed, setAuthed] = useState(false)
    56→  const [checkingAuth, setCheckingAuth] = useState(true)
    57→  const [authError, setAuthError] = useState('')
    58→
    59→  // On mount, check if a valid cookie session already exists.
    60→  useEffect(() => {
    61→    let cancelled = false
    62→    fetch('/api/admin/feed-monitor')
    63→      .then((res) => {
    64→        if (cancelled) return
    65→        if (res.ok) {
    66→          setAuthed(true)
    67→        } else {
    68→          setAuthed(false)
    69→        }
    70→      })
    71→      .catch(() => {
    72→        if (!cancelled) setAuthed(false)
    73→      })
    74→      .finally(() => {
    75→        if (!cancelled) setCheckingAuth(false)
    76→      })
    77→    return () => { cancelled = true }
    78→  }, [])
    79→
    80→  const handleAuth = () => {
    81→    if (!password.trim()) {
    82→      setAuthError('Password is required')
    83→      return
    84→    }
    85→    // POST the password to /api/admin/login — the server sets an HttpOnly
    86→    // cookie. The client never persists the password.
    87→    fetch('/api/admin/login', {
    88→      method: 'POST',
    89→      headers: { 'Content-Type': 'application/json' },
    90→      body: JSON.stringify({ password }),
    91→    })
    92→      .then((res) => {
    93→        if (res.ok) {
    94→          setAuthed(true)
    95→          setAuthError('')
    96→          setPassword('') // clear from memory — no longer needed
    97→        } else {
    98→          setAuthError('Invalid password')
    99→          setAuthed(false)
   100→        }
   101→      })
   102→      .catch(() => {
   103→        setAuthError('Network error — try again')
   104→      })
   105→  }
   106→
   107→  const handleLogout = () => {
   108→    fetch('/api/admin/logout', { method: 'POST' })
   109→      .catch(() => {})
   110→      .finally(() => {
   111→        setAuthed(false)
   112→        setPassword('')
   113→      })
   114→  }
   115→
   116→  if (checkingAuth) {
   117→    return (
   118→      <div className="min-h-screen flex items-center justify-center bg-[#0A0A0A]">
   119→        <div className="animate-spin w-8 h-8 rounded-full border-2 border-white/10 border-t-[#6C2BD9]" />
   120→      </div>
   121→    )
   122→  }
   123→
   124→  if (!authed) {
   125→    return (
   126→      <div className="min-h-screen flex items-center justify-center bg-[#0A0A0A] p-4">
   127→        <motion.div
   128→          initial={{ opacity: 0, y: 20 }}
   129→          animate={{ opacity: 1, y: 0 }}
   130→          className="w-full max-w-md bg-[#1A1A1A] border border-white/10 rounded-2xl p-8 shadow-2xl"
   131→        >
   132→          <div className="flex items-center gap-3 mb-6">
   133→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center text-white font-black text-xl">
   134→              F
   135→            </div>
   136→            <div>
   137→              <h1 className="text-white font-bold text-lg">Fan Pulse Admin</h1>
   138→              <p className="text-white/50 text-xs">Feed Monitor Pipeline</p>
   139→            </div>
   140→          </div>
   141→          <label className="block text-xs font-bold text-white/70 mb-2 uppercase tracking-wider">
   142→            Admin Password
   143→          </label>
   144→          <input
   145→            type="password"
   146→            value={password}
   147→            onChange={(e) => setPassword(e.target.value)}
   148→            onKeyDown={(e) => e.key === 'Enter' && handleAuth()}
   149→            placeholder="Enter admin password"
   150→            className="w-full bg-[#0A0A0A] border border-white/10 rounded-lg px-4 py-3 text-white text-sm focus:outline-none focus:border-[#6C2BD9] focus:ring-2 focus:ring-[#6C2BD9]/30 transition-all"
   151→            autoFocus
   152→          />
   153→          {authError && (
   154→            <p className="mt-2 text-xs text-[#EF4444] flex items-center gap-1.5">
   155→              <AlertCircle className="size-3.5" /> {authError}
   156→            </p>
   157→          )}
   158→          <button
   159→            onClick={handleAuth}
   160→            className="mt-4 w-full bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white font-bold py-3 rounded-lg hover:opacity-90 transition-opacity flex items-center justify-center gap-2 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   161→          >
   162→            <Lock className="size-4" /> Unlock Admin
   163→          </button>
   164→          <p className="mt-4 text-xs text-white/40 text-center">
   165→            Admin password required. Set ADMIN_PASSWORD in your environment.
   166→          </p>
   167→        </motion.div>
   168→      </div>
   169→    )
   170→  }
   171→
   172→  return <MonitorDashboard onLogout={handleLogout} />
   173→}

   175→// ── Monitor Dashboard (shown when authed) ────────────────────────────────────

   177→function MonitorDashboard({ onLogout }: { onLogout: () => void }) {
   178→  const [monitors, setMonitors] = useState<Monitor[]>([])
   179→  const [loading, setLoading] = useState(true)
   180→  const [showCreateForm, setShowCreateForm] = useState(false)
   181→  const [expandedMonitorId, setExpandedMonitorId] = useState<string | null>(null)
   182→  const [refreshing, setRefreshing] = useState<string | null>(null)
   183→  const [toast, setToast] = useState<{ msg: string; type: 'success' | 'error' } | null>(null)
   184→
   185→  const fetchMonitors = useCallback(async () => {
   186→    try {
   187→      // H4: cookie sent automatically — no manual auth header needed.
   188→      const res = await fetch('/api/admin/feed-monitor')
   189→      if (res.ok) {
   190→        const data = await res.json()
   191→        setMonitors(data.monitors || [])
   192→      }
   193→    } catch (err) {
   194→      console.error('Failed to fetch monitors:', err)
   195→    } finally {
   196→      setLoading(false)
   197→    }
   198→  }, [])
   199→
   200→  useEffect(() => {
   201→    fetchMonitors()
   202→    // Auto-refresh the monitor list every 30s so admins see live updates
   203→    const interval = setInterval(fetchMonitors, 30000)
   204→    return () => clearInterval(interval)
   205→  }, [fetchMonitors])
   206→
   207→  const handleRefresh = async (id: string) => {
   208→    setRefreshing(id)
   209→    try {
   210→      const res = await fetch(`/api/admin/feed-monitor/${id}/refresh`, {
   211→        method: 'POST',
   212→      })
   213→      const data = await res.json()
   214→      if (res.ok) {
   215→        const r = data.refresh
   216→        setToast({
   217→          msg: `Refreshed: ${r.newPosts} new posts, ${r.playersUpdated} players updated in ${Math.round(r.durationMs / 1000)}s`,
   218→          type: 'success',
   219→        })
   220→        fetchMonitors()
   221→      } else {
   222→        setToast({ msg: data.error || 'Refresh failed', type: 'error' })
   223→      }
   224→    } catch (err) {
   225→      setToast({ msg: 'Network error', type: 'error' })
   226→    } finally {
   227→      setRefreshing(null)
   228→      setTimeout(() => setToast(null), 5000)
   229→    }
   230→  }
   231→
   232→  const handleStatusChange = async (id: string, status: 'active' | 'paused' | 'ended') => {
   233→    try {
   234→      const res = await fetch(`/api/admin/feed-monitor/${id}`, {
   235→        method: 'PATCH',
   236→        headers: {
   237→          'Content-Type': 'application/json',
   238→        },
   239→        body: JSON.stringify({ status }),
   240→      })
   241→      if (res.ok) {
   242→        fetchMonitors()
   243→        setToast({ msg: `Monitor ${status}`, type: 'success' })
   244→        setTimeout(() => setToast(null), 3000)
   245→      }
   246→    } catch (err) {
   247→      setToast({ msg: 'Status update failed', type: 'error' })
   248→    }
   249→  }
   250→
   251→  const handleDelete = async (id: string) => {
   252→    if (!confirm('Delete this monitor and all its scraped posts? Player sentiment aggregates will be preserved.')) {
   253→      return
   254→    }
   255→    try {
   256→      const res = await fetch(`/api/admin/feed-monitor/${id}`, {
   257→        method: 'DELETE',
   258→      })
   259→      if (res.ok) {
   260→        fetchMonitors()
   261→        setToast({ msg: 'Monitor deleted', type: 'success' })
   262→        setTimeout(() => setToast(null), 3000)
   263→      }
   264→    } catch (err) {
   265→      setToast({ msg: 'Delete failed', type: 'error' })
   266→    }
   267→  }
   268→
   269→  return (
   270→    <div className="min-h-screen bg-[#0A0A0A] text-white">
   271→      {/* Header */}
   272→      <header className="border-b border-white/10 bg-[#1A1A1A] sticky top-0 z-20">
   273→        <div className="max-w-6xl mx-auto px-4 py-4 flex items-center justify-between">
   274→          <div className="flex items-center gap-3">
   275→            <div className="w-9 h-9 rounded-lg bg-gradient-to-br from-[#6C2BD9] to-[#10B981] flex items-center justify-center font-black">
   276→              F
   277→            </div>
   278→            <div>
   279→              <h1 className="font-bold text-sm">Feed Monitor Pipeline</h1>
   280→              <p className="text-white/50 text-xs">Admin · Fan Sentiment</p>
   281→            </div>
   282→          </div>
   283→          <div className="flex items-center gap-2">
   284→            <button
   285→              onClick={() => setShowCreateForm(true)}
   286→              className="bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white text-xs font-bold px-4 py-2 rounded-lg hover:opacity-90 transition-opacity flex items-center gap-1.5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   287→            >
   288→              <Plus className="size-3.5" /> New Monitor
   289→            </button>
   290→            <button
   291→              onClick={onLogout}
   292→              className="text-white/60 hover:text-white text-xs font-bold px-3 py-2 rounded-lg border border-white/10 hover:border-white/20 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   293→            >
   294→              Logout
   295→            </button>
   296→          </div>
   297→        </div>
   298→      </header>
   299→
   300→      <main className="max-w-6xl mx-auto px-4 py-6">
   301→        {/* Stats overview */}
   302→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mb-6">
   303→          <StatCard label="Active Monitors" value={monitors.filter((m) => m.status === 'active').length} icon={<Activity className="size-4 text-[#10B981]" />} />
   304→          <StatCard label="Total Posts Scraped" value={monitors.reduce((s, m) => s + m.postCount, 0)} icon={<MessageCircle className="size-4 text-[#8B5CF6]" />} />
   305→          <StatCard label="Players Tracked" value={monitors.reduce((s, m) => s + m.playerSentimentCount, 0)} icon={<Hash className="size-4 text-[#FF6B35]" />} />
   306→          <StatCard label="Ended Monitors" value={monitors.filter((m) => m.status === 'ended').length} icon={<Clock className="size-4 text-white/50" />} />
   307→        </div>
   308→
   309→        {/* Monitor list */}
   310→        {loading ? (
   311→          <div className="flex items-center justify-center py-20">
   312→            <div className="animate-spin w-8 h-8 rounded-full border-2 border-white/10 border-t-[#6C2BD9]" />
   313→          </div>
   314→        ) : monitors.length === 0 ? (
   315→          <div className="text-center py-20">
   316→            <div className="w-16 h-16 mx-auto rounded-2xl bg-white/5 flex items-center justify-center mb-4">
   317→              <MessageCircle className="size-8 text-white/30" />
   318→            </div>
   319→            <h2 className="text-lg font-bold mb-1">No monitors yet</h2>
   320→            <p className="text-white/50 text-sm mb-4">Create a feed monitor to start scraping real fan sentiment.</p>
   321→            <button
   322→              onClick={() => setShowCreateForm(true)}
   323→              className="bg-gradient-to-r from-[#6C2BD9] to-[#8B5CF6] text-white text-sm font-bold px-5 py-2.5 rounded-lg hover:opacity-90 transition-opacity inline-flex items-center gap-2 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   324→            >
   325→              <Plus className="size-4" /> Create First Monitor
   326→            </button>
   327→          </div>
   328→        ) : (
   329→          <div className="space-y-3">
   330→            {monitors.map((m) => (
   331→              <MonitorCard
   332→                key={m.id}
   333→                monitor={m}
   334→                expanded={expandedMonitorId === m.id}
   335→                onToggle={() => setExpandedMonitorId(expandedMonitorId === m.id ? null : m.id)}
   336→                onRefresh={() => handleRefresh(m.id)}
   337→                onStatusChange={(s) => handleStatusChange(m.id, s)}
   338→                onDelete={() => handleDelete(m.id)}
   339→                refreshing={refreshing === m.id}
   340→              />
   341→            ))}
   342→          </div>
   343→        )}
   344→      </main>
   345→
   346→      {/* Create form modal */}
   347→      <AnimatePresence>
   348→        {showCreateForm && (
   349→          <CreateMonitorModal
   350→            onClose={() => setShowCreateForm(false)}
   351→            onCreated={() => {
   352→              setShowCreateForm(false)
   353→              fetchMonitors()
   354→              setToast({ msg: 'Monitor created! Initial refresh running in background — check back in 30-60s.', type: 'success' })
   355→              setTimeout(() => setToast(null), 6000)
   356→            }}
   357→          />
   358→        )}
   359→      </AnimatePresence>
   360→
   361→      {/* Toast */}
   362→      <AnimatePresence>
   363→        {toast && (
   364→          <motion.div
   365→            initial={{ opacity: 0, y: 20 }}
   366→            animate={{ opacity: 1, y: 0 }}
   367→            exit={{ opacity: 0, y: 20 }}
   368→            className={`fixed bottom-6 left-1/2 -translate-x-1/2 z-50 px-5 py-3 rounded-xl shadow-2xl border text-sm font-medium ${
   369→              toast.type === 'success'
   370→                ? 'bg-[#10B981]/10 border-[#10B981]/30 text-[#10B981]'
   371→                : 'bg-[#EF4444]/10 border-[#EF4444]/30 text-[#EF4444]'
   372→            }`}
   373→          >
   374→            <div className="flex items-center gap-2">
   375→              {toast.type === 'success' ? <CheckCircle2 className="size-4" /> : <AlertCircle className="size-4" />}
   376→              {toast.msg}
   377→            </div>
   378→          </motion.div>
   379→        )}
   380→      </AnimatePresence>
   381→    </div>
   382→  )
   383→}

   385→function StatCard({ label, value, icon }: { label: string; value: number; icon: React.ReactNode }) {
   386→  return (
   387→    <div className="bg-[#1A1A1A] border border-white/10 rounded-xl p-4">
   388→      <div className="flex items-center justify-between mb-1">
   389→        <span className="text-xs font-bold text-white/60 uppercase tracking-wider">{label}</span>
   390→        {icon}
   391→      </div>
   392→      <div className="text-2xl font-black">{value.toLocaleString()}</div>

   397→function MonitorCard({
   398→  monitor,
   399→  expanded,
   400→  onToggle,
   401→  onRefresh,
   402→  onStatusChange,
   403→  onDelete,
   404→  refreshing,
   405→}: {
   406→  monitor: Monitor
   407→  expanded: boolean
   408→  onToggle: () => void
   409→  onRefresh: () => void
   410→  onStatusChange: (status: 'active' | 'paused' | 'ended') => void
   411→  onDelete: () => void
   412→  refreshing: boolean
   413→}) {
   414→  const statusColors = {
   415→    active: 'bg-[#10B981]/10 text-[#10B981] border-[#10B981]/30',
   416→    paused: 'bg-[#FF6B35]/10 text-[#FF6B35] border-[#FF6B35]/30',
   417→    ended: 'bg-white/5 text-white/50 border-white/10',
   418→  }
   419→
   420→  const lastRefreshed = monitor.lastRefreshedAt
   421→    ? new Date(monitor.lastRefreshedAt).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
   422→    : 'never'
   423→
   424→  const endsAt = new Date(monitor.endsAt)
   425→  const timeLeft = endsAt.getTime() - Date.now()
   426→  const timeLeftLabel = timeLeft > 0 ? `${Math.floor(timeLeft / 3600000)}h ${Math.floor((timeLeft % 3600000) / 60000)}m left` : 'ended'
   427→
   428→  return (
   429→    <div className="bg-[#1A1A1A] border border-white/10 rounded-xl overflow-hidden">
   430→      <div className="p-4 flex items-center gap-3">
   431→        <button onClick={onToggle} className="text-white/40 hover:text-white transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2 rounded">
   432→          {expanded ? <ChevronDown className="size-4" /> : <ChevronRight className="size-4" />}
   433→        </button>
   434→        <div className="flex-1 min-w-0">
   435→          <div className="flex items-center gap-2 mb-1">
   436→            <h3 className="font-bold text-sm truncate">{monitor.matchLabel}</h3>
   437→            <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full border uppercase ${statusColors[monitor.status]}`}>
   438→              {monitor.status}
   439→            </span>
   440→          </div>
   441→          <div className="flex items-center gap-3 text-xs text-white/50 flex-wrap">
   442→            <span className="flex items-center gap-1">
   443→              <Hash className="size-3" /> {monitor.hashtags.length} tags
   444→            </span>
   445→            <span className="flex items-center gap-1">
   446→              <MessageCircle className="size-3" /> {monitor.postCount} posts
   447→            </span>
   448→            <span className="flex items-center gap-1">
   449→              <Activity className="size-3" /> {monitor.playerSentimentCount} players
   450→            </span>
   451→            <span className="flex items-center gap-1">
   452→              <Clock className="size-3" /> refreshed {lastRefreshed}
   453→            </span>
   454→            <span className="text-white/40">· {timeLeftLabel}</span>
   455→          </div>
   456→        </div>
   457→        <div className="flex items-center gap-1">
   458→          <button
   459→            onClick={onRefresh}
   460→            disabled={refreshing || monitor.status === 'ended'}
   461→            title="Manual refresh"
   462→            className="p-2 rounded-lg hover:bg-white/5 text-white/60 hover:text-white disabled:opacity-30 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   463→          >
   464→            <RefreshCw className={`size-4 ${refreshing ? 'animate-spin' : ''}`} />
   465→          </button>
   466→          {monitor.status === 'active' ? (
   467→            <button
   468→              onClick={() => onStatusChange('paused')}
   469→              title="Pause"
   470→              className="p-2 rounded-lg hover:bg-white/5 text-white/60 hover:text-white transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   471→            >
   472→              <Pause className="size-4" />
   473→            </button>
   474→          ) : monitor.status === 'paused' ? (
   475→            <button
   476→              onClick={() => onStatusChange('active')}
   477→              title="Resume"
   478→              className="p-2 rounded-lg hover:bg-white/5 text-[#10B981] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   479→            >
   480→              <Play className="size-4" />
   481→            </button>
   482→          ) : null}
   483→          <button
   484→            onClick={() => onStatusChange('ended')}
   485→            disabled={monitor.status === 'ended'}
   486→            title="End monitor"
   487→            className="p-2 rounded-lg hover:bg-white/5 text-white/60 hover:text-white disabled:opacity-30 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   488→          >
   489→            <Square className="size-4" />
   490→          </button>
   491→          <button
   492→            onClick={onDelete}
   493→            title="Delete"
   494→            className="p-2 rounded-lg hover:bg-[#EF4444]/10 text-white/60 hover:text-[#EF4444] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   495→          >
   496→            <Trash2 className="size-4" />

   607→function PostRow({ post }: { post: Post }) {
   608→  const [expanded, setExpanded] = useState(false)
   609→
   610→  const platformColors = {
   611→    twitter: 'bg-[#1DA1F2]/10 text-[#1DA1F2] border-[#1DA1F2]/30',
   612→    reddit: 'bg-[#FF4500]/10 text-[#FF4500] border-[#FF4500]/30',
   613→    web: 'bg-white/5 text-white/60 border-white/10',
   614→  }
   615→
   616→  const sentimentColor =
   617→    post.sentimentScore >= 70
   618→      ? 'text-[#10B981]'
   619→      : post.sentimentScore >= 45
   620→        ? 'text-[#FF6B35]'
   621→        : 'text-[#EF4444]'
   622→
   623→  return (
   624→    <div className="bg-[#1A1A1A] border border-white/5 rounded-lg p-3">
   625→      <div className="flex items-start justify-between gap-2 mb-1">
   626→        <div className="flex items-center gap-2 flex-wrap">
   627→          <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full border uppercase ${platformColors[post.platform as keyof typeof platformColors] || platformColors.web}`}>
   628→            {post.platform}
   629→          </span>
   630→          <span className="text-xs font-bold text-white/80">{post.author || 'unknown'}</span>
   631→          <span className="text-xs text-white/40">{new Date(post.postedAt).toLocaleDateString()}</span>
   632→        </div>
   633→        <div className="flex items-center gap-2 shrink-0">
   634→          <span className={`text-sm font-black ${sentimentColor}`}>{Math.round(post.sentimentScore)}</span>
   635→          <a
   636→            href={post.url}
   637→            target="_blank"
   638→            rel="noopener noreferrer"
   639→            className="text-white/40 hover:text-white transition-colors"
   640→          >
   641→            <ExternalLink className="size-3.5" />
   642→          </a>
   643→        </div>
   644→      </div>
   645→      <p className={`text-xs text-white/70 ${expanded ? '' : 'line-clamp-2'}`}>
   646→        {post.content}
   647→      </p>
   648→      {post.content.length > 200 && (
   649→        <button
   650→          onClick={() => setExpanded(!expanded)}
   651→          className="text-xs text-[#8B5CF6] hover:underline mt-1 rounded focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2"
   652→        >
   653→          {expanded ? 'show less' : 'show more'}
   654→        </button>
   655→      )}
   656→      {post.topQuote && (
   657→        <div className="mt-2 px-3 py-2 bg-[#6C2BD9]/5 border-l-2 border-[#6C2BD9] rounded-r">
   658→          <p className="text-xs italic text-white/80">&ldquo;{post.topQuote}&rdquo;</p>
   659→        </div>
   660→      )}
   661→      {post.mentionedPlayers.length > 0 && (
   662→        <div className="mt-2 flex items-center gap-1 flex-wrap">
   663→          <span className="text-[10px] text-white/40">mentions:</span>
   664→          {post.mentionedPlayers.slice(0, 3).map((id, i) => (
   665→            <span key={i} className="text-[10px] text-[#FF6B35] font-mono">
   666→              {id.slice(-6)}
   667→            </span>
   668→          ))}
   669→        </div>
   670→      )}
   671→    </div>
   672→  )
   673→}

   677→function CreateMonitorModal({
   678→  onClose,
   679→  onCreated,
   680→}: {
   681→  onClose: () => void
   682→  onCreated: () => void
   683→}) {
   684→  const [form, setForm] = useState({
   685→    matchLabel: '',
   686→    teamCodes: '',
   687→    hashtags: '',
   688→    seedUrls: '',
   689→    playerIds: '',
   690→    refreshInterval: '5',
   691→    durationHours: '6',
   692→  })
   693→  const [submitting, setSubmitting] = useState(false)
   694→  const [error, setError] = useState('')
   695→
   696→  const handleSubmit = async () => {
   697→    setError('')
   698→    if (!form.matchLabel.trim()) {
   699→      setError('Match label is required (e.g. "ESP vs KSA — Matchday 2")')
   700→      return
   701→    }
   702→    const teamCodes = form.teamCodes
   703→      .split(/[,\n\s]+/)
   704→      .map((s) => s.trim().toUpperCase())
   705→      .filter(Boolean)
   706→    const hashtags = form.hashtags
   707→      .split(/[,\n]/)
   708→      .map((s) => s.trim())
   709→      .filter(Boolean)
   710→    if (teamCodes.length === 0) {
   711→      setError('Add at least one team code (e.g. ESP, KSA)')
   712→      return
   713→    }
   714→    if (hashtags.length === 0) {
   715→      setError('Add at least one hashtag (e.g. #LaRoja, #LamineYamal)')
   716→      return
   717→    }
   718→
   719→    const seedUrls = form.seedUrls
   720→      .split(/[,\n\s]+/)
   721→      .map((s) => s.trim())
   722→      .filter((s) => s.startsWith('http'))
   723→    const playerIds = form.playerIds
   724→      .split(/[,\n\s]+/)
   725→      .map((s) => s.trim())
   726→      .filter(Boolean)
   727→
   728→    setSubmitting(true)
   729→    try {
   730→      const res = await fetch('/api/admin/feed-monitor', {
   731→        method: 'POST',
   732→        headers: {
   733→          'Content-Type': 'application/json',
   734→        },
   735→        body: JSON.stringify({
   736→          matchLabel: form.matchLabel.trim(),
   737→          teamCodes,
   738→          hashtags,
   739→          seedUrls,
   740→          playerIds,
   741→          refreshInterval: Number(form.refreshInterval),
   742→          durationHours: Number(form.durationHours),
   743→        }),
   744→      })
   745→      const data = await res.json()
   746→      if (res.ok) {
   747→        onCreated()
   748→      } else {
   749→        setError(data.error || 'Failed to create monitor')
   750→      }
   751→    } catch (err) {
   752→      setError('Network error — try again')
   753→    } finally {
   754→      setSubmitting(false)
   755→    }
   756→  }
   757→
   758→  return (
   759→    <motion.div
   760→      initial={{ opacity: 0 }}
   761→      animate={{ opacity: 1 }}
   762→      exit={{ opacity: 0 }}
   763→      className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4"
   764→      onClick={onClose}
   765→    >
   766→      <motion.div
   767→        initial={{ scale: 0.95, opacity: 0 }}
   768→        animate={{ scale: 1, opacity: 1 }}
   769→        exit={{ scale: 0.95, opacity: 0 }}
   770→        className="bg-[#1A1A1A] border border-white/10 rounded-2xl max-w-2xl w-full max-h-[90vh] overflow-y-auto"
   771→        onClick={(e) => e.stopPropagation()}
   772→      >
   773→        <div className="sticky top-0 bg-[#1A1A1A] border-b border-white/10 px-6 py-4 flex items-center justify-between">
   774→          <div>
   775→            <h2 className="font-bold text-base">Create Feed Monitor</h2>
   776→            <p className="text-xs text-white/50">Scrape real fan sentiment for a match</p>
   777→          </div>
   778→          <button onClick={onClose} className="p-1 rounded-lg hover:bg-white/5 text-white/60 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#6C2BD9] focus-visible:ring-offset-2">
   779→            <X className="size-5" />
   780→          </button>
   781→        </div>
   782→
   783→        <div className="p-6 space-y-4">
   784→          <Field
   785→            label="Match Label"
   786→            placeholder="e.g. ESP vs KSA — Matchday 2"
   787→            value={form.matchLabel}
   788→            onChange={(v) => setForm({ ...form, matchLabel: v })}
   789→            hint="Human-readable label shown in the monitor list"
   790→          />
   791→          <Field
   792→            label="Team Codes"
   793→            placeholder="ESP, KSA"
   794→            value={form.teamCodes}
   795→            onChange={(v) => setForm({ ...form, teamCodes: v })}
   796→            hint="Comma-separated FIFA team codes (ESP, KSA, BRA, etc.)"
   797→          />
   798→          <Field
   799→            label="Hashtags"
   800→            placeholder={"#LaRoja, #LamineYamal, #ESPKSA\n#VamosEspaña"}
   801→            value={form.hashtags}

   803→            hint="One per line or comma-separated. These anchor the search."
   804→            textarea
   805→          />

   808→            placeholder={"https://x.com/.../status/1234...\nhttps://reddit.com/r/soccer/comments/..."}
   809→            value={form.seedUrls}
   810→            onChange={(v) => setForm({ ...form, seedUrls: v })}
   811→            hint="Viral tweets that anchor the conversation. 3-5 URLs ideal."
   812→            textarea
   813→          />
   814→          <Field
   815→            label="Player IDs to track (optional)"
   816→            placeholder="cmqj36lcp00bordjrdro0geqk, cmqj36lci00b6rdjrsv75dxon"
   817→            value={form.playerIds}

   827→              hint="1-60 min. 5 min recommended for live matches."
   828→            />
   829→            <Field
   830→              label="Duration (hours)"
   831→              placeholder="6"
   832→              value={form.durationHours}
   833→              onChange={(v) => setForm({ ...form, durationHours: v })}
   834→              hint="1-48 hours. Auto-end after this window."
   835→            />
   836→          </div>

   838→          {error && (
