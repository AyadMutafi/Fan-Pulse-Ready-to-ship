    98→    98→function timeAgo(dateStr: string | null, isAr: boolean): string {
    99→    99→  if (!dateStr) return ''
   100→   100→  const now = new Date()
   101→   101→  const date = new Date(dateStr)
   102→   102→  const diffMs = now.getTime() - date.getTime()
   103→   103→  const diffMin = Math.floor(diffMs / 60000)
   104→   104→  const diffHr = Math.floor(diffMs / 3600000)
   105→   105→  const diffDay = Math.floor(diffMs / 86400000)
   106→   106→
   107→   107→  if (isAr) {
   108→   108→    if (diffMin < 1) return 'الآن'
   109→   109→    if (diffMin < 60) return `منذ ${diffMin} د`
   110→   110→    if (diffHr < 24) return `منذ ${diffHr} س`
   111→   111→    return `منذ ${diffDay} ي`
   112→   112→  }
   113→   113→  if (diffMin < 1) return 'now'
   114→   114→  if (diffMin < 60) return `${diffMin}m ago`
   115→   115→  if (diffHr < 24) return `${diffHr}h ago`
   116→   116→  return `${diffDay}d ago`
   117→   117→}

   155→   155→function CircularGauge({ value, size = 100 }: { value: number; size?: number }) {
   156→   156→  const strokeWidth = 8
   157→   157→  const radius = (size - strokeWidth) / 2
   158→   158→  const circumference = 2 * Math.PI * radius
   159→   159→  const progress = (value / 100) * circumference
   160→   160→  const color = getSentimentColor(value)
   161→   161→
   162→   162→  return (
   163→   163→    <div className="relative" style={{ width: size, height: size }}>
   164→   164→      <svg width={size} height={size} className="-rotate-90">
   165→   165→        <circle
   166→   166→          cx={size / 2}
   167→   167→          cy={size / 2}
   168→   168→          r={radius}
   169→   169→          fill="none"
   170→   170→          stroke="currentColor"
   171→   171→          strokeWidth={strokeWidth}
   172→   172→          className="text-[#E0E0E0] dark:text-white/10"
   173→   173→        />
   174→   174→        <motion.circle
   175→   175→          cx={size / 2}
   176→   176→          cy={size / 2}
   177→   177→          r={radius}
   178→   178→          fill="none"
   179→   179→          stroke={color}
   180→   180→          strokeWidth={strokeWidth}
   181→   181→          strokeLinecap="round"
   182→   182→          strokeDasharray={circumference}
   183→   183→          initial={{ strokeDashoffset: circumference }}
   184→   184→          animate={{ strokeDashoffset: circumference - progress }}
   185→   185→          transition={{ duration: 1.2, ease: 'easeOut' }}
   186→   186→        />
   187→   187→      </svg>
   188→   188→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   189→   189→        <motion.span
   190→   190→          className="text-2xl font-black"
   191→   191→          style={{ color }}
   192→   192→          initial={{ opacity: 0, scale: 0.5 }}
   193→   193→          animate={{ opacity: 1, scale: 1 }}
   194→   194→          transition={{ delay: 0.3, duration: 0.4 }}
   195→   195→        >
   196→   196→          {Math.round(value)}
   197→   197→        </motion.span>
   198→   198→        <span className="text-[11px] font-medium text-[#666] dark:text-[#CCCCCC]">/ 100</span>
   199→   199→      </div>
   200→   200→    </div>
   201→   201→  )
   202→   202→}

   245→   245→export default function FanPulseTab() {

   256→   256→  const [platformFilter, setPlatformFilter] = useState<PlatformFilter>('all')

   259→   259→  const [visiblePosts, setVisiblePosts] = useState(10)
   260→   260→
   261→   261→  // ── Fetch Data ──
   262→   262→  const fetchData = useCallback(async () => {

   264→   264→      setError(null)

   271→   271→      if (!res.ok) throw new Error('Failed to fetch data')
   272→   272→      const data = await res.json()
   273→   273→
   274→   274→      setPosts(data.posts || [])
   275→   275→      setSummaries(data.summaries || [])

   278→   278→      setError(isAr ? 'فشل تحميل البيانات' : 'Failed to load data')
   279→   279→    } finally {
   280→   280→      setIsLoading(false)
   281→   281→      setIsRefreshing(false)
   282→   282→    }
   283→   283→  }, [platformFilter, languageFilter, isAr])
   284→   284→
   285→   285→  useEffect(() => {
   286→   286→    fetchData()
   287→   287→  }, [fetchData])
   288→   288→
   289→   289→  // ── Refresh Handler ──

   291→   291→    setIsRefreshing(true)
   292→   292→    try {
   293→   293→      await fetch('/api/social-sentiment', {
   294→   294→        method: 'POST',

   301→   301→      await fetchData()
   302→   302→    } catch (err) {
   303→   303→      console.error('Refresh failed:', err)
   304→   304→      setIsRefreshing(false)
   305→   305→    }
   306→   306→  }, [languageFilter, fetchData])
   307→   307→

   339→   339→      } catch { /* ignore */ }
   340→   340→    })
   341→   341→    return [...topicMap.entries()]
   342→   342→      .sort((a, b) => b[1] - a[1])
   343→   343→      .slice(0, 5)
   344→   344→      .map(([topic]) => topic)
   345→   345→  }, [summaries])
   346→   346→
   347→   347→  const mostActiveLanguage = useMemo(() => {

   373→   373→      negative: Math.round((negative / total) * 100),
   374→   374→    }
   375→   375→  }, [filteredPosts])

   381→   381→      langSentiments: Map<string, { avg: number; count: number }>

   383→   383→      overallSentiment: number
   384→   384→    }>()

   386→   386→    summaries.forEach(s => {
   387→   387→      if (!teamMap.has(s.teamCode)) {
   388→   388→        teamMap.set(s.teamCode, {
   389→   389→          teamCode: s.teamCode,
   390→   390→          langSentiments: new Map(),
   391→   391→          totalPosts: 0,
   392→   392→          overallSentiment: 0,
   393→   393→        })
   394→   394→      }
   395→   395→      const team = teamMap.get(s.teamCode)!
   396→   396→      team.langSentiments.set(s.language, { avg: s.avgSentiment, count: s.postCount })
   397→   397→      team.totalPosts += s.postCount
   398→   398→      team.overallSentiment = (
   399→   399→        (team.overallSentiment * (team.totalPosts - s.postCount) + s.avgSentiment * s.postCount) /
   400→   400→        team.totalPosts
   401→   401→      )
   402→   402→    })
   403→   403→
   404→   404→    return [...teamMap.values()]
   405→   405→      .sort((a, b) => b.totalPosts - a.totalPosts)
   406→   406→      .slice(0, 10)
   407→   407→  }, [summaries])

   409→   409→  const toggleTeam = (teamCode: string) => {

   412→   412→      if (next.has(teamCode)) next.delete(teamCode)

   414→   414→      return next
   415→   415→    })
   416→   416→  }
   417→   417→
   418→   418→  const togglePost = (postId: string) => {
   419→   419→    setExpandedPosts(prev => {
   420→   420→      const next = new Set(prev)
   421→   421→      if (next.has(postId)) next.delete(postId)
   422→   422→      else next.add(postId)
   423→   423→      return next
   424→   424→    })
   425→   425→  }
   426→   426→
   427→   427→  const loadMore = () => {

   429→   429→  }
   430→   430→
   431→   431→  // ── Render ──
   432→   432→

   435→   435→      {/* ── 1. Header ──────────────────────────────────────── */}
   436→   436→      <motion.div
   437→   437→        initial={{ opacity: 0, y: 20 }}
   438→   438→        animate={{ opacity: 1, y: 0 }}
   439→   439→        className="flex items-start justify-between gap-3"
   440→   440→      >
   441→   441→        <div className="space-y-1">
   442→   442→          <div className="flex items-center gap-2.5">
   443→   443→            <h2 className="text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
