   158→type PosGroup = 'GK' | 'DEF' | 'MID' | 'FWD'
   159→function posGroup(p: { position: string }): PosGroup {
   160→  if (p.position === 'GK') return 'GK'
   161→  if (['CB', 'LB', 'RB'].includes(p.position)) return 'DEF'
   162→  if (['CM', 'CAM', 'CDM'].includes(p.position)) return 'MID'
   163→  return 'FWD' // LW, RW, ST, CF
   164→}

   166→/**
   167→ * A player merged across all stages where they appeared.
   168→ *
   169→ * `eliteScore` aggregates their hero performances (group-stage Elite pulse +
   170→ * R32 advanced-team buzz). `crisisScore` aggregates their villain performances
   171→ * (group-stage Crisis pulse + R32 eliminated-team buzz). A player can appear in
   172→ * BOTH lists (e.g. Weghorst was a Crisis pick in the group stage AND his team
   173→ * was eliminated in R32) — the two scores are tracked independently so the same
   174→ * verified player can anchor either XI.
   175→ */
   176→interface MergedPlayer {
   177→  name: string
   178→  nationCode: string
   179→  position: GroupStageEntry['position']
   180→  /** Hero-path composite score, or null if the player never appeared as a hero. */
   181→  eliteScore: number | null
   182→  /** Villain-path composite score, or null if the player never appeared as a villain. */
   183→  crisisScore: number | null
   184→  /** Best hero fact (for Elite XI matchInfo). */
   185→  eliteMatchInfo: string | null
   186→  /** Worst villain fact (for Crisis XI matchInfo). */
   187→  crisisMatchInfo: string | null
   188→  /** Most recent trend across stages. */
   189→  trend: Trend
   190→}
   191→
   192→// Trend bonus: rising = 80, stable = 50, falling = 20.
   193→function trendBonus(trend: Trend): number {
   194→  if (trend === 'rising') return 80
   195→  if (trend === 'stable') return 50
   196→  return 20
   197→}
   198→
   199→/**
   200→ * Merge the group-stage pools and the R32 VERIFIED_POOL into one list of
   201→ * MergedPlayer records, keyed by player name.
   202→ *
   203→ * Scoring (per the spec):
   204→ *   tournamentScore = (groupPulse × 0.4) + (r32Buzz × 0.4) + (trendBonus × 0.2)
   205→ *
   206→ * A player who only appeared in one stage gets a neutral 50 for the missing
   207→ * stage's component, so their composite is still comparable. The `tier` of each
   208→ * appearance (elite vs crisis) routes the score into `eliteScore` or
   209→ * `crisisScore` — a player can have both.
   210→ */
   211→function mergeAllPlayers(): Map<string, MergedPlayer> {
   212→  const map = new Map<string, MergedPlayer>()
   213→
   214→  const ensure = (name: string, nationCode: string, position: GroupStageEntry['position'], trend: Trend): MergedPlayer => {
   215→    let p = map.get(name)
   216→    if (!p) {
   217→      p = {
   218→        name, nationCode, position,
   219→        eliteScore: null, crisisScore: null,
   220→        eliteMatchInfo: null, crisisMatchInfo: null,
   221→        trend,
   222→      }
   223→      map.set(name, p)
   224→    }
   225→    return p
   226→  }
   227→
   228→  // ── Group-stage Elite appearances → eliteScore ──
   229→  for (const g of GROUP_STAGE_ELITE) {
   230→    const p = ensure(g.name, g.nationCode, g.position, g.trend)
   231→    // R32 buzz defaults to 50 (neutral) unless this player also has an R32 record.
   232→    const r32Buzz = 50
   233→    const score = Math.round(g.pulseScore * 0.4 + r32Buzz * 0.4 + trendBonus(g.trend) * 0.2)
   234→    p.eliteScore = score
   235→    p.eliteMatchInfo = g.matchInfo
   236→    p.trend = g.trend
   237→  }
   238→
   239→  // ── Group-stage Crisis appearances → crisisScore ──
   240→  for (const g of GROUP_STAGE_CRISIS) {
   241→    const p = ensure(g.name, g.nationCode, g.position, g.trend)
   242→    const r32Buzz = 50
   243→    const score = Math.round(g.pulseScore * 0.4 + r32Buzz * 0.4 + trendBonus(g.trend) * 0.2)
   244→    p.crisisScore = score
   245→    p.crisisMatchInfo = g.matchInfo
   246→    p.trend = g.trend
   247→  }
   248→
   249→  // ── R32 VERIFIED_POOL appearances ──
   250→  // Advanced-team players feed eliteScore; eliminated-team players feed crisisScore.
   251→  // Group-stage pulse defaults to 50 (neutral) unless this player also has a
   252→  // group-stage record (handled by reading the existing merged entry).
   253→  for (const r of VERIFIED_POOL) {
   254→    const position = r.position as GroupStageEntry['position']
   255→    // Derive an R32 trend: advancing teams trend up, eliminated teams trend down.
   256→    const r32Trend: Trend = r.teamStatus === 'advanced' ? 'rising' : 'falling'
   257→    const p = ensure(r.name, r.nationCode, position, r32Trend)
   258→
   259→    if (r.teamStatus === 'advanced') {
   260→      // Group pulse: use the existing elite group-stage pulse if present, else 50.
   261→      const groupEntry = GROUP_STAGE_ELITE.find(g => g.name === r.name)
   262→      const groupPulse = groupEntry ? groupEntry.pulseScore : 50
   263→      const score = Math.round(groupPulse * 0.4 + r.baselineBuzz * 0.4 + trendBonus(r32Trend) * 0.2)
   264→      // If the player already has an eliteScore from the group stage, prefer the
   265→      // HIGHER of the two (their best tournament moment). Update matchInfo to
   266→      // the R32 fact if it's more decisive (R32 facts cite the knockout result).
   267→      if (p.eliteScore === null || score > p.eliteScore) {
   268→        p.eliteScore = score
   269→      }
   270→      if (!p.eliteMatchInfo) p.eliteMatchInfo = r.r32Fact
   271→      else if (r.r32Fact && !p.eliteMatchInfo.includes('R32')) p.eliteMatchInfo = r.r32Fact
   272→    } else if (r.teamStatus === 'eliminated') {
   273→      const groupEntry = GROUP_STAGE_CRISIS.find(g => g.name === r.name)
   274→      const groupPulse = groupEntry ? groupEntry.pulseScore : 50
   275→      const score = Math.round(groupPulse * 0.4 + r.baselineBuzz * 0.4 + trendBonus(r32Trend) * 0.2)
   276→      // For crisis, prefer the LOWER score (their worst tournament moment).
   277→      if (p.crisisScore === null || score < p.crisisScore) {
   278→        p.crisisScore = score
   279→      }
   280→      if (!p.crisisMatchInfo) p.crisisMatchInfo = r.r32Fact
   281→      else if (r.r32Fact && r.r32Fact.includes('ELIMINATED')) p.crisisMatchInfo = r.r32Fact
   282→    }
   283→    // 'upcoming' R32 teamStatus is ignored — those players have no result yet.
   284→  }
   285→
   286→  // ── KNOCKOUT_HEROES appearances ──
   287→  // These are verified QF/SF/3rd/Final hero/villain performances. The pulseScore
   288→  // here is app-internal (0-100) and reflects the knockout-stage moment only — it
   289→  // is combined with the existing group/R32 score (if any) using the same
   290→  // best-moment routing: Elite keeps the HIGHER score, Crisis keeps the LOWER.
   291→  for (const k of KNOCKOUT_HEROES) {
   292→    const p = ensure(k.name, k.nationCode, k.position, k.trend)
   293→    // Knockout buzz: high for heroes (esp. Final winners), low for villains.
   294→    // Use the entry's pulseScore directly as the "buzz" component.
   295→    const knockoutBuzz = k.pulseScore
   296→    // Group-stage pulse: use the SAME-TIER group-stage pulse if present, else 50.
   297→    // (Don't cross-contaminate — e.g. Lamine Yamal was a Crisis pick in the group
   298→    // stage but an Elite hero in the knockouts; his eliteScore should NOT inherit
   299→    // his group-stage crisis pulse of 30.)
   300→    const groupEliteEntry = GROUP_STAGE_ELITE.find(g => g.name === k.name)
   301→    const groupCrisisEntry = GROUP_STAGE_CRISIS.find(g => g.name === k.name)
   302→    const r32Entry = VERIFIED_POOL.find(r => r.name === k.name)
   303→    let groupPulse: number
   304→    if (k.tier === 'elite') {
   305→      groupPulse = groupEliteEntry?.pulseScore ?? (r32Entry && r32Entry.teamStatus === 'advanced' ? 50 : 50)
   306→    } else {
   307→      groupPulse = groupCrisisEntry?.pulseScore ?? (r32Entry && r32Entry.teamStatus === 'eliminated' ? 50 : 50)
   308→    }
   309→    const score = Math.round(groupPulse * 0.4 + knockoutBuzz * 0.4 + trendBonus(k.trend) * 0.2)
   310→
   311→    if (k.tier === 'elite') {
   312→      // Hero: keep the HIGHER score (their best tournament moment).
   313→      if (p.eliteScore === null || score > p.eliteScore) {
   314→        p.eliteScore = score
   315→      }
   316→      // Prefer the knockout matchInfo (more decisive) — but only overwrite if the
   317→      // existing one doesn't already cite the Final/SF.
   318→      if (!p.eliteMatchInfo) p.eliteMatchInfo = k.matchInfo
   319→      else if (!p.eliteMatchInfo.includes('FINAL') && !p.eliteMatchInfo.includes('SF') && !p.eliteMatchInfo.includes('3rd')) {
   320→        p.eliteMatchInfo = k.matchInfo
   321→      }
   322→      p.trend = k.trend
   323→    } else {
   324→      // Crisis: keep the LOWER score (their worst tournament moment).
   325→      if (p.crisisScore === null || score < p.crisisScore) {
   326→        p.crisisScore = score
   327→      }
   328→      if (!p.crisisMatchInfo) p.crisisMatchInfo = k.matchInfo
   329→      else if (!p.crisisMatchInfo.includes('FINAL') && !p.crisisMatchInfo.includes('SF') && !p.crisisMatchInfo.includes('3rd')) {
   330→        p.crisisMatchInfo = k.matchInfo
   331→      }
   332→      p.trend = k.trend
   333→    }
   334→  }
   335→
   336→  return map
   337→}

   341→export interface RetroPick {
   342→  /** Stable id derived from name + position group. */
   343→  id: string
   344→  name: string
   345→  nationCode: string
   346→  position: string
   347→  /** App-internal composite 0-100. NOT a verified statistic. */
   348→  tournamentScore: number
   349→  /** Verified match fact (copied verbatim from the source pool). */
   350→  matchInfo: string | null
   351→  trend: Trend
   352→}

   354→export interface RetroSide {
   355→  formation: string
   356→  players: RetroPick[]
   357→}

   359→export interface TournamentRetroResult {
   360→  elite: RetroSide
   361→  crisis: RetroSide
   362→  generatedAt: string
   363→}
   364→
   365→// ── Formation slot config (4-3-3) ────────────────────────────────────────────
   366→const FORMATION = '4-3-3'
   367→const SLOTS: { group: PosGroup; count: number }[] = [
   368→  { group: 'GK', count: 1 },
   369→  { group: 'DEF', count: 4 },
   370→  { group: 'MID', count: 3 },
   371→  { group: 'FWD', count: 3 },
   372→]
   373→const XI_SIZE = SLOTS.reduce((n, s) => n + s.count, 0) // 11
   374→
   375→/**
   376→ * Pick an XI (4-3-3) from a candidate pool.
   377→ *
   378→ * @param candidates  players with a non-null score for this side
   379→ * @param descending  true for Elite (highest score first), false for Crisis (lowest first)
   380→ *
   381→ * ANTI-HALLUCINATION: if a position group is short, we fill from the next-best
   382→ * available verified player. If the entire pool is exhausted, the slot is
   383→ * returned as a null placeholder (name: 'N/A') — we never fabricate.
   384→ */
   385→function pickXI(
   386→  candidates: MergedPlayer[],
   387→  scoreOf: (p: MergedPlayer) => number | null,
   388→  descending: boolean,
   389→): RetroPick[] {
   390→  // Filter to players that actually have a score for this side, then sort.
   391→  const eligible = candidates
   392→    .filter(p => scoreOf(p) !== null)
   393→    .sort((a, b) => {
   394→      const sa = scoreOf(a)!
   395→      const sb = scoreOf(b)!
   396→      return descending ? sb - sa : sa - sb
   397→    })
   398→
   399→  const used = new Set<string>()
   400→  const picks: (MergedPlayer | null)[] = []
   401→
   402→  // Pass 1: fill each position-group slot from players of that group.
   403→  for (const slot of SLOTS) {
   404→    let filled = 0
   405→    for (const p of eligible) {
   406→      if (filled >= slot.count) break
   407→      if (used.has(p.name)) continue
   408→      if (posGroup(p) === slot.group) {
   409→        picks.push(p)
   410→        used.add(p.name)
   411→        filled++
   412→      }
   413→    }
   414→    // Pad with nulls if this group was under-filled.
   415→    while (filled < slot.count) {
   416→      picks.push(null)
   417→      filled++
   418→    }
   419→  }
   420→
   421→  // Pass 2: replace any null placeholders with the next-best unused verified
   422→  // player (regardless of position group) so the XI fills 11 real names.
   423→  for (let i = 0; i < picks.length; i++) {
   424→    if (picks[i] !== null) continue
   425→    const next = eligible.find(p => !used.has(p.name))
   426→    if (next) {
   427→      picks[i] = next
   428→      used.add(next.name)
   429→    }
   430→    // If no unused player remains, the slot stays null → 'N/A'.
   431→  }
   432→
   433→  // Map to RetroPick, assigning an order within the formation.
   434→  return picks.map((p, i) => {
   435→    const slotIndex = i
   436→    const group = SLOTS.reduce<{ group: PosGroup; count: number } | null>((acc, s) => {
   437→      if (acc) return acc
   438→      return s
   439→    }, null)
   440→    void group // (slot metadata derived from index instead)
   441→    if (!p) {
   442→      return {
   443→        id: `na-${slotIndex}`,
   444→        name: 'N/A',
   445→        nationCode: '',
   446→        position: '',
   447→        tournamentScore: 0,
   448→        matchInfo: null,
   449→        trend: 'stable' as Trend,
   450→      }
   451→    }
   452→    const score = scoreOf(p)!
   453→    const matchInfo = descending ? p.eliteMatchInfo : p.crisisMatchInfo
   454→    return {
   455→      id: `${p.name.toLowerCase().replace(/[^a-z0-9]+/g, '-')}-${slotIndex}`,
   456→      name: p.name,
   457→      nationCode: p.nationCode,
   458→      position: p.position,
   459→      tournamentScore: score,
   460→      matchInfo,
   461→      trend: p.trend,
   462→    }
   463→  })
   464→}
   465→
   466→// ── Main entry point ─────────────────────────────────────────────────────────
   467→
   468→/**
   469→ * Compute the Team of the Tournament Retro (Elite XI + Crisis XI).
   470→ *
   471→ * Pure function — no DB, no SDK. The result is deterministic for a given
   472→ * VERIFIED_POOL + group-stage pool, so the API route caches it for 1 hour.
   473→ */
   474→export function computeTournamentRetro(): TournamentRetroResult {
   475→  const merged = mergeAllPlayers()
   476→  const all = Array.from(merged.values())
   477→
   478→  const elite = pickXI(all, p => p.eliteScore, true)
   479→  const crisis = pickXI(all, p => p.crisisScore, false)
   480→
   481→  return {
   482→    elite: { formation: FORMATION, players: elite },
   483→    crisis: { formation: FORMATION, players: crisis },
   484→    generatedAt: new Date().toISOString(),
   485→  }
   486→}

   488→// ── Anti-hallucination self-check (exported for tests / dev scripts) ─────────
   489→
   490→/**
   491→ * Returns the set of every player name that is eligible to appear in the retro.
   492→ * Used by the API route's self-check to guarantee no fabricated name slipped in.
   493→ */
   494→export function getAllVerifiedNames(): Set<string> {
   495→  const names = new Set<string>()
   496→  for (const g of GROUP_STAGE_ELITE) names.add(g.name)
   497→  for (const g of GROUP_STAGE_CRISIS) names.add(g.name)
   498→  for (const r of VERIFIED_POOL) names.add(r.name)
   499→  for (const k of KNOCKOUT_HEROES) names.add(k.name)
   500→  return names
   501→}
