   121→function getTrendIcon(trend: string) {
   122→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   123→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   124→  return <Minus className="size-3 text-[#FF6B35]" />
   125→}

   127→function getSentimentColor(score: number) {
   128→  if (score >= 80) return 'text-[#10B981]'
   129→  if (score >= 50) return 'text-[#FF6B35]'
   130→  return 'text-[#EF4444]'
   131→}
   132→
   133→function getSentimentBg(score: number) {
   134→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   135→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   136→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'

   930→                    <span className="text-2xl">{activeView === 'elite' ? '🌟' : '⚠️'}</span>

   994→            </motion.div>

   996→        </>
