   150→// ── Helpers ──────────────────────────────────────────────────

   152→function getFlag(nationCode: string): string {
   153→  const team = findNationalTeam(nationCode)
   154→  return team?.flag ?? '🏳️'
   155→}
   156→
   157→function getTrendIcon(trend: string) {
   158→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   159→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   160→  return <Minus className="size-3 text-[#FF6B35]" />
   161→}
   162→
   163→function getSentimentColor(score: number) {
   164→  if (score >= 80) return 'text-[#10B981]'
   165→  if (score >= 50) return 'text-[#FF6B35]'
   166→  return 'text-[#EF4444]'
   167→}
   168→
   169→function getSentimentBg(score: number) {
   170→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   171→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   172→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   173→}
   174→
   175→function getProgressClass(score: number) {
   176→  if (score >= 80) return 'progress-emerald'
   177→  if (score >= 50) return 'progress-amber'
   178→  return 'progress-red'
   179→}
   180→
   181→// Face emoji mapping for /10 rating system
   182→function getPulseFaceEmoji(pulseScore: number): string {
   183→  if (pulseScore >= 90) return '🤩'
   184→  if (pulseScore >= 70) return '😊'
   185→  if (pulseScore >= 50) return '😐'
   186→  if (pulseScore >= 30) return '😟'
   187→  return '😵'
   188→}
   189→
   190→function getRatingColor(rating: number): string {
   191→  if (rating >= 9) return '#10B981'
   192→  if (rating >= 7) return '#6C2BD9'
   193→  if (rating >= 5) return '#FF6B35'
   194→  if (rating >= 3) return '#EF4444'
   195→  return '#DC2626'
   196→}

   765→function FormationPlayerCard({
   766→  player,
   767→  type,
   768→  stageStatus,
   769→}: {
   770→  player: WCSelectionPlayer
   771→  type: 'elite' | 'crisis'
   772→  stageStatus: string
   773→}) {
