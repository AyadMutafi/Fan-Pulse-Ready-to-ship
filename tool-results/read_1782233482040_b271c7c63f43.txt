   216→interface FanVoteAgg {

   222→function HomeTab() {
   223→  const { t } = useLanguage()
   224→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   225→  const [apiMatches, setApiMatches] = useState<Array<{
   226→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   227→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   228→  }>>([])
   229→
   230→  // Fan vote state
   231→  const [sessionId, setSessionId] = useState<string>('')
   232→  const [fanVotes, setFanVotes] = useState<FanVoteAgg[]>([])
   233→  const [myVotes, setMyVotes] = useState<Array<{ teamCode: string; score: number }>>([])
   234→  const [votesLoading, setVotesLoading] = useState(true)
   235→  const [selectedVoteTeam, setSelectedVoteTeam] = useState<string | null>(null)
   236→  const [submitting, setSubmitting] = useState(false)
   237→  const [toast, setToast] = useState<{ msg: string; emoji: string } | null>(null)
   238→
   239→  useEffect(() => {
   240→    async function fetchMatches() {
   241→      try {
   242→        const res = await fetch('/api/matches?league=WC')
   243→        if (res.ok) {
   244→          const data = await res.json()
   245→          const mapped = (data.matches || []).map((m: any) => ({
   246→            id: m.id,
   247→            home: m.homeTeam.code,
   248→            away: m.awayTeam.code,
   249→            homeFlag: m.homeTeam.flag || '',
   250→            awayFlag: m.awayTeam.flag || '',
   251→            score: m.score,
   252→            homeSentiment: Math.round(m.homeTeam.sentiment),
   253→            awaySentiment: Math.round(m.awayTeam.sentiment),
   254→            live: m.status === 'live',
   255→            league: m.league === 'WC' ? `WC Group ${m.group}` : m.league,
   256→          }))
   257→          setApiMatches(mapped)
   258→        }
   259→      } catch (err) {
   260→        console.error('Failed to fetch matches:', err)
   261→      }
   262→    }
   263→    fetchMatches()
   264→  }, [])
   265→
   266→  // SSR-safe: read/create sessionId inside useEffect, never during render.
   267→  useEffect(() => {
   268→    try {
   269→      const existing = typeof window !== 'undefined' ? window.localStorage.getItem('fan_session_id') : null
   270→      if (existing && existing.length > 0) {
   271→        setSessionId(existing)
   272→        return
   273→      }
   274→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
   275→        ? crypto.randomUUID()
   276→        : `anon-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`
   277→      if (typeof window !== 'undefined') {
   278→        window.localStorage.setItem('fan_session_id', newId)
   279→      }
   280→      setSessionId(newId)
   281→    } catch {
   282→      setSessionId(`anon-${Date.now()}`)
   283→    }
   284→  }, [])
   285→
   286→  // Fetch fan votes whenever sessionId changes.
   287→  useEffect(() => {
   288→    if (!sessionId) return
   289→    let cancelled = false
   290→    async function loadVotes() {
   291→      setVotesLoading(true)
   292→      try {
   293→        const res = await fetch(`/api/fan-vote?session=${encodeURIComponent(sessionId)}`)
   294→        if (!res.ok) return
   295→        const data = await res.json()
   296→        if (cancelled) return
   297→        setFanVotes(Array.isArray(data.votes) ? data.votes : [])
   298→        setMyVotes(Array.isArray(data.myVotes) ? data.myVotes : [])
   299→      } catch (err) {
   300→        console.error('Failed to fetch fan votes:', err)
   301→      } finally {
   302→        if (!cancelled) setVotesLoading(false)
   303→      }
   304→    }
   305→    loadVotes()
   306→    return () => { cancelled = true }
   307→  }, [sessionId])
   308→
   309→  const totalVoteCount = fanVotes.reduce((sum, v) => sum + (v.count || 0), 0)
   310→
   311→  const handleVote = async (teamCode: string, score: number) => {
   312→    if (!sessionId || submitting) return
   313→    setSubmitting(true)
   314→    // Optimistic update: bump my vote immediately
   315→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   316→    setMyVotes(prev => {
   317→      const without = prev.filter(v => v.teamCode !== teamCode)
   318→      return [...without, { teamCode, score }]
   319→    })
   320→    setFanVotes(prev => {
   321→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   322→      if (idx === -1) {
   323→        return [...prev, { teamCode, score, count: 1 }]
   324→      }
   325→      const next = [...prev]
   326→      const wasMine = prevMyVote !== null
   327→      // Adjust running average: if I had voted, swap my old score for the new.
   328→      const current = next[idx]
   329→      const totalScore = current.score * current.count
   330→      const newScore = wasMine
   331→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   332→        : Math.round((totalScore + score) / (current.count + 1))
   333→      const newCount = wasMine ? current.count : current.count + 1
   334→      next[idx] = { teamCode, score: newScore, count: newCount }
   335→      return next
   336→    })
   337→    setSelectedVoteTeam(null)
   338→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   339→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   340→    setTimeout(() => setToast(null), 2500)
   341→    try {
   342→      await fetch('/api/fan-vote', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({ teamCode, score, sessionId }),
   346→      })
   347→    } catch (err) {
   348→      console.error('Failed to submit vote:', err)
   349→      // Revert on failure
   350→      setMyVotes(prev => {
   351→        const without = prev.filter(v => v.teamCode !== teamCode)
   352→        if (prevMyVote !== null) return [...without, { teamCode, score: prevMyVote }]
   353→        return without
   354→      })
   355→      setToast({ msg: 'Vote failed — please retry', emoji: '⚠️' })
   356→    } finally {
   357→      setSubmitting(false)
   358→    }
   359→  }
   360→
   361→  const filteredMatches = (matchFilter === 'ALL'
   362→    ? apiMatches
   363→    : apiMatches.filter(m => m.league.startsWith('WC'))
   364→  ).slice(0, 24) // Limit to 24 cards max for performance
   365→
   366→  const fanVoteIntelText = totalVoteCount === 0
   367→    ? 'Be the first to vote in the Fan Mood section below'
   368→    : `${totalVoteCount.toLocaleString()} fan votes tallied for World Cup 2026 Group Stage`
   369→
   370→  const moodTeamEntries = FAN_MOOD_TEAM_CODES.map(code => {
   371→    const team = NATIONAL_TEAMS.find(t => t.code === code)
   372→    const vote = fanVotes.find(v => v.teamCode === code)
   373→    const myVote = myVotes.find(v => v.teamCode === code)
   374→    return {
   375→      code,
   376→      flag: team?.flag ?? '🏳️',
   377→      name: team?.name ?? code,
   378→      score: vote?.score ?? 50,
   379→      count: vote?.count ?? 0,
   380→      myVote: myVote?.score ?? null,
   381→    }
   382→  })
   383→
   384→  return (
   385→    <div className="space-y-6">
   386→      {/* Hero */}
   387→      <motion.div
   388→        initial={{ opacity: 0, y: 20 }}
   389→        animate={{ opacity: 1, y: 0 }}
   390→        transition={{ duration: 0.5 }}
   391→        className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-[#6C2BD9]/8 via-white dark:via-[#1A1A1A] to-[#FF6B35]/5 p-6 border border-[#6C2BD9]/10 dark:border-[#6C2BD9]/20"
   392→      >
   393→        <div className="absolute -right-8 -top-8 size-32 rounded-full bg-[#6C2BD9]/5 blur-3xl" />
   394→        <div className="absolute -bottom-8 -left-8 size-32 rounded-full bg-[#FF6B35]/5 blur-3xl" />
   395→        <h2 className="relative text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
   396→          {t('home.your_pulse')} <span className="text-[#6C2BD9]">⚡</span>
   397→        </h2>
   398→        <p className="relative mt-2 text-sm text-[#666] dark:text-[#CCCCCC]">
   399→          {t('home.mood_desc')}
   400→        </p>
   401→        <div className="relative mt-4 flex items-center gap-3">
   402→          <div className="flex items-center gap-1.5 rounded-full bg-[#10B981]/10 px-3 py-1.5 text-xs font-semibold text-[#10B981]">
   403→            <Zap className="size-3.5" />
   404→            78% {t('home.positive')}
   405→          </div>
   406→          <div className="flex items-center gap-1.5 rounded-full bg-[#EF4444]/10 px-3 py-1.5 text-xs font-semibold text-[#EF4444]">
   407→            <Activity className="size-3.5" />
   408→            {apiMatches.filter(m => m.live).length} {t('home.live')}
   409→          </div>
   410→        </div>
   411→      </motion.div>
   412→
   413→      {/* Featured Matches */}
   414→      <div>
   415→        <div className="mb-3 flex items-center justify-between">
   416→          <h3 className="text-xs font-bold uppercase tracking-wider text-[#666] dark:text-[#CCCCCC]">
   417→            {t('home.featured')}
   418→          </h3>
   419→          <div className="flex gap-1.5">
   420→            {(['WC', 'ALL'] as const).map((filter) => (
   421→              <button
   422→                key={filter}
   423→                onClick={() => setMatchFilter(filter)}
   424→                className={`
   425→                  rounded-full px-3 py-1 text-[10px] font-bold transition-all duration-200
