   124→function getTrendIcon(trend: string) {
   125→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   126→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   127→  return <Minus className="size-3 text-[#FF6B35]" />

   129→
   130→function getSentimentColor(score: number) {
   131→  if (score >= 80) return 'text-[#10B981]'
   132→  if (score >= 50) return 'text-[#FF6B35]'
   133→  return 'text-[#EF4444]'
   134→}
   135→
   136→function getSentimentBg(score: number) {
   137→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   138→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   139→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   140→}
   141→
   142→function getProgressClass(score: number) {
   143→  if (score >= 80) return 'progress-emerald'
   144→  if (score >= 50) return 'progress-amber'
   145→  return 'progress-red'
   146→}

   148→// Face emoji mapping for /10 rating system
   149→function getPulseFaceEmoji(pulseScore: number): string {
   150→  if (pulseScore >= 90) return '🤩'
   151→  if (pulseScore >= 70) return '😊'
   152→  if (pulseScore >= 50) return '😐'
   153→  if (pulseScore >= 30) return '😟'
   154→  return '😵'
   155→}

   157→function getRatingColor(rating: number): string {
   158→  if (rating >= 9) return '#10B981'
   159→  if (rating >= 7) return '#6C2BD9'
   160→  if (rating >= 5) return '#FF6B35'
   161→  if (rating >= 3) return '#EF4444'
   162→  return '#DC2626'
