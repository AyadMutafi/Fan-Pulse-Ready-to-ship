   222→// ── Seed handler ─────────────────────────────────────────────────────────────
   223→// AUTH REQUIRED: this route wipes and re-creates most tables. Only the admin
   224→// may call it. The frontend no longer calls this route; it runs once at deploy.
   225→export async function POST(request: Request) {
   226→  // ── Auth gate ──
   227→  if (!isAdminAuthorized(request)) {
   228→    return unauthorizedResponse()
   229→  }
   230→
   231→  try {
   232→    // ── Check if already seeded (skip if ?force=true) ──
   233→    const { searchParams } = new URL(request.url)
   234→    const force = searchParams.get('force') === 'true'
   235→    if (!force) {
   236→      const existingStages = await db.wCStage.count()
   237→      if (existingStages > 0) {
   238→        return NextResponse.json({ message: 'Already seeded', stages: existingStages })
   239→      }
   240→    }
   241→
   242→    // ── 1. Clean up existing data ──
   243→    // NOTE: fanVote is intentionally NOT wiped — user votes are the product and
   244→    // must survive re-seeds. pulseBreakdown + sentimentSummary are safe to wipe
   245→    // because they are fully recomputed by the pulse engine below.
   246→    await db.wCSelectionPlayer.deleteMany()
   247→    await db.wCSelection.deleteMany()
   248→    await db.wCStage.deleteMany()
   249→    await db.match.deleteMany()
   250→    await db.nationalTeam.deleteMany()
   251→    await db.pulseBreakdown.deleteMany().catch((e) => {
   252→      console.warn('pulseBreakdown deleteMany failed (continuing):', e)
   253→    })
   254→    await db.sentimentSummary.deleteMany().catch((e) => {
   255→      console.warn('sentimentSummary deleteMany failed (continuing):', e)
   256→    })
   257→    // Intentionally NOT deleting fanVote — user votes are preserved.
   258→
   259→    // ── 2. Create 7 stages — Group Stage completed, Round of 32 live ──
   260→    const stagesData = [
   261→      { name: 'Group Stage', nameAr: 'دور المجموعات', order: 1, status: 'completed', key: 'group-stage', startedAt: new Date('2026-06-11'), completedAt: new Date('2026-06-27') },
   262→      { name: 'Round of 32', nameAr: 'دور الـ 32', order: 2, status: 'live', key: 'round-of-32', startedAt: new Date('2026-06-28'), completedAt: undefined as Date | undefined },
   263→      { name: 'Round of 16', nameAr: 'دور الـ 16', order: 3, status: 'upcoming', key: 'round-of-16', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   264→      { name: 'Quarter Finals', nameAr: 'ربع النهائي', order: 4, status: 'upcoming', key: 'quarter-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   265→      { name: 'Semi Finals', nameAr: 'نصف النهائي', order: 5, status: 'upcoming', key: 'semi-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },

   269→

   271→    for (const sd of stagesData) {
   272→      const stage = await db.wCStage.create({
   273→        data: {
   274→          name: sd.name,
   275→          nameAr: sd.nameAr,
   276→          order: sd.order,
   277→          status: sd.status,
   278→          startedAt: sd.startedAt,
   279→          completedAt: sd.completedAt,
   280→        }
   281→      })
   282→      stages.push({ ...stage, key: sd.key })
   283→    }
   284→
   285→    // ── 3. Seed national teams (48 WC 2026 teams) ──
   286→    const { NATIONAL_TEAMS } = await import('@/lib/national-teams')
   287→    for (const team of NATIONAL_TEAMS) {
   288→      await db.nationalTeam.upsert({
   289→        where: { code: team.code },
   290→        update: {},
   291→        create: {
   292→          id: team.id,
   293→          name: team.name,
   294→          nameAr: team.nameAr,
   295→          code: team.code,
   296→          flag: team.flag,
   297→          group: team.group,
   298→          fifaRank: team.fifaRank,
   299→          primaryColor: team.primaryColor,
   300→          region: team.region,
   301→        }
   302→      })
   303→    }
   304→
   305→    // ── 4. Seed VERIFIED WC 2026 matches ──
   306→    let matchesCreated = 0
   307→    for (const m of MATCHES_DATA) {
   308→      const homeInfo = TEAM_INFO[m.homeCode]
   309→      const awayInfo = TEAM_INFO[m.awayCode]
   310→      if (!homeInfo || !awayInfo) {
   311→        console.warn(`Skipping match: unknown team code ${m.homeCode} or ${m.awayCode}`)
   312→        continue
   313→      }
   314→
   315→      await db.match.create({
   316→        data: {
   317→          homeTeamCode: m.homeCode,
   318→          homeTeamName: homeInfo.name,
   319→          homeTeamFlag: homeInfo.flag,
   320→          awayTeamCode: m.awayCode,
   321→          awayTeamName: awayInfo.name,
   322→          awayTeamFlag: awayInfo.flag,
   323→          homeScore: m.homeScore,
   324→          awayScore: m.awayScore,
   325→          status: m.status,
   326→          league: 'WC',
   327→          group: m.group,

   329→          homeSentiment: m.homeSentiment,
   330→          awaySentiment: m.awaySentiment,
   331→        }

   335→
   336→    // ── 5. Seed Elite & Crisis selections (group-stage only) ──
   337→    // R32 has NO Elite/Crisis XI — the stage is still live (6 matches Jul 3).
   338→    for (const stage of stages) {
   339→      const elitePool = ELITE_PLAYERS[stage.key]
   340→      const crisisPool = CRISIS_PLAYERS[stage.key]
   341→

   344→          data: {
   345→            type: 'elite',

   349→          }
   350→        })
   351→
   352→        for (const p of elitePool) {
   353→          await db.wCSelectionPlayer.create({
   354→            data: {
   355→              selectionId: eliteSelection.id,
   356→              playerName: p.name,
   357→              nationCode: p.nationCode,
   358→              position: p.position,
   359→              pulseScore: p.pulseScore,
   360→              sentiment: p.sentiment,
   361→              trend: p.trend,
   362→              isLive: p.isLive,
   363→              matchInfo: p.matchInfo,
   364→              order: p.order,
   365→            }
   366→          })
   367→        }
   368→      }
   369→
   370→      if (crisisPool && crisisPool.length > 0) {
   371→        const crisisSelection = await db.wCSelection.create({
   372→          data: {
   373→            type: 'crisis',
   374→            stageId: stage.id,
   375→            formation: '4-3-3',
   376→            locked: stage.key === 'group-stage',
   377→          }
   378→        })
   379→
   380→        for (const p of crisisPool) {
   381→          await db.wCSelectionPlayer.create({
   382→            data: {
   383→              selectionId: crisisSelection.id,
   384→              playerName: p.name,
   385→              nationCode: p.nationCode,
   386→              position: p.position,
   387→              pulseScore: p.pulseScore,
   388→              sentiment: p.sentiment,
   389→              trend: p.trend,
   390→              isLive: p.isLive,
   391→              matchInfo: p.matchInfo,
   392→              order: p.order,
   393→            }
   394→          })
   395→        }
   396→      }
   397→    }
   398→
   399→    // ── 6. Run the Pulse Score engine (group-stage players only at this point) ──
   400→    // R32 is seeded AFTER the pulse engine so the engine's 4-component computed
   401→    // scores don't clobber the R32 web-buzz scores. R32 pulseScore = buzzScore.
   402→    let pulseResult: { playersComputed: number; breakdownsWritten: number; errors: string[] } | null = null
   403→    try {
   404→      pulseResult = await computeAllPulseScores(getDb())
   405→    } catch (err) {
   406→      console.error('Pulse engine failed during seed:', err)
   407→    }
   408→
   409→    // ── 7. Seed R32 Elite XI + Crisis XI from the VERIFIED_POOL baseline ──────
   410→    // R32 teams are seeded from the VERIFIED_POOL baseline (real web-buzz captured
   411→    // 2026-07-02). A background cron refreshes with live web_search periodically —
   412→    // see /api/world-cup/r32-cron. The pool contains only web-verified players;
   413→    // see src/lib/r32-buzz-ranker.ts for the anti-hallucination contract.
   414→    // Seeded AFTER the pulse engine so R32 buzz scores are preserved.
   415→    let r32Seeded: { elite: number; crisis: number; buzzSource: string } | null = null
   416→    try {
   417→      const r32Stage = stages.find((s) => s.key === 'round-of-32')
   418→      if (r32Stage) {
   419→        const r32Result = await rankR32Teams(false) // baseline, no SDK calls — instant
   420→        await seedR32Teams(db, r32Result, r32Stage.id)
   421→        r32Seeded = {
   422→          elite: r32Result.elite.length,
   423→          crisis: r32Result.crisis.length,
   424→          buzzSource: r32Result.buzzSource,
   425→        }
   426→      }
   427→    } catch (err) {
   428→      console.error('R32 seed failed (continuing with group stage only):', err)
   429→    }
   430→
   431→    return NextResponse.json({
   432→      success: true,
   433→      message: 'Database seeded with VERIFIED World Cup 2026 data (see /VERIFIED_DATA.md for sources)',
   434→      stages: stages.length,
   435→      nationalTeams: NATIONAL_TEAMS.length,
   436→      matches: matchesCreated,
   437→      matchdayInfo: 'Group Stage completed Jun 27 · Round of 32 LIVE (10/16 played, 6 upcoming Jul 3) · 24 group-stage matches + 16 R32 matches seeded',
   438→      pulse: pulseResult,
   439→      r32: r32Seeded,
   440→    })
   441→  } catch (error) {
   442→    console.error('Seed failed:', error)
   443→    return NextResponse.json({ error: 'Seed failed', details: String(error) }, { status: 500 })
   444→  }
   445→}
