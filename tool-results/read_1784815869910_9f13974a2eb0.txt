   241→    if (!force) {
   242→      const existingStages = await db.wCStage.count()
   243→      if (existingStages > 0) {
   244→        return NextResponse.json({ message: 'Already seeded', stages: existingStages })
   245→      }
   246→    }
   247→
   248→    // ── 1. Clean up existing data ──
   249→    // NOTE: fanVote is intentionally NOT wiped — user votes are the product and
   250→    // must survive re-seeds. pulseBreakdown + sentimentSummary are safe to wipe
   251→    // because they are fully recomputed by the pulse engine below.
   252→    await db.wCSelectionPlayer.deleteMany()
   253→    await db.wCSelection.deleteMany()
   254→    await db.wCStage.deleteMany()
   255→    await db.match.deleteMany()
   256→    await db.nationalTeam.deleteMany()
   257→    await db.pulseBreakdown.deleteMany().catch((e) => {
   258→      console.warn('pulseBreakdown deleteMany failed (continuing):', e)
   259→    })
   260→    await db.sentimentSummary.deleteMany().catch((e) => {
   261→      console.warn('sentimentSummary deleteMany failed (continuing):', e)
   262→    })
   263→    // Intentionally NOT deleting fanVote — user votes are preserved.
   264→
   265→    // ── 2. Create 7 stages — Group Stage completed, Round of 32 live ──
   266→    const stagesData = [
   267→      { name: 'Group Stage', nameAr: 'دور المجموعات', order: 1, status: 'completed', key: 'group-stage', startedAt: new Date('2026-06-11'), completedAt: new Date('2026-06-27') },
   268→      { name: 'Round of 32', nameAr: 'دور الـ 32', order: 2, status: 'live', key: 'round-of-32', startedAt: new Date('2026-06-28'), completedAt: undefined as Date | undefined },
   269→      { name: 'Round of 16', nameAr: 'دور الـ 16', order: 3, status: 'upcoming', key: 'round-of-16', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   270→      { name: 'Quarter Finals', nameAr: 'ربع النهائي', order: 4, status: 'upcoming', key: 'quarter-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   271→      { name: 'Semi Finals', nameAr: 'نصف النهائي', order: 5, status: 'upcoming', key: 'semi-finals', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   272→      { name: 'Third Place', nameAr: 'مركز الثالث', order: 6, status: 'upcoming', key: 'third-place', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   273→      { name: 'Final', nameAr: 'النهائي', order: 7, status: 'upcoming', key: 'final', startedAt: undefined as Date | undefined, completedAt: undefined as Date | undefined },
   274→    ]
   275→
   276→    const stages: Array<{ id: string; name: string; order: number; status: string; key: string }> = []
   277→    for (const sd of stagesData) {
   278→      const stage = await db.wCStage.create({
   279→        data: {
   280→          name: sd.name,
   281→          nameAr: sd.nameAr,
   282→          order: sd.order,
   283→          status: sd.status,
   284→          startedAt: sd.startedAt,
   285→          completedAt: sd.completedAt,
   286→        }
   287→      })
   288→      stages.push({ ...stage, key: sd.key })
   289→    }
   290→
   291→    // ── 3. Seed national teams (48 WC 2026 teams) ──
   292→    const { NATIONAL_TEAMS } = await import('@/lib/national-teams')
   293→    for (const team of NATIONAL_TEAMS) {
   294→      await db.nationalTeam.upsert({
   295→        where: { code: team.code },
   296→        update: {},
   297→        create: {
   298→          id: team.id,
   299→          name: team.name,
   300→          nameAr: team.nameAr,
   301→          code: team.code,
   302→          flag: team.flag,
   303→          group: team.group,
   304→          fifaRank: team.fifaRank,
   305→          primaryColor: team.primaryColor,
   306→          region: team.region,
   307→        }
   308→      })
   309→    }
   310→
   311→    // ── 4. Seed VERIFIED WC 2026 matches ──
   312→    let matchesCreated = 0
   313→    for (const m of MATCHES_DATA) {
   314→      const homeInfo = TEAM_INFO[m.homeCode]
   315→      const awayInfo = TEAM_INFO[m.awayCode]
   316→      if (!homeInfo || !awayInfo) {
   317→        console.warn(`Skipping match: unknown team code ${m.homeCode} or ${m.awayCode}`)
   318→        continue
   319→      }
   320→
   321→      await db.match.create({
   322→        data: {
   323→          homeTeamCode: m.homeCode,
   324→          homeTeamName: homeInfo.name,
   325→          homeTeamFlag: homeInfo.flag,
   326→          awayTeamCode: m.awayCode,
   327→          awayTeamName: awayInfo.name,
   328→          awayTeamFlag: awayInfo.flag,
   329→          homeScore: m.homeScore,

   337→        }

   341→
   342→    // ── 5. Seed Elite & Crisis selections (group-stage only) ──
   343→    // R32 has NO Elite/Crisis XI — the stage is still live (6 matches Jul 3).
   344→    for (const stage of stages) {
   345→      const elitePool = ELITE_PLAYERS[stage.key]
   346→      const crisisPool = CRISIS_PLAYERS[stage.key]
   347→
   348→      if (elitePool && elitePool.length > 0) {
   349→        const eliteSelection = await db.wCSelection.create({
   350→          data: {
   351→            type: 'elite',
   352→            stageId: stage.id,
   353→            formation: '4-3-3',
   354→            locked: stage.key === 'group-stage', // group stage completed → locked
   355→          }
   356→        })
   357→
   358→        for (const p of elitePool) {
   359→          await db.wCSelectionPlayer.create({
   360→            data: {
   361→              selectionId: eliteSelection.id,
   362→              playerName: p.name,
   363→              nationCode: p.nationCode,
   364→              position: p.position,
   365→              pulseScore: p.pulseScore,
   366→              sentiment: p.sentiment,
   367→              trend: p.trend,
   368→              isLive: p.isLive,
   369→              matchInfo: p.matchInfo,
   370→              order: p.order,
   371→            }
   372→          })
   373→        }
   374→      }
   375→
   376→      if (crisisPool && crisisPool.length > 0) {
   377→        const crisisSelection = await db.wCSelection.create({
   378→          data: {
   379→            type: 'crisis',
   380→            stageId: stage.id,
   381→            formation: '4-3-3',
   382→            locked: stage.key === 'group-stage',
   383→          }
   384→        })
   385→
   386→        for (const p of crisisPool) {
   387→          await db.wCSelectionPlayer.create({
   388→            data: {
   389→              selectionId: crisisSelection.id,
   390→              playerName: p.name,
   391→              nationCode: p.nationCode,
   392→              position: p.position,
   393→              pulseScore: p.pulseScore,
   394→              sentiment: p.sentiment,
   395→              trend: p.trend,
   396→              isLive: p.isLive,
   397→              matchInfo: p.matchInfo,
   398→              order: p.order,
   399→            }
   400→          })
   401→        }
   402→      }
   403→    }
   404→
   405→    // ── 6. Run the Pulse Score engine (group-stage players only at this point) ──
   406→    // R32 is seeded AFTER the pulse engine so the engine's 4-component computed
   407→    // scores don't clobber the R32 web-buzz scores. R32 pulseScore = buzzScore.
   408→    let pulseResult: { playersComputed: number; breakdownsWritten: number; errors: string[] } | null = null
   409→    try {
   410→      pulseResult = await computeAllPulseScores(getDb())
   411→    } catch (err) {
   412→      console.error('Pulse engine failed during seed:', err)
   413→    }
   414→
   415→    // ── 7. Seed R32 Elite XI + Crisis XI from the VERIFIED_POOL baseline ──────
   416→    // R32 teams are seeded from the VERIFIED_POOL baseline (real web-buzz captured
   417→    // 2026-07-02). A background cron refreshes with live web_search periodically —
   418→    // see /api/world-cup/r32-cron. The pool contains only web-verified players;
   419→    // see src/lib/r32-buzz-ranker.ts for the anti-hallucination contract.
   420→    // Seeded AFTER the pulse engine so R32 buzz scores are preserved.
   421→    let r32Seeded: { elite: number; crisis: number; buzzSource: string } | null = null
   422→    try {
   423→      const r32Stage = stages.find((s) => s.key === 'round-of-32')
   424→      if (r32Stage) {
   425→        const r32Result = await rankR32Teams(false) // baseline, no SDK calls — instant
   426→        await seedR32Teams(db, r32Result, r32Stage.id)
   427→        r32Seeded = {
   428→          elite: r32Result.elite.length,
   429→          crisis: r32Result.crisis.length,
   430→          buzzSource: r32Result.buzzSource,
   431→        }
   432→      }
   433→    } catch (err) {
   434→      console.error('R32 seed failed (continuing with group stage only):', err)
   435→    }
   436→
   437→    return NextResponse.json({
   438→      success: true,
   439→      message: 'Database seeded with VERIFIED World Cup 2026 data (see /VERIFIED_DATA.md for sources)',
   440→      stages: stages.length,
   441→      nationalTeams: NATIONAL_TEAMS.length,
   442→      matches: matchesCreated,
   443→      matchdayInfo: 'Group Stage completed Jun 27 · Round of 32 LIVE (10/16 played, 6 upcoming Jul 3) · 24 group-stage matches + 16 R32 matches seeded',
   444→      pulse: pulseResult,
   445→      r32: r32Seeded,
   446→    })
   447→  } catch (error) {
   448→    return NextResponse.json(safeErrorResponse(error, 'world-cup/seed'), { status: 500 })
   449→  }
