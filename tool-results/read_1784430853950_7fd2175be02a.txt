   223→// ── Seed handler ─────────────────────────────────────────────────────────────
   224→// AUTH REQUIRED: this route wipes and re-creates most tables. Only the admin
   225→// may call it. The frontend no longer calls this route; it runs once at deploy.
   226→export async function POST(request: Request) {
   227→  // ── Auth gate ──
   228→  if (!isAdminAuthorized(request)) {
   229→    return unauthorizedResponse()
   230→  }
   231→
   232→  try {
   233→    // ── Check if already seeded (skip if ?force=true) ──
   234→    const { searchParams } = new URL(request.url)
   235→    const force = searchParams.get('force') === 'true'
   236→    if (!force) {
   237→      const existingStages = await db.wCStage.count()
   238→      if (existingStages > 0) {
   239→        return NextResponse.json({ message: 'Already seeded', stages: existingStages })
   240→      }
   241→    }
   242→
   243→    // ── 1. Clean up existing data ──
   244→    // NOTE: fanVote is intentionally NOT wiped — user votes are the product and
   245→    // must survive re-seeds. pulseBreakdown + sentimentSummary are safe to wipe
   246→    // because they are fully recomputed by the pulse engine below.

   258→    // Intentionally NOT deleting fanVote — user votes are preserved.

   307→    let matchesCreated = 0
