    98→function timeAgo(dateStr: string | null, isAr: boolean): string {
    99→  if (!dateStr) return ''
   100→  const now = new Date()
   101→  const date = new Date(dateStr)
   102→  const diffMs = now.getTime() - date.getTime()
   103→  const diffMin = Math.floor(diffMs / 60000)
   104→  const diffHr = Math.floor(diffMs / 3600000)
   105→  const diffDay = Math.floor(diffMs / 86400000)
   106→
   107→  if (isAr) {
   108→    if (diffMin < 1) return 'الآن'
   109→    if (diffMin < 60) return `منذ ${diffMin} د`
   110→    if (diffHr < 24) return `منذ ${diffHr} س`
   111→    return `منذ ${diffDay} ي`
   112→  }
   113→  if (diffMin < 1) return 'now'
   114→  if (diffMin < 60) return `${diffMin}m ago`
   115→  if (diffHr < 24) return `${diffHr}h ago`
   116→  return `${diffDay}d ago`
   117→}

   119→function formatNumber(n: number): string {
   120→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`
   121→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`
   122→  return n.toString()
   123→}

   153→// ── Circular Gauge Component ─────────────────────────────────

   155→function CircularGauge({ value, size = 100 }: { value: number; size?: number }) {
   156→  const strokeWidth = 8
   157→  const radius = (size - strokeWidth) / 2
   158→  const circumference = 2 * Math.PI * radius
   159→  const progress = (value / 100) * circumference
   160→  const color = getSentimentColor(value)
   161→
   162→  return (
   163→    <div className="relative" style={{ width: size, height: size }}>
   164→      <svg width={size} height={size} className="-rotate-90">
   165→        <circle
   166→          cx={size / 2}
   167→          cy={size / 2}
   168→          r={radius}
   169→          fill="none"
   170→          stroke="currentColor"
   171→          strokeWidth={strokeWidth}
   172→          className="text-[#E0E0E0] dark:text-white/10"
   173→        />
   174→        <motion.circle
   175→          cx={size / 2}
   176→          cy={size / 2}
   177→          r={radius}
   178→          fill="none"
   179→          stroke={color}
   180→          strokeWidth={strokeWidth}
   181→          strokeLinecap="round"
   182→          strokeDasharray={circumference}
   183→          initial={{ strokeDashoffset: circumference }}
   184→          animate={{ strokeDashoffset: circumference - progress }}
   185→          transition={{ duration: 1.2, ease: 'easeOut' }}
   186→        />
   187→      </svg>
   188→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   189→        <motion.span
   190→          className="text-2xl font-black"
   191→          style={{ color }}
   192→          initial={{ opacity: 0, scale: 0.5 }}
   193→          animate={{ opacity: 1, scale: 1 }}
   194→          transition={{ delay: 0.3, duration: 0.4 }}
   195→        >
   196→          {Math.round(value)}
   197→        </motion.span>
   198→        <span className="text-[9px] font-medium text-[#666] dark:text-[#CCCCCC]">/ 100</span>
   199→      </div>
   200→    </div>
   201→  )
   202→}

   260→

   263→    try {
   264→      setError(null)
   265→      const params = new URLSearchParams()
   266→      params.set('platform', platformFilter)
   267→      params.set('period', '24h')
   268→      if (languageFilter !== 'all') params.set('lang', languageFilter)
   269→
   270→      const res = await fetch(`/api/social-sentiment?${params.toString()}`)
   271→      if (!res.ok) throw new Error('Failed to fetch data')
   272→      const data = await res.json()
   273→
   274→      setPosts(data.posts || [])
   275→      setSummaries(data.summaries || [])
   276→    } catch (err) {
   277→      console.error('Failed to fetch sentiment data:', err)
   278→      setError(isAr ? 'فشل تحميل البيانات' : 'Failed to load data')
   279→    } finally {
   280→      setIsLoading(false)
   281→      setIsRefreshing(false)
   282→    }
   283→  }, [platformFilter, languageFilter, isAr])
   284→
   285→  useEffect(() => {
   286→    fetchData()

   288→
   289→  // ── Refresh Handler ──
   290→  const handleRefresh = useCallback(async () => {
   291→    setIsRefreshing(true)
   292→    try {
   293→      await fetch('/api/social-sentiment', {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify({
   297→          teamCode: '',
   298→          language: languageFilter === 'all' ? '' : languageFilter,
   299→        }),
   300→      })
   301→      await fetchData()
   302→    } catch (err) {
   303→      console.error('Refresh failed:', err)
   304→      setIsRefreshing(false)
   305→    }
   306→  }, [languageFilter, fetchData])
   307→
   308→  // ── Computed Data ──
   309→  const filteredPosts = useMemo(() => {
   310→    let result = posts
   311→    if (platformFilter !== 'all') {
   312→      result = result.filter(p => p.platform === platformFilter)
   313→    }
   314→    if (languageFilter !== 'all') {
   315→      result = result.filter(p => p.language === languageFilter)
   316→    }
   317→    // Sort by recency
   318→    result = [...result].sort((a, b) => {
   319→      const dateA = a.postedAt ? new Date(a.postedAt).getTime() : 0
   320→      const dateB = b.postedAt ? new Date(b.postedAt).getTime() : 0
   321→      return dateB - dateA
   322→    })
   323→    return result
   324→  }, [posts, platformFilter, languageFilter])
   325→
   326→  const globalAvgSentiment = useMemo(() => {
   327→    if (summaries.length === 0) return 0
   328→    const total = summaries.reduce((sum, s) => sum + s.avgSentiment * s.postCount, 0)
   329→    const count = summaries.reduce((sum, s) => sum + s.postCount, 0)
   330→    return count > 0 ? total / count : 0
   331→  }, [summaries])
   332→
   333→  const topTopics = useMemo(() => {
   334→    const topicMap = new Map<string, number>()
   335→    summaries.forEach(s => {
   336→      try {
   337→        const topics: string[] = JSON.parse(s.topTopics || '[]')
   338→        topics.forEach(t => topicMap.set(t, (topicMap.get(t) || 0) + s.postCount))
   339→      } catch { /* ignore */ }
   340→    })
   341→    return [...topicMap.entries()]
   342→      .sort((a, b) => b[1] - a[1])
   343→      .slice(0, 5)
   344→      .map(([topic]) => topic)
   345→  }, [summaries])
   346→
   347→  const mostActiveLanguage = useMemo(() => {
   348→    if (summaries.length === 0) return null
   349→    const langCounts = new Map<string, number>()
   350→    summaries.forEach(s => {
   351→      langCounts.set(s.language, (langCounts.get(s.language) || 0) + s.postCount)
   352→    })
   353→    let maxLang = ''
   354→    let maxCount = 0
   355→    langCounts.forEach((count, lang) => {
   356→      if (count > maxCount) {
   357→        maxCount = count
   358→        maxLang = lang
   359→      }
   360→    })
   361→    return { code: maxLang, count: maxCount }
   362→  }, [summaries])
   363→
   364→  const sentimentSplit = useMemo(() => {
   365→    if (filteredPosts.length === 0) return { positive: 33, neutral: 34, negative: 33 }
   366→    const positive = filteredPosts.filter(p => p.sentiment > 60).length
   367→    const neutral = filteredPosts.filter(p => p.sentiment >= 40 && p.sentiment <= 60).length
   368→    const negative = filteredPosts.filter(p => p.sentiment < 40).length
   369→    const total = filteredPosts.length
   370→    return {
   371→      positive: Math.round((positive / total) * 100),
   372→      neutral: Math.round((neutral / total) * 100),
   373→      negative: Math.round((negative / total) * 100),
   374→    }
   375→  }, [filteredPosts])
   376→
   377→  // Team sentiment data grouped by team
   378→  const teamSentiments = useMemo(() => {
   379→    const teamMap = new Map<string, {
   380→      teamCode: string
   381→      langSentiments: Map<string, { avg: number; count: number }>
   382→      totalPosts: number
   383→      overallSentiment: number
   384→    }>()
   385→
   386→    summaries.forEach(s => {
   387→      if (!teamMap.has(s.teamCode)) {
   388→        teamMap.set(s.teamCode, {
   389→          teamCode: s.teamCode,
   390→          langSentiments: new Map(),
   391→          totalPosts: 0,
   392→          overallSentiment: 0,
   393→        })

   399→        (team.overallSentiment * (team.totalPosts - s.postCount) + s.avgSentiment * s.postCount) /
   400→        team.totalPosts
   401→      )
   402→    })
   403→
   404→    return [...teamMap.values()]
