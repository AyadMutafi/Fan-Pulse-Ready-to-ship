    27→// ── World Cup stage label helper ─────────────────────────────
    28→// Renders an accurate label for a WC match based on its `group` value.
    29→// Group-stage matches use letters A-L; knockout rounds use R32/R16/QF/SF/Final.
    30→// (Previously everything was labelled "WC Group {x}", which produced the
    31→// nonsensical "WC Group R32" for knockout matches.)
    32→function wcStageLabel(group: string | null | undefined): string {
    33→  if (!group) return 'World Cup 2026'
    34→  if (/^[A-L]$/.test(group)) return `WC Group ${group}`
    35→  switch (group) {
    36→    case 'R32': return 'WC Round of 32'
    37→    case 'R16': return 'WC Round of 16'
    38→    case 'QF': return 'WC Quarter Finals'
    39→    case 'SF': return 'WC Semi Finals'
    40→    case 'Final': return 'WC Final'
    41→    default: return 'World Cup 2026'
    42→  }
    43→}

   235→function HomeTab() {
   236→  const { t } = useLanguage()
   237→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   238→  const [apiMatches, setApiMatches] = useState<Array<{
   239→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   240→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   241→    status: string; group: string; matchDate: string
   242→  }>>([])
   243→
   244→  // Fan vote state
   245→  const [sessionId, setSessionId] = useState<string>('')
   246→  const [fanVotes, setFanVotes] = useState<FanVoteAgg[]>([])
   247→  const [myVotes, setMyVotes] = useState<Array<{ teamCode: string; score: number }>>([])
   248→  const [votesLoading, setVotesLoading] = useState(true)
   249→  const [selectedVoteTeam, setSelectedVoteTeam] = useState<string | null>(null)
   250→  const [submitting, setSubmitting] = useState(false)
   251→  const [toast, setToast] = useState<{ msg: string; emoji: string } | null>(null)
   252→  const [fanCardOffer, setFanCardOffer] = useState<{ teamCode: string; score: number } | null>(null)
   253→
   254→  useEffect(() => {
   255→    async function fetchMatches() {
   256→      try {
   257→        const res = await fetch('/api/matches?league=WC')
   258→        if (res.ok) {
   259→          const data = await res.json()
   260→          const mapped = (data.matches || []).map((m: any) => ({
   261→            id: m.id,
   262→            home: m.homeTeam.code,
   263→            away: m.awayTeam.code,
   264→            homeFlag: m.homeTeam.flag || '',
   265→            awayFlag: m.awayTeam.flag || '',
   266→            score: m.score,
   267→            homeSentiment: Math.round(m.homeTeam.sentiment),
   268→            awaySentiment: Math.round(m.awayTeam.sentiment),
   269→            live: m.status === 'live',
   270→            league: m.league === 'WC' ? wcStageLabel(m.group) : m.league,
   271→            status: m.status || 'upcoming',
   272→            group: m.group || '',
   273→            matchDate: m.matchDate || '',
   274→          }))
   275→          setApiMatches(mapped)
   276→        }
   277→      } catch (err) {
   278→        console.error('Failed to fetch matches:', err)
   279→      }
   280→    }
   281→    fetchMatches()
   282→  }, [])
   283→
   284→  // SSR-safe: read/create sessionId inside useEffect, never during render.
   285→  useEffect(() => {
   286→    try {
   287→      const existing = typeof window !== 'undefined' ? window.localStorage.getItem('fan_session_id') : null
   288→      if (existing && existing.length > 0) {
   289→        setSessionId(existing)
   290→        return
   291→      }
   292→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
   293→        ? crypto.randomUUID()
   294→        : `anon-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`
   295→      if (typeof window !== 'undefined') {
   296→        window.localStorage.setItem('fan_session_id', newId)
   297→      }
   298→      setSessionId(newId)
   299→    } catch {
   300→      setSessionId(`anon-${Date.now()}`)
   301→    }
   302→  }, [])
   303→
   304→  // Fetch fan votes whenever sessionId changes.
   305→  useEffect(() => {
   306→    if (!sessionId) return
   307→    let cancelled = false
   308→    async function loadVotes() {
   309→      setVotesLoading(true)
   310→      try {
   311→        const res = await fetch(`/api/fan-vote?session=${encodeURIComponent(sessionId)}`)
   312→        if (!res.ok) return
   313→        const data = await res.json()
   314→        if (cancelled) return
   315→        setFanVotes(Array.isArray(data.votes) ? data.votes : [])
   316→        setMyVotes(Array.isArray(data.myVotes) ? data.myVotes : [])
   317→      } catch (err) {
   318→        console.error('Failed to fetch fan votes:', err)
   319→      } finally {
   320→        if (!cancelled) setVotesLoading(false)
   321→      }
   322→    }
   323→    loadVotes()
   324→    return () => { cancelled = true }
   325→  }, [sessionId])
   326→
   327→  // Close voting modal on Escape key (accessibility — modal was only closable
   328→  // via the explicit Close button or selecting a mood)
   329→  useEffect(() => {
   330→    if (!selectedVoteTeam) return
   331→    const onKey = (e: KeyboardEvent) => {
   332→      if (e.key === 'Escape' && !submitting) setSelectedVoteTeam(null)
   333→    }
   334→    window.addEventListener('keydown', onKey)
   335→    return () => window.removeEventListener('keydown', onKey)
   336→  }, [selectedVoteTeam, submitting])
   337→
   338→  const totalVoteCount = fanVotes.reduce((sum, v) => sum + (v.count || 0), 0)
   339→
   340→  const handleVote = async (teamCode: string, score: number) => {
   341→    if (!sessionId || submitting) return
   342→    setSubmitting(true)
   343→    // Optimistic update: bump my vote immediately
   344→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   345→    setMyVotes(prev => {
   346→      const without = prev.filter(v => v.teamCode !== teamCode)
   347→      return [...without, { teamCode, score }]
   348→    })
   349→    setFanVotes(prev => {
   350→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   351→      if (idx === -1) {
   352→        return [...prev, { teamCode, score, count: 1 }]
   353→      }
   354→      const next = [...prev]
   355→      const wasMine = prevMyVote !== null
   356→      // Adjust running average: if I had voted, swap my old score for the new.
   357→      const current = next[idx]
   358→      const totalScore = current.score * current.count
   359→      const newScore = wasMine
   360→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   361→        : Math.round((totalScore + score) / (current.count + 1))
   362→      const newCount = wasMine ? current.count : current.count + 1
   363→      next[idx] = { teamCode, score: newScore, count: newCount }
   364→      return next
   365→    })
   366→    setSelectedVoteTeam(null)
   367→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   368→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   369→    setTimeout(() => setToast(null), 2500)
   370→    // Show the Fan Card offer for 8 seconds — long enough to click "Get Fan Card"
   371→    setFanCardOffer({ teamCode, score })
   372→    const offerTimer = setTimeout(() => setFanCardOffer(null), 8000)
   373→    try {
   374→      await fetch('/api/fan-vote', {
   375→        method: 'POST',
   376→        headers: { 'Content-Type': 'application/json' },
   377→        body: JSON.stringify({ teamCode, score, sessionId }),
   378→      })
   379→    } catch (err) {
   380→      console.error('Failed to submit vote:', err)
   381→      // Revert on failure
   382→      setMyVotes(prev => {
   383→        const without = prev.filter(v => v.teamCode !== teamCode)
   384→        if (prevMyVote !== null) return [...without, { teamCode, score: prevMyVote }]
   385→        return without
   386→      })
   387→      setToast({ msg: 'Vote failed — please retry', emoji: '⚠️' })
   388→      setFanCardOffer(null)
   389→      clearTimeout(offerTimer)
   390→    } finally {
   391→      setSubmitting(false)
   392→    }
   393→  }
   394→
   395→  const filteredMatches = (matchFilter === 'ALL'
   396→    ? apiMatches
   397→    : apiMatches.filter(m => m.league.startsWith('WC'))
   398→  ).slice(0, 24) // Limit to 24 cards max for performance
   399→
   400→  const fanVoteIntelText = totalVoteCount === 0
   401→    ? 'Be the first to vote in the Fan Mood section below'
   402→    : `${totalVoteCount.toLocaleString()} fan votes tallied for World Cup 2026 Group Stage`
   403→
   404→  // ── Arena Intelligence: VERIFIED insights only ──────────────────────────
   405→  // Every insight below is a historical fact sourced from VERIFIED_DATA.md
   406→  // (the project's single source of truth, cross-checked against Wikipedia,
   407→  // ESPN, Olympics.com, FIFA.com on 2026-07-02). We deliberately do NOT
   408→  // auto-generate insights from apiMatches because that approach produced two
   409→  // verifiable hallucinations in production:
   410→  //   (1) "World Cup 2026 kicked off with ESP 0-0 AUT in the opener" — FALSE.
   411→  //       The old code used parsed[0] (first array element) as "the opener",
   412→  //       but the first row happened to be an UPCOMING R32 match (ESP vs AUT,
   413→  //       scheduled Jul 3, not yet played). The real opener was MEX 2-0 RSA
   414→  //       on Jun 11 (the earliest matchDate among completed matches).
   415→  //   (2) "Shock in Group Stage: ESP 0-0 AUT" — FALSE. The old code used
   416→  //       league.includes('WC') to label the stage as "Group Stage", but ALL
   417→  //       WC matches (including R32) match that check. ESP vs AUT is an R32
   418→  //       match, not a group-stage match. The real group-stage shock was
   419→  //       ESP 0-0 CPV (Spain held scoreless by Cape Verde).
   420→  //
   421→  // Fix: hardcode the verified insight set with explicit VERIFIED_DATA.md
   422→  // citations. Only the fan-vote count stays dynamic (it is live data).
   423→  // This guarantees we never describe an upcoming match as played, never
   424→  // mislabel a knockout match as a group-stage match, and never invent stats.

  1167→function RateTab() {
  1168→  const { t } = useLanguage()
  1169→  const [players, setPlayers] = useState<Array<{
  1170→    id: string
  1171→    playerName: string
  1172→    nationCode: string
  1173→    position: string
  1174→    avgRating: number
  1175→    totalRatings: number
  1176→  }>>([])
  1177→  const [myRatings, setMyRatings] = useState<Record<string, number>>({})
  1178→  const [loading, setLoading] = useState(true)
  1179→  const [submittingId, setSubmittingId] = useState<string | null>(null)
  1180→
  1181→  // ── Load real players from the DB + this session's ratings ──
  1182→  useEffect(() => {
  1183→    let cancelled = false
  1184→    async function load() {
  1185→      setLoading(true)
  1186→      try {
  1187→        const res = await fetch('/api/ratings')
  1188→        if (!res.ok) return
  1189→        const data = await res.json()
  1190→        if (cancelled) return
  1191→        const list = Array.isArray(data.ratings) ? data.ratings : []
  1192→        setPlayers(list)
  1193→      } catch {
  1194→        /* ignore — toast shown on submit failure */
  1195→      } finally {
  1196→        if (!cancelled) setLoading(false)
  1197→      }
  1198→    }
  1199→    load()
  1200→    return () => { cancelled = true }
  1201→  }, [])
  1202→
  1203→  // ── Per-browser session ID (crypto.randomUUID, persisted in localStorage) ──
  1204→  // Required by the API — the old 'anonymous' default is rejected. A real
  1205→  // session ID makes the (sessionId, playerId) unique constraint meaningful,
  1206→  // enforcing one rating per session per player.
  1207→  const [sessionId, setSessionId] = useState<string>('')
  1208→  useEffect(() => {
  1209→    try {
  1210→      const existing = window.localStorage.getItem('fp_session_id')
  1211→      if (existing && existing.length >= 8) {
  1212→        setSessionId(existing)
  1213→        return
  1214→      }
  1215→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
  1216→        ? crypto.randomUUID()
  1217→        : `fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`
  1218→      window.localStorage.setItem('fp_session_id', newId)
  1219→      setSessionId(newId)
  1220→    } catch {
  1221→      setSessionId(`fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`)
  1222→    }
  1223→  }, [])
  1224→
  1225→  const handleRate = async (playerId: string, rating: number) => {
  1226→    if (!sessionId || submittingId === playerId) return
  1227→    setSubmittingId(playerId)
  1228→    // Optimistically update local state so the stars fill instantly.
  1229→    setMyRatings(prev => ({ ...prev, [playerId]: rating }))
  1230→    try {
  1231→      const res = await fetch('/api/ratings', {
  1232→        method: 'POST',
  1233→        headers: { 'Content-Type': 'application/json' },
  1234→        body: JSON.stringify({ playerId, rating, sessionId }),
  1235→      })
  1236→      const data = await res.json().catch(() => null)
  1237→      if (!res.ok) {
  1238→        // Revert on failure.
  1239→        setMyRatings(prev => {
  1240→          const next = { ...prev }
  1241→          delete next[playerId]
  1242→          return next
  1243→        })
  1244→        if (res.status === 429) {
  1245→          toast.error('Too many ratings — please slow down.')
  1246→        } else {
  1247→          toast.error(data?.error || 'Failed to submit rating')
  1248→        }
  1249→        return
  1250→      }
  1251→      toast.success(data?.updated ? 'Rating updated!' : 'Rating submitted!', {
  1252→        description: `${rating}/10`,

  1255→      const fresh = await fetch('/api/ratings')
  1256→      if (fresh.ok) {
  1257→        const fd = await fresh.json().catch(() => null)
  1258→        if (Array.isArray(fd?.ratings)) setPlayers(fd.ratings)
  1259→      }
  1260→    } catch {
  1261→      setMyRatings(prev => {
  1262→        const next = { ...prev }
  1263→        delete next[playerId]
  1264→        return next
  1265→      })
  1266→      toast.error('Network error — please try again')

  1268→      setSubmittingId(null)
  1269→    }
  1270→  }
  1271→
  1272→  return (
  1273→    <div className="space-y-6">
  1274→      <motion.div
  1275→        initial={{ opacity: 0, y: 20 }}
  1276→        animate={{ opacity: 1, y: 0 }}

  1279→        <h2 className="text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
  1280→          {t('ratings.title')}
  1281→        </h2>
  1282→        <p className="text-sm text-[#666] dark:text-[#CCCCCC]">{t('ratings.desc')}</p>
  1283→      </motion.div>

  1286→        {loading ? (
  1287→          <Card className="border-[#E0E0E0]/50 dark:border-white/5">

  1289→              Loading players…
  1290→            </CardContent>
  1291→          </Card>
  1292→        ) : players.length === 0 ? (
  1293→          <Card className="border-[#E0E0E0]/50 dark:border-white/5">
  1294→            <CardContent className="p-6 text-center text-sm text-[#666] dark:text-[#CCCCCC]">
  1295→              No rateable players found.
  1296→            </CardContent>
  1297→          </Card>
  1298→        ) : (
  1299→          players.map((player, i) => (
  1300→            <motion.div
  1301→              key={player.id}
  1302→              initial={{ opacity: 0, y: 10 }}
