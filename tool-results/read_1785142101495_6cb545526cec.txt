    33→// ── Color thresholds ────────────────────────────────────────

    37→// ── Chart geometry (viewBox 0 0 360 160) ─────────────────────
    38→const CHART_W = 360
    39→const CHART_H = 160
    40→const PAD_L = 26
    41→const PAD_R = 8
    42→const PAD_T = 10
    43→const PAD_B = 22
    44→const X0 = PAD_L
    45→const X1 = CHART_W - PAD_R
    46→const Y0 = PAD_T
    47→const Y1 = CHART_H - PAD_B
    48→const W_INNER = X1 - X0
    49→const H_INNER = Y1 - Y0
    50→
    51→const xForMin = (m: number) => X0 + (m / 90) * W_INNER
    52→const yForVal = (v: number) => Y1 - (Math.max(0, Math.min(100, v)) / 100) * H_INNER
    53→
    54→// ── Smooth path (Catmull-Rom → Bezier) ──────────────────────
    55→function buildSmoothPath(pts: { x: number; y: number }[]): string {
    56→  if (pts.length === 0) return ''
    57→  if (pts.length === 1) return `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`
    58→  let d = `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`
    59→  for (let i = 0; i < pts.length - 1; i++) {
    60→    const p0 = pts[i - 1] ?? pts[i]
    61→    const p1 = pts[i]
    62→    const p2 = pts[i + 1]
    63→    const p3 = pts[i + 2] ?? p2
    64→    const c1x = p1.x + (p2.x - p0.x) / 6
    65→    const c1y = p1.y + (p2.y - p0.y) / 6
    66→    const c2x = p2.x - (p3.x - p1.x) / 6
    67→    const c2y = p2.y - (p3.y - p1.y) / 6
    68→    d += ` C ${c1x.toFixed(2)} ${c1y.toFixed(2)}, ${c2x.toFixed(2)} ${c2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`
    69→  }
    70→  return d
    71→}
    72→
    73→function buildAreaPath(pts: { x: number; y: number }[]): string {
    74→  if (pts.length === 0) return ''
    75→  const line = buildSmoothPath(pts)
    76→  const last = pts[pts.length - 1]
    77→  const first = pts[0]
    78→  return `${line} L ${last.x.toFixed(2)} ${Y1.toFixed(2)} L ${first.x.toFixed(2)} ${Y1.toFixed(2)} Z`
    79→}
    80→
    81→// Interpolated value at a target minute (linear between samples)
    82→function valueAtMinute(minutes: number[], values: number[], target: number): number {
    83→  if (values.length === 0) return 0
    84→  if (target <= minutes[0]) return values[0]
    85→  if (target >= minutes[minutes.length - 1]) return values[values.length - 1]
    86→  for (let i = 0; i < minutes.length - 1; i++) {
    87→    if (target >= minutes[i] && target <= minutes[i + 1]) {
    88→      const span = minutes[i + 1] - minutes[i] || 1
    89→      const t = (target - minutes[i]) / span
    90→      return values[i] + (values[i + 1] - values[i]) * t
    91→    }
    92→  }
    93→  return values[values.length - 1]
    94→}

   105→function TrendArrow({ trend, className = '' }: { trend: 'up' | 'down' | 'flat'; className?: string }) {
   106→  if (trend === 'up') return <TrendingUp className={`size-3 ${className}`} />
   107→  if (trend === 'down') return <TrendingDown className={`size-3 ${className}`} />
   108→  return <Minus className={`size-3 ${className}`} />
   109→}
