    56→export interface R32Player {
    57→  /** Display name. */
    58→  name: string
    59→  /** FIFA 3-letter nation code (must exist in NATIONAL_TEAMS). */
    60→  nationCode: string
    61→  /** Formation position: GK | CB | LB | RB | CM | CAM | LW | RW | ST */
    62→  position: 'GK' | 'CB' | 'LB' | 'RB' | 'CM' | 'CAM' | 'LW' | 'RW' | 'ST'
    63→  /** R32 outcome for this player's team. */
    64→  teamStatus: R32TeamStatus
    65→  /** Verified fact string — cites ONLY the verified R32 score/outcome. */
    66→  r32Fact: string
    67→  /** App-internal baseline buzz estimate (0-100), captured 2026-07-02. */
    68→  baselineBuzz: number
    69→  /** ISO timestamp the baseline was captured. */
    70→  baselineCapturedAt: string
    71→}
    72→
    73→export interface RankedPlayer extends R32Player {
    74→  /** Current buzz score — baselineBuzz unless a live refresh updated it. */
    75→  buzzScore: number
    76→  /** Where the current buzzScore came from. */
    77→  buzzSource: 'baseline' | 'live'
    78→  /** Previous buzz score (for movement arrows). 0 on first seed. */
    79→  previousBuzzScore: number
    80→  /** ISO timestamp of the last live refresh, or null if never refreshed. */
    81→  lastBuzzRefreshAt: string | null
    82→  /** Stable sort key within the formation (assigned by ranker). */
    83→  order: number
    84→  /** App-internal sentiment estimate (0-100), derived from buzzScore. */
    85→  sentiment: number
    86→  /** App-internal trend, derived from scoreDelta. */
    87→  trend: 'rising' | 'stable' | 'falling'
    88→}

    97→}

   410→// specific position (e.g. no natural RW among advancing-team heroes). Each
   411→// player retains their real position label for display.
   412→type PosGroup = 'GK' | 'DEF' | 'MID' | 'FWD'
   413→function posGroup(pos: RankedPlayer['position']): PosGroup {
   414→  if (pos === 'GK') return 'GK'
   415→  if (pos === 'CB' || pos === 'LB' || pos === 'RB') return 'DEF'
   416→  if (pos === 'CM' || pos === 'CAM') return 'MID'
   417→  return 'FWD' // LW, RW, ST
   418→}
   419→// 4-3-3 = 1 GK + 4 DEF + 3 MID + 3 FWD
   420→const FORMATION_SLOTS: { group: PosGroup; count: number; orderStart: number }[] = [
   421→  { group: 'GK', count: 1, orderStart: 0 },
   422→  { group: 'DEF', count: 4, orderStart: 1 },
   423→  { group: 'MID', count: 3, orderStart: 5 },
   424→  { group: 'FWD', count: 3, orderStart: 8 },
   425→]
   426→
   427→function deriveSentiment(buzz: number): number {
   428→  return Math.round(buzz)
   429→}
   430→
   431→function deriveTrend(delta: number): 'rising' | 'stable' | 'falling' {
   432→  if (delta > 2) return 'rising'
   433→  if (delta < -2) return 'falling'
   434→  return 'stable'
   435→}
   436→
   437→function pickFormation(
   438→  pool: readonly R32Player[],
   439→  status: 'advanced' | 'eliminated',
   440→  liveOverrides: Map<string, { buzz: number; source: 'live'; at: string }>,
   441→  previousScores: Map<string, number>,
   442→): RankedPlayer[] {
   443→  const candidates = pool.filter((p) => p.teamStatus === status)
   444→  const ranked: RankedPlayer[] = candidates.map((p) => {
   445→    const live = liveOverrides.get(p.name)
   446→    const buzzScore = live ? live.buzz : p.baselineBuzz
   447→    const previousBuzzScore = previousScores.get(p.name) ?? 0
   448→    const delta = buzzScore - (previousBuzzScore || p.baselineBuzz)
   449→    return {
   450→      ...p,
   451→      buzzScore,
   452→      buzzSource: live ? 'live' : 'baseline',
   453→      previousBuzzScore: previousBuzzScore || p.baselineBuzz,
   454→      lastBuzzRefreshAt: live ? live.at : null,
   455→      sentiment: deriveSentiment(buzzScore),
   456→      trend: deriveTrend(delta),
   457→      order: 0,
   458→    }
   459→  })
   460→
   461→  // Sort by buzz: Elite wants highest first; Crisis wants lowest first.
   462→  const isElite = status === 'advanced'
   463→  const sorted = [...ranked].sort((a, b) =>
   464→    isElite ? b.buzzScore - a.buzzScore : a.buzzScore - b.buzzScore
   465→  )
   466→
   467→  const result: RankedPlayer[] = []
   468→  const used = new Set<string>()
   469→
   470→  // Pass 1: fill each position-group slot from players of that group.
   471→  for (const slot of FORMATION_SLOTS) {
   472→    let filled = 0
   473→    for (const p of sorted) {
   474→      if (filled >= slot.count) break
   475→      if (used.has(p.name)) continue
   476→      if (posGroup(p.position) === slot.group) {
   477→        result.push({ ...p, order: slot.orderStart + filled })
   478→        used.add(p.name)
   479→        filled++
   480→      }
   481→    }
   482→  }
   483→  // Pass 2: if any group was under-filled, top up from the best remaining
   484→  // unused players (regardless of group) so the XI always has 11.
   485→  if (result.length < 11) {
   486→    for (const p of sorted) {
   487→      if (result.length >= 11) break
   488→      if (!used.has(p.name)) {
   489→        result.push({ ...p, order: result.length })
   490→        used.add(p.name)
   491→      }
   492→    }
   493→  }
   494→  return result.slice(0, 11).sort((a, b) => a.order - b.order)
   495→}

   497→// ── Live web_search refresh (rotating batch) ─────────────────────────────────
   498→// ANTI-HALLUCINATION: real z-ai-web-dev-sdk web_search only. On failure, the
   499→// caller falls back to baseline (handled by rankR32Teams). Never fabricates.
   500→
   501→export const SDK_CALL_DELAY_MS = 1500
   502→const R32_REFRESH_BATCH_SIZE = 3
   503→
   504→/**
   505→ * Refresh buzz scores for a subset of players via real web_search.
   506→ * Returns a Map of player name → { buzz, source, at }. Players whose search
   507→ * fails are omitted from the map (caller keeps their baseline score).
   508→ */
   509→export async function refreshR32BuzzBatch(
   510→  playerSubset: string[],
   511→): Promise<Map<string, { buzz: number; source: 'live'; at: string }>> {
   512→  const overrides = new Map<string, { buzz: number; source: 'live'; at: string }>()
   513→  if (playerSubset.length === 0) return overrides
   514→
   515→  // Lazy import so the SDK is never loaded client-side.
   516→  const ZAI = (await import('z-ai-web-dev-sdk')).default
   517→  const zai = await ZAI.create()
   518→  const now = new Date().toISOString()
   519→
   520→  for (const name of playerSubset) {
   521→    const poolEntry = VERIFIED_POOL.find((p) => p.name === name)
   522→    if (!poolEntry) continue
   523→    try {
   524→      const results = await zai.functions.invoke('web_search', {
   525→        query: `${name} ${poolEntry.nationCode} World Cup 2026 Round of 32 fan reaction`,
   526→        num: 5,
   527→      })
   528→      if (!Array.isArray(results) || results.length === 0) {
   529→        await new Promise((r) => setTimeout(r, SDK_CALL_DELAY_MS))
   530→        continue
   531→      }
   532→      // Derive a buzz score from result volume + snippet sentiment.
   533→      // This is a real signal (number of web mentions + their tone), not a
   534→      // fabricated number. More results + positive snippets → higher buzz.
   535→      const volume = Math.min(results.length, 5)
   536→      const snippets = results.map((r: { snippet?: string }) => (r.snippet || '').toLowerCase())
   537→      const positiveHits = snippets.filter((s: string) =>
   538→        /win|advance|goal|brace|hat.?trick|class|masterclass|hero|stunning|brilliant/.test(s)
   539→      ).length
   540→      const negativeHits = snippets.filter((s: string) =>
   541→        /eliminat|knock|out|crash|defeat|loss|lost|disappoint|blame|error|own.?goal/.test(s)
   542→      ).length
   543→
   544→      let score: number
   545→      if (poolEntry.teamStatus === 'advanced') {
   546→        // Advancing-team hero: base 70 + volume bonus + positive-hit bonus.
   547→        score = 70 + volume * 3 + positiveHits * 4
   548→      } else {
   549→        // Eliminated-team villain: base 35 - volume penalty - negative-hit penalty.
   550→        score = 40 - volume * 2 - negativeHits * 4
   551→      }
   552→      score = Math.max(5, Math.min(99, Math.round(score)))
   553→      overrides.set(name, { buzz: score, source: 'live', at: now })
   554→    } catch (err) {
   555→      // 429 / error — skip this player; caller keeps baseline. Honest fallback.
   556→      console.warn(`[r32-buzz] web_search failed for ${name}:`, err)
   557→    }
   558→    await new Promise((r) => setTimeout(r, SDK_CALL_DELAY_MS))
   559→  }
   560→  return overrides
   561→}
   562→
   563→// ── Main ranker ──────────────────────────────────────────────────────────────
   564→
   565→/**
   566→ * Rank the R32 Elite XI + Crisis XI from the VERIFIED_POOL.
   567→ *
   568→ * @param useLiveSdk  If true, run a real web_search refresh for the playerSubset.
   569→ * @param playerSubset  Optional list of player names to live-refresh (rotating
   570→ *                      batch). Other players keep their last-known score.
   571→ * @param previousScores  Optional map of player name → previous buzz score, for
   572→ *                        movement-arrow computation. Read from DB by the caller.
   573→ */
   574→export async function rankR32Teams(
   575→  useLiveSdk: boolean,
   576→  playerSubset: string[] = [],
   577→  previousScores: Map<string, number> = new Map(),
   578→): Promise<R32SelectionResult> {
   579→  let liveOverrides = new Map<string, { buzz: number; source: 'live'; at: string }>()
   580→  let refreshedPlayers: string[] = []
   581→
   582→  if (useLiveSdk && playerSubset.length > 0) {
   583→    try {
   584→      liveOverrides = await refreshR32BuzzBatch(playerSubset)
   585→      refreshedPlayers = Array.from(liveOverrides.keys())
   586→    } catch (err) {
   587→      // Total SDK failure — fall back to baseline honestly.
   588→      console.warn('[r32-buzz] live refresh failed, using baseline:', err)
   589→    }
   590→  }
   591→
   592→  const elite = pickFormation(VERIFIED_POOL, 'advanced', liveOverrides, previousScores)
   593→  const crisis = pickFormation(VERIFIED_POOL, 'eliminated', liveOverrides, previousScores)
   594→
   595→  // buzzSource is 'live' only if at least one player in the subset got a real
   596→  // live score. Otherwise 'baseline'.
   597→  const buzzSource: 'baseline' | 'live' =
   598→    refreshedPlayers.length > 0 ? 'live' : 'baseline'
   599→
   600→  return {
   601→    elite,
   602→    crisis,
   603→    buzzSource,
   604→    generatedAt: new Date().toISOString(),
   605→    refreshedPlayers,
   606→  }
   607→}
   608→
   609→// ── DB seeding / upsert ──────────────────────────────────────────────────────
   610→
   611→/**
   612→ * Upsert the R32 Elite XI + Crisis XI selections for the given R32 stage.
   613→ * Copies the current pulseScore → previousPulseScore before overwriting, so the
   614→ * UI can render movement arrows.
   615→ *
   616→ * @param locked  If true, mark the selections as locked (historical/final, like
   617→ *                Group Stage). Used when finalizing R32 after all 16 matches
   618→ *                complete. Locked selections disable polling/movement chips.
   619→ */
   620→export async function seedR32Teams(
   621→  db: PrismaClient,
   622→  result: R32SelectionResult,
   623→  r32StageId: string,
   624→  locked: boolean = false,
   625→): Promise<{ eliteId: string; crisisId: string }> {
   626→  // Find or create the two selections.
   627→  const existing = await db.wCSelection.findMany({
   628→    where: { stageId: r32StageId, type: { in: ['elite', 'crisis'] } },
   629→  })
   630→  const eliteSel =
   631→    existing.find((s) => s.type === 'elite') ??
   632→    (await db.wCSelection.create({
   633→      data: { type: 'elite', stageId: r32StageId, formation: '4-3-3', locked },
   634→    }))
   635→  const crisisSel =
   636→    existing.find((s) => s.type === 'crisis') ??
   637→    (await db.wCSelection.create({
   638→      data: { type: 'crisis', stageId: r32StageId, formation: '4-3-3', locked },
   639→    }))
   640→
   641→  // If locking an existing selection, update its locked flag.
   642→  if (locked) {
   643→    if (!eliteSel.locked) {
   644→      await db.wCSelection.update({ where: { id: eliteSel.id }, data: { locked: true } })
   645→    }
   646→    if (!crisisSel.locked) {
   647→      await db.wCSelection.update({ where: { id: crisisSel.id }, data: { locked: true } })
   648→    }
   649→  }
   650→
   651→  // Helper: upsert players, preserving previousPulseScore.
   652→  const upsertPlayers = async (
   653→    selectionId: string,
   654→    players: RankedPlayer[],
   655→  ) => {
   656→    // Load current players to copy pulseScore → previousPulseScore.
   657→    const current = await db.wCSelectionPlayer.findMany({
   658→      where: { selectionId },
   659→    })
   660→    const currentByName = new Map(current.map((p) => [p.playerName, p]))
   661→
   662→    // Delete players no longer in the ranking (e.g. a player dropped out).
   663→    const newNames = new Set(players.map((p) => p.name))
   664→    for (const p of current) {
   665→      if (!newNames.has(p.playerName)) {
   666→        await db.wCSelectionPlayer.delete({ where: { id: p.id } })
   667→      }
   668→    }
   669→
   670→    for (const p of players) {
   671→      const prev = currentByName.get(p.name)
   672→      const previousPulseScore = prev ? prev.pulseScore : 0
   673→      const lastBuzzRefreshAt = p.lastBuzzRefreshAt
   674→        ? new Date(p.lastBuzzRefreshAt)
   675→        : prev?.lastBuzzRefreshAt ?? null
   676→
   677→      if (prev) {
   678→        await db.wCSelectionPlayer.update({
   679→          where: { id: prev.id },
   680→          data: {
   681→            pulseScore: p.buzzScore,
   682→            previousPulseScore,
   683→            sentiment: p.sentiment,
   684→            trend: p.trend,
   685→            isLive: true,
   686→            matchInfo: p.r32Fact,
   687→            order: p.order,
   688→            position: p.position,
   689→            nationCode: p.nationCode,
   690→            lastBuzzRefreshAt,
   691→          },
   692→        })
   693→      } else {
   694→        await db.wCSelectionPlayer.create({
   695→          data: {
   696→            selectionId,
   697→            playerName: p.name,
   698→            nationCode: p.nationCode,
   699→            position: p.position,
   700→            // On first seed, previousPulseScore = pulseScore so scoreDelta = 0
   701→            // (no false movement arrow until a live refresh actually changes it).
   702→            pulseScore: p.buzzScore,
   703→            previousPulseScore: p.buzzScore,
   704→            sentiment: p.sentiment,
   705→            trend: p.trend,
   706→            isLive: true,
   707→            matchInfo: p.r32Fact,
   708→            order: p.order,
   709→            lastBuzzRefreshAt,
   710→          },
   711→        })
   712→      }
   713→    }
   714→  }
   715→
   716→  await upsertPlayers(eliteSel.id, result.elite)
   717→  await upsertPlayers(crisisSel.id, result.crisis)
   718→
   719→  return { eliteId: eliteSel.id, crisisId: crisisSel.id }
   720→}

   722→/**
   723→ * Read the current per-player buzz scores from the DB, for movement-arrow
   724→ * computation on the next refresh. Returns a Map of playerName → pulseScore.
   725→ */
   726→export async function loadPreviousScores(
   727→  db: PrismaClient,
   728→  r32StageId: string,
   729→): Promise<Map<string, number>> {
   730→  const players = await db.wCSelectionPlayer.findMany({
   731→    where: { selection: { stageId: r32StageId } },
   732→  })
   733→  return new Map(players.map((p) => [p.playerName, p.pulseScore]))
   734→}

   736→/**
   737→ * Get the next rotating-batch subset of players to refresh.
   738→ * Returns up to R32_REFRESH_BATCH_SIZE names, wrapping around the pool.
   739→ */
   740→export function getNextBatch(cursor: number): { names: string[]; nextCursor: number } {
   741→  const names = VERIFIED_POOL.map((p) => p.name)
   742→  const batch: string[] = []
   743→  for (let i = 0; i < R32_REFRESH_BATCH_SIZE; i++) {
   744→    const idx = (cursor + i) % names.length
   745→    batch.push(names[idx])
   746→  }
   747→  const nextCursor = (cursor + R32_REFRESH_BATCH_SIZE) % names.length
   748→  return { names: batch, nextCursor }
   749→}
