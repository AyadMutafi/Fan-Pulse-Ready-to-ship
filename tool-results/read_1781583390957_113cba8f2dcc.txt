   152→function getTrendIcon(trend: string) {
   153→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   154→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   155→  return <Minus className="size-3 text-[#FF6B35]" />
   156→}
   157→
   158→function getSentimentColor(score: number) {
   159→  if (score >= 80) return 'text-[#10B981]'
   160→  if (score >= 50) return 'text-[#FF6B35]'
   161→  return 'text-[#EF4444]'
   162→}
   163→
   164→function getSentimentBg(score: number) {
   165→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   166→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   167→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   168→}
   169→
   170→function getProgressClass(score: number) {
   171→  if (score >= 80) return 'progress-emerald'
   172→  if (score >= 50) return 'progress-amber'
   173→  return 'progress-red'
   174→}
   175→
   176→// Face emoji mapping for /10 rating system
   177→function getPulseFaceEmoji(pulseScore: number): string {
   178→  if (pulseScore >= 90) return '🤩'
   179→  if (pulseScore >= 70) return '😊'
   180→  if (pulseScore >= 50) return '😐'
   181→  if (pulseScore >= 30) return '😟'
   182→  return '😵'
   183→}
   184→
   185→function getRatingColor(rating: number): string {
   186→  if (rating >= 9) return '#10B981'
   187→  if (rating >= 7) return '#6C2BD9'
   188→  if (rating >= 5) return '#FF6B35'
   189→  if (rating >= 3) return '#EF4444'
   190→  return '#DC2626'
   191→}

   242→  const [apiMatches, setApiMatches] = useState<Array<{
   243→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   244→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   245→  }>>([])
   246→
   247→  useEffect(() => {
   248→    async function fetchMatches() {
   249→      try {
   250→        const res = await fetch('/api/matches?league=WC')
   251→        if (res.ok) {
   252→          const data = await res.json()
   253→          const mapped = (data.matches || []).map((m: any) => ({
   254→            id: m.id,
   255→            home: m.homeTeam.code,
   256→            away: m.awayTeam.code,
   257→            homeFlag: m.homeTeam.flag || '',
   258→            awayFlag: m.awayTeam.flag || '',
   259→            score: m.score,
   260→            homeSentiment: Math.round(m.homeTeam.sentiment),
   261→            awaySentiment: Math.round(m.awayTeam.sentiment),
   262→            live: m.status === 'live',
   263→            league: m.league === 'WC' ? `WC Group ${m.group}` : m.league,
   264→          }))
   265→          setApiMatches(mapped)
   266→        }
   267→      } catch (err) {
   268→        console.error('Failed to fetch matches:', err)
   269→      }
   270→    }
   271→    fetchMatches()
   272→  }, [])
   273→
   274→  const filteredMatches = (matchFilter === 'ALL'
   275→    ? apiMatches
