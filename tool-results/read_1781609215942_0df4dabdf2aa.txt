   153→function getTrendIcon(trend: string) {
   154→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   155→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   156→  return <Minus className="size-3 text-[#FF6B35]" />
   157→}
   158→
   159→function getSentimentColor(score: number) {
   160→  if (score >= 80) return 'text-[#10B981]'
   161→  if (score >= 50) return 'text-[#FF6B35]'
   162→  return 'text-[#EF4444]'
   163→}
   164→
   165→function getSentimentBg(score: number) {
   166→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   167→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   168→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   169→}
   170→
   171→function getProgressClass(score: number) {
   172→  if (score >= 80) return 'progress-emerald'
   173→  if (score >= 50) return 'progress-amber'
   174→  return 'progress-red'
   175→}

   177→// Face emoji mapping for /10 rating system
   178→function getPulseFaceEmoji(pulseScore: number): string {
   179→  if (pulseScore >= 90) return '🤩'
   180→  if (pulseScore >= 70) return '😊'
   181→  if (pulseScore >= 50) return '😐'
   182→  if (pulseScore >= 30) return '😟'
   183→  return '😵'
   184→}
   185→
   186→function getRatingColor(rating: number): string {
   187→  if (rating >= 9) return '#10B981'
   188→  if (rating >= 7) return '#6C2BD9'
   189→  if (rating >= 5) return '#FF6B35'
   190→  if (rating >= 3) return '#EF4444'
   191→  return '#DC2626'
   192→}

   240→function HomeTab() {
   241→  const { t } = useLanguage()
   242→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   243→  const [apiMatches, setApiMatches] = useState<Array<{
   244→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   245→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   246→  }>>([])
   247→
   248→  useEffect(() => {
   249→    async function fetchMatches() {
   250→      try {
   251→        const res = await fetch('/api/matches?league=WC')
   252→        if (res.ok) {
   253→          const data = await res.json()
   254→          const mapped = (data.matches || []).map((m: any) => ({
   255→            id: m.id,
   256→            home: m.homeTeam.code,
   257→            away: m.awayTeam.code,
   258→            homeFlag: m.homeTeam.flag || '',
   259→            awayFlag: m.awayTeam.flag || '',
   260→            score: m.score,
   261→            homeSentiment: Math.round(m.homeTeam.sentiment),
   262→            awaySentiment: Math.round(m.awayTeam.sentiment),
   263→            live: m.status === 'live',
   264→            league: m.league === 'WC' ? `WC Group ${m.group}` : m.league,
   265→          }))
   266→          setApiMatches(mapped)
   267→        }
   268→      } catch (err) {
   269→        console.error('Failed to fetch matches:', err)
   270→      }
   271→    }
   272→    fetchMatches()
   273→  }, [])
   274→
   275→  const filteredMatches = (matchFilter === 'ALL'
   276→    ? apiMatches
   277→    : apiMatches.filter(m => m.league.startsWith('WC'))
   278→  ).slice(0, 24) // Limit to 24 cards max for performance
   279→
   280→  return (
   281→    <div className="space-y-6">
   282→      {/* Hero */}
   283→      <motion.div
   284→        initial={{ opacity: 0, y: 20 }}
   285→        animate={{ opacity: 1, y: 0 }}
   286→        transition={{ duration: 0.5 }}
   287→        className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-[#6C2BD9]/8 via-white dark:via-[#1A1A1A] to-[#FF6B35]/5 p-6 border border-[#6C2BD9]/10 dark:border-[#6C2BD9]/20"
   288→      >
   289→        <div className="absolute -right-8 -top-8 size-32 rounded-full bg-[#6C2BD9]/5 blur-3xl" />
   290→        <div className="absolute -bottom-8 -left-8 size-32 rounded-full bg-[#FF6B35]/5 blur-3xl" />
   291→        <h2 className="relative text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
   292→          {t('home.your_pulse')} <span className="text-[#6C2BD9]">⚡</span>
   293→        </h2>
   294→        <p className="relative mt-2 text-sm text-[#666] dark:text-[#CCCCCC]">
   295→          {t('home.mood_desc')}
   296→        </p>
   297→        <div className="relative mt-4 flex items-center gap-3">
   298→          <div className="flex items-center gap-1.5 rounded-full bg-[#10B981]/10 px-3 py-1.5 text-xs font-semibold text-[#10B981]">
   299→            <Zap className="size-3.5" />
   300→            78% {t('home.positive')}
   301→          </div>
   302→          <div className="flex items-center gap-1.5 rounded-full bg-[#EF4444]/10 px-3 py-1.5 text-xs font-semibold text-[#EF4444]">
   303→            <Activity className="size-3.5" />
   304→            {apiMatches.filter(m => m.live).length} {t('home.live')}
   305→          </div>
   306→        </div>
   307→      </motion.div>
   308→
   309→      {/* Featured Matches */}
   310→      <div>
   311→        <div className="mb-3 flex items-center justify-between">
   312→          <h3 className="text-xs font-bold uppercase tracking-wider text-[#666] dark:text-[#CCCCCC]">
   313→            {t('home.featured')}
   314→          </h3>
   315→          <div className="flex gap-1.5">
   316→            {(['WC', 'ALL'] as const).map((filter) => (
   317→              <button
   318→                key={filter}
   319→                onClick={() => setMatchFilter(filter)}
   320→                className={`
   321→                  rounded-full px-3 py-1 text-[10px] font-bold transition-all duration-200
   322→                  ${matchFilter === filter
   323→                    ? 'bg-[#6C2BD9] text-white shadow-sm'
   324→                    : 'bg-[#F8F9FA] dark:bg-[#2D2D2D] text-[#666] dark:text-[#CCCCCC] border border-[#E0E0E0] dark:border-white/10'
   325→                  }
   326→                `}
   327→              >
   328→                {filter === 'ALL' ? '⚽ All' : '🏆 World Cup'}
   329→              </button>

   331→          </div>
   332→        </div>
   333→        <div className="grid gap-3 sm:grid-cols-2">
   334→          {filteredMatches.map((match, i) => (
