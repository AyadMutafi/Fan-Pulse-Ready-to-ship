   153→function getFlag(nationCode: string): string {
   154→  const team = findNationalTeam(nationCode)
   155→  return team?.flag ?? '🏳️'
   156→}
   157→
   158→function getTrendIcon(trend: string) {
   159→  if (trend === 'rising') return <TrendingUp className="size-3 text-[#10B981]" />
   160→  if (trend === 'falling') return <TrendingDown className="size-3 text-[#EF4444]" />
   161→  return <Minus className="size-3 text-[#FF6B35]" />
   162→}
   163→
   164→function getSentimentColor(score: number) {
   165→  if (score >= 80) return 'text-[#10B981]'
   166→  if (score >= 50) return 'text-[#FF6B35]'
   167→  return 'text-[#EF4444]'
   168→}
   169→
   170→function getSentimentBg(score: number) {
   171→  if (score >= 80) return 'bg-[#10B981]/5 border-[#10B981]/15'
   172→  if (score >= 50) return 'bg-[#FF6B35]/5 border-[#10B981]/15'
   173→  return 'bg-[#EF4444]/5 border-[#EF4444]/15'
   174→}
   175→
   176→function getProgressClass(score: number) {
   177→  if (score >= 80) return 'progress-emerald'
   178→  if (score >= 50) return 'progress-amber'
   179→  return 'progress-red'
   180→}
   181→
   182→// Face emoji mapping for /10 rating system
   183→function getPulseFaceEmoji(pulseScore: number): string {
   184→  if (pulseScore >= 90) return '🤩'
   185→  if (pulseScore >= 70) return '😊'
   186→  if (pulseScore >= 50) return '😐'
   187→  if (pulseScore >= 30) return '😟'
   188→  return '😵'
   189→}
   190→
   191→function getRatingColor(rating: number): string {
   192→  if (rating >= 9) return '#10B981'
   193→  if (rating >= 7) return '#6C2BD9'
   194→  if (rating >= 5) return '#FF6B35'
   195→  if (rating >= 3) return '#EF4444'
   196→  return '#DC2626'
   197→}
