    26→// ── World Cup stage label helper ─────────────────────────────
    27→// Renders an accurate label for a WC match based on its `group` value.
    28→// Group-stage matches use letters A-L; knockout rounds use R32/R16/QF/SF/Final.
    29→// (Previously everything was labelled "WC Group {x}", which produced the
    30→// nonsensical "WC Group R32" for knockout matches.)
    31→function wcStageLabel(group: string | null | undefined): string {
    32→  if (!group) return 'World Cup 2026'
    33→  if (/^[A-L]$/.test(group)) return `WC Group ${group}`
    34→  switch (group) {
    35→    case 'R32': return 'WC Round of 32'
    36→    case 'R16': return 'WC Round of 16'
    37→    case 'QF': return 'WC Quarter Finals'
    38→    case 'SF': return 'WC Semi Finals'
    39→    case 'Final': return 'WC Final'
    40→    default: return 'World Cup 2026'
    41→  }
    42→}

   234→function HomeTab() {
   235→  const { t } = useLanguage()
   236→  const [matchFilter, setMatchFilter] = useState<'ALL' | 'WC'>('WC')
   237→  const [apiMatches, setApiMatches] = useState<Array<{
   238→    id: string; home: string; away: string; homeFlag: string; awayFlag: string
   239→    score: string; homeSentiment: number; awaySentiment: number; live: boolean; league: string
   240→    status: string; group: string; matchDate: string
   241→  }>>([])
   242→
   243→  // Fan vote state
   244→  const [sessionId, setSessionId] = useState<string>('')
   245→  const [fanVotes, setFanVotes] = useState<FanVoteAgg[]>([])
   246→  const [myVotes, setMyVotes] = useState<Array<{ teamCode: string; score: number }>>([])
   247→  const [votesLoading, setVotesLoading] = useState(true)
   248→  const [selectedVoteTeam, setSelectedVoteTeam] = useState<string | null>(null)
   249→  const [submitting, setSubmitting] = useState(false)
   250→  const [toast, setToast] = useState<{ msg: string; emoji: string } | null>(null)
   251→  const [fanCardOffer, setFanCardOffer] = useState<{ teamCode: string; score: number } | null>(null)
   252→
   253→  useEffect(() => {
   254→    async function fetchMatches() {
   255→      try {
   256→        const res = await fetch('/api/matches?league=WC')
   257→        if (res.ok) {
   258→          const data = await res.json()
   259→          const mapped = (data.matches || []).map((m: any) => ({
   260→            id: m.id,
   261→            home: m.homeTeam.code,
   262→            away: m.awayTeam.code,
   263→            homeFlag: m.homeTeam.flag || '',
   264→            awayFlag: m.awayTeam.flag || '',
   265→            score: m.score,
   266→            homeSentiment: Math.round(m.homeTeam.sentiment),
   267→            awaySentiment: Math.round(m.awayTeam.sentiment),
   268→            live: m.status === 'live',
   269→            league: m.league === 'WC' ? wcStageLabel(m.group) : m.league,
   270→            status: m.status || 'upcoming',
   271→            group: m.group || '',
   272→            matchDate: m.matchDate || '',
   273→          }))
   274→          setApiMatches(mapped)
   275→        }
   276→      } catch (err) {
   277→        console.error('Failed to fetch matches:', err)
   278→      }
   279→    }
   280→    fetchMatches()
   281→  }, [])
   282→
   283→  // SSR-safe: read/create sessionId inside useEffect, never during render.
   284→  useEffect(() => {
   285→    try {
   286→      const existing = typeof window !== 'undefined' ? window.localStorage.getItem('fan_session_id') : null
   287→      if (existing && existing.length > 0) {
   288→        setSessionId(existing)
   289→        return
   290→      }
   291→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
   292→        ? crypto.randomUUID()
   293→        : `anon-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`
   294→      if (typeof window !== 'undefined') {
   295→        window.localStorage.setItem('fan_session_id', newId)
   296→      }
   297→      setSessionId(newId)
   298→    } catch {
   299→      setSessionId(`anon-${Date.now()}`)
   300→    }
   301→  }, [])
   302→
   303→  // Fetch fan votes whenever sessionId changes.
   304→  useEffect(() => {
   305→    if (!sessionId) return
   306→    let cancelled = false
   307→    async function loadVotes() {
   308→      setVotesLoading(true)
   309→      try {
   310→        const res = await fetch(`/api/fan-vote?session=${encodeURIComponent(sessionId)}`)
   311→        if (!res.ok) return
   312→        const data = await res.json()
   313→        if (cancelled) return
   314→        setFanVotes(Array.isArray(data.votes) ? data.votes : [])
   315→        setMyVotes(Array.isArray(data.myVotes) ? data.myVotes : [])
   316→      } catch (err) {
   317→        console.error('Failed to fetch fan votes:', err)
   318→      } finally {
   319→        if (!cancelled) setVotesLoading(false)
   320→      }
   321→    }
   322→    loadVotes()
   323→    return () => { cancelled = true }
   324→  }, [sessionId])
   325→
   326→  // Close voting modal on Escape key (accessibility — modal was only closable
   327→  // via the explicit Close button or selecting a mood)
   328→  useEffect(() => {
   329→    if (!selectedVoteTeam) return
   330→    const onKey = (e: KeyboardEvent) => {
   331→      if (e.key === 'Escape' && !submitting) setSelectedVoteTeam(null)
   332→    }
   333→    window.addEventListener('keydown', onKey)
   334→    return () => window.removeEventListener('keydown', onKey)
   335→  }, [selectedVoteTeam, submitting])
   336→
   337→  const totalVoteCount = fanVotes.reduce((sum, v) => sum + (v.count || 0), 0)
   338→
   339→  const handleVote = async (teamCode: string, score: number) => {
   340→    if (!sessionId || submitting) return
   341→    setSubmitting(true)
   342→    // Optimistic update: bump my vote immediately
   343→    const prevMyVote = myVotes.find(v => v.teamCode === teamCode)?.score ?? null
   344→    setMyVotes(prev => {
   345→      const without = prev.filter(v => v.teamCode !== teamCode)
   346→      return [...without, { teamCode, score }]
   347→    })
   348→    setFanVotes(prev => {
   349→      const idx = prev.findIndex(v => v.teamCode === teamCode)
   350→      if (idx === -1) {
   351→        return [...prev, { teamCode, score, count: 1 }]
   352→      }
   353→      const next = [...prev]
   354→      const wasMine = prevMyVote !== null
   355→      // Adjust running average: if I had voted, swap my old score for the new.
   356→      const current = next[idx]
   357→      const totalScore = current.score * current.count
   358→      const newScore = wasMine
   359→        ? Math.round((totalScore - prevMyVote! + score) / current.count)
   360→        : Math.round((totalScore + score) / (current.count + 1))
   361→      const newCount = wasMine ? current.count : current.count + 1
   362→      next[idx] = { teamCode, score: newScore, count: newCount }
   363→      return next
   364→    })
   365→    setSelectedVoteTeam(null)
   366→    const mood = MOOD_EMOJI_OPTIONS.find(o => o.score === score)
   367→    setToast({ msg: `Vote recorded for ${teamCode}`, emoji: mood?.emoji ?? '✓' })
   368→    setTimeout(() => setToast(null), 2500)
   369→    // Show the Fan Card offer for 8 seconds — long enough to click "Get Fan Card"
   370→    setFanCardOffer({ teamCode, score })
   371→    const offerTimer = setTimeout(() => setFanCardOffer(null), 8000)
   372→    try {
   373→      await fetch('/api/fan-vote', {
   374→        method: 'POST',
   375→        headers: { 'Content-Type': 'application/json' },
   376→        body: JSON.stringify({ teamCode, score, sessionId }),
   377→      })
   378→    } catch (err) {
   379→      console.error('Failed to submit vote:', err)
   380→      // Revert on failure
   381→      setMyVotes(prev => {
   382→        const without = prev.filter(v => v.teamCode !== teamCode)
   383→        if (prevMyVote !== null) return [...without, { teamCode, score: prevMyVote }]
   384→        return without
   385→      })
   386→      setToast({ msg: 'Vote failed — please retry', emoji: '⚠️' })
   387→      setFanCardOffer(null)
   388→      clearTimeout(offerTimer)
   389→    } finally {
   390→      setSubmitting(false)
   391→    }
   392→  }
   393→
   394→  const filteredMatches = (matchFilter === 'ALL'
   395→    ? apiMatches
   396→    : apiMatches.filter(m => m.league.startsWith('WC'))
   397→  ).slice(0, 24) // Limit to 24 cards max for performance
   398→
   399→  const fanVoteIntelText = totalVoteCount === 0
   400→    ? 'Be the first to vote in the Fan Mood section below'
   401→    : `${totalVoteCount.toLocaleString()} fan votes tallied for World Cup 2026 Group Stage`
   402→
   403→  // ── Arena Intelligence: VERIFIED insights only ──────────────────────────
   404→  // Every insight below is a historical fact sourced from VERIFIED_DATA.md
   405→  // (the project's single source of truth, cross-checked against Wikipedia,
   406→  // ESPN, Olympics.com, FIFA.com on 2026-07-02). We deliberately do NOT
   407→  // auto-generate insights from apiMatches because that approach produced two
   408→  // verifiable hallucinations in production:
   409→  //   (1) "World Cup 2026 kicked off with ESP 0-0 AUT in the opener" — FALSE.
   410→  //       The old code used parsed[0] (first array element) as "the opener",
   411→  //       but the first row happened to be an UPCOMING R32 match (ESP vs AUT,
   412→  //       scheduled Jul 3, not yet played). The real opener was MEX 2-0 RSA
   413→  //       on Jun 11 (the earliest matchDate among completed matches).
   414→  //   (2) "Shock in Group Stage: ESP 0-0 AUT" — FALSE. The old code used
   415→  //       league.includes('WC') to label the stage as "Group Stage", but ALL
   416→  //       WC matches (including R32) match that check. ESP vs AUT is an R32
   417→  //       match, not a group-stage match. The real group-stage shock was
   418→  //       ESP 0-0 CPV (Spain held scoreless by Cape Verde).
   419→  //
   420→  // Fix: hardcode the verified insight set with explicit VERIFIED_DATA.md
   421→  // citations. Only the fan-vote count stays dynamic (it is live data).
   422→  // This guarantees we never describe an upcoming match as played, never
   423→  // mislabel a knockout match as a group-stage match, and never invent stats.
   424→  const arenaIntel = useMemo<Array<{ icon: typeof Sparkles; text: string; color: string; stage: string }>>(() => {

   426→

  1166→function RateTab() {
  1167→  const { t } = useLanguage()
  1168→  const [players, setPlayers] = useState<Array<{
  1169→    id: string
  1170→    playerName: string
  1171→    nationCode: string
  1172→    position: string
  1173→    avgRating: number
  1174→    totalRatings: number
  1175→  }>>([])
  1176→  const [myRatings, setMyRatings] = useState<Record<string, number>>({})
  1177→  const [loading, setLoading] = useState(true)
  1178→  const [submittingId, setSubmittingId] = useState<string | null>(null)
  1179→
  1180→  // ── Load real players from the DB + this session's ratings ──
  1181→  useEffect(() => {
  1182→    let cancelled = false
  1183→    async function load() {
  1184→      setLoading(true)
  1185→      try {
  1186→        const res = await fetch('/api/ratings')
  1187→        if (!res.ok) return
  1188→        const data = await res.json()
  1189→        if (cancelled) return
  1190→        const list = Array.isArray(data.ratings) ? data.ratings : []
  1191→        setPlayers(list)
  1192→      } catch {
  1193→        /* ignore — toast shown on submit failure */
  1194→      } finally {
  1195→        if (!cancelled) setLoading(false)
  1196→      }
  1197→    }
  1198→    load()
  1199→    return () => { cancelled = true }
  1200→  }, [])
  1201→
  1202→  // ── Per-browser session ID (crypto.randomUUID, persisted in localStorage) ──
  1203→  // Required by the API — the old 'anonymous' default is rejected. A real
  1204→  // session ID makes the (sessionId, playerId) unique constraint meaningful,
  1205→  // enforcing one rating per session per player.
  1206→  const [sessionId, setSessionId] = useState<string>('')
  1207→  useEffect(() => {
  1208→    try {
  1209→      const existing = window.localStorage.getItem('fp_session_id')
  1210→      if (existing && existing.length >= 8) {
  1211→        setSessionId(existing)
  1212→        return
  1213→      }
  1214→      const newId = (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function')
  1215→        ? crypto.randomUUID()
  1216→        : `fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`
  1217→      window.localStorage.setItem('fp_session_id', newId)
  1218→      setSessionId(newId)
  1219→    } catch {
  1220→      setSessionId(`fp-${Date.now()}-${Math.random().toString(36).slice(2, 12)}`)
  1221→    }
  1222→  }, [])
  1223→
  1224→  const handleRate = async (playerId: string, rating: number) => {
  1225→    if (!sessionId || submittingId === playerId) return
  1226→    setSubmittingId(playerId)
  1227→    // Optimistically update local state so the stars fill instantly.
  1228→    setMyRatings(prev => ({ ...prev, [playerId]: rating }))
  1229→    try {
  1230→      const res = await fetch('/api/ratings', {
  1231→        method: 'POST',
  1232→        headers: { 'Content-Type': 'application/json' },
  1233→        body: JSON.stringify({ playerId, rating, sessionId }),
  1234→      })
  1235→      const data = await res.json().catch(() => null)
  1236→      if (!res.ok) {
  1237→        // Revert on failure.
  1238→        setMyRatings(prev => {
  1239→          const next = { ...prev }
  1240→          delete next[playerId]
  1241→          return next
  1242→        })
  1243→        if (res.status === 429) {
  1244→          toast.error('Too many ratings — please slow down.')
  1245→        } else {
  1246→          toast.error(data?.error || 'Failed to submit rating')
  1247→        }
  1248→        return
  1249→      }
  1250→      toast.success(data?.updated ? 'Rating updated!' : 'Rating submitted!', {
  1251→        description: `${rating}/10`,
  1252→      })
  1253→      // Refresh aggregates so the avg display updates.
  1254→      const fresh = await fetch('/api/ratings')
  1255→      if (fresh.ok) {
  1256→        const fd = await fresh.json().catch(() => null)
  1257→        if (Array.isArray(fd?.ratings)) setPlayers(fd.ratings)
  1258→      }
  1259→    } catch {
  1260→      setMyRatings(prev => {
  1261→        const next = { ...prev }
  1262→        delete next[playerId]
  1263→        return next
  1264→      })
  1265→      toast.error('Network error — please try again')
  1266→    } finally {
  1267→      setSubmittingId(null)
  1268→    }
  1269→  }
  1270→
  1271→  return (
  1272→    <div className="space-y-6">
  1273→      <motion.div
  1274→        initial={{ opacity: 0, y: 20 }}
  1275→        animate={{ opacity: 1, y: 0 }}

  1278→        <h2 className="text-2xl font-bold tracking-tight text-[#1A1A1A] dark:text-white">
  1279→          {t('ratings.title')}

  1282→      </motion.div>
  1283→

  1285→        {loading ? (
  1286→          <Card className="border-[#E0E0E0]/50 dark:border-white/5">
  1287→            <CardContent className="p-6 text-center text-sm text-[#666] dark:text-[#CCCCCC]">
  1288→              Loading players…
  1289→            </CardContent>
  1290→          </Card>
  1291→        ) : players.length === 0 ? (
  1292→          <Card className="border-[#E0E0E0]/50 dark:border-white/5">
  1293→            <CardContent className="p-6 text-center text-sm text-[#666] dark:text-[#CCCCCC]">
  1294→              No rateable players found.
  1295→            </CardContent>
  1296→          </Card>
  1297→        ) : (
  1298→          players.map((player, i) => (
  1299→            <motion.div
  1300→              key={player.id}
  1301→              initial={{ opacity: 0, y: 10 }}
  1302→              animate={{ opacity: 1, y: 0 }}
