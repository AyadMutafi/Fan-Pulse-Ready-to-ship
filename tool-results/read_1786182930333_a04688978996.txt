     1→/**
     2→ * Transfer Pulse — Discovery pipeline.
     3→ *
     4→ * For each tracked player, asks Grok's x_search tool for X posts about their
     5→ * transfer, then keeps ONLY the posts authored by a Tier 1 journalist
     6→ * (handle verified against TIER1_HANDLES). Each such post anchors a saga.
     7→ *
     8→ * ANTI-HALLUCINATION CONTRACT:
     9→ *   - A saga is ONLY created if a Tier 1 journalist's OWN X post is found.
    10→ *     A fan quoting Fabrizio Romano is rejected (the fan's handle != Romano's).
    11→ *   - The destination club is extracted from the journalist's post text by an
    12→ *     LLM. If the LLM cannot confidently determine a destination, the post is
    13→ *     discarded — we never guess.
    14→ *   - ENTITY RESOLUTION (added 2026-07-22): before a saga is created, an LLM
    15→ *     verifies that the player's current club matches `fromClubName`. This
    16→ *     prevents the "Ederson" entity-confusion incident where Tier 1 posts
    17→ *     about the Atalanta MF Ederson were misattributed to the Man City GK
    18→ *     Ederson, fabricating a "Man City → Atalanta" transfer that doesn't
    19→ *     exist. If verification fails, the post is rejected.
    20→ *   - The journalist's real X post URL becomes the TransferSource.url. The
    21→ *     URL must match the real X post pattern (enforced in grok-x-search).
    22→ *   - Duplicate sources (same URL) are never double-counted (@unique).
    23→ *   - Debunked sagas are never deleted — only their status changes.
    24→ */

    56→/**
    57→ * Discover transfer sagas for a batch of tracked players.
    58→ *
    59→ * Processes players in a rotating window so the cron can cycle through the
    60→ * full watchlist without hammering the xAI API.
    61→ *
    62→ * SYSTEMIC STALENESS GUARD (added 2026-07-26):
    63→ *   Before running discovery for a player, `checkPlayerAlreadyMoved()` asks
    64→ *   the AI whether the player has already completed a transfer AWAY from
    65→ *   their watchlist `fromClubName`. If yes, ALL of that player's active
    66→ *   sagas are marked `completed` (or `debunked` if the actual destination
    67→ *   differs from the saga's `toClubName`), and the player is SKIPPED for
    68→ *   the rest of this discovery run. This prevents the "Wirtz → Man City"
    69→ *   class of bug where a player who already moved keeps getting surfaced
    70→ *   as an active rumor based on stale pre-move Tier 1 posts.
    71→ *
    72→ * @param opts.maxPlayers  how many players to scan this run (default 5)
    73→ * @param opts.offset      starting index into TRACKED_PLAYERS (default 0)
    74→ * @param opts.playerName  scan a single named player instead of a batch
    75→ */
    76→export async function discoverTransferSagas(opts: {
    77→  maxPlayers?: number
    78→  offset?: number
    79→  playerName?: string
    80→} = {}): Promise<DiscoveryResult> {
    81→  const startedAt = Date.now()
    82→  const result: DiscoveryResult = {
    83→    playersScanned: 0,
    84→    sagasCreated: 0,
    85→    sagasUpdated: 0,
    86→    sourcesAdded: 0,
    87→    skipped: 0,
    88→    errors: [],
    89→    durationMs: 0,
    90→  }
    91→
    92→  // Resolve the batch of players to scan
    93→  let batch: TrackedPlayer[]
    94→  if (opts.playerName) {
    95→    const found = TRACKED_PLAYERS.find(
    96→      (p) => p.name.toLowerCase() === opts.playerName!.toLowerCase(),
    97→    )
    98→    batch = found ? [found] : []
    99→    if (batch.length === 0) {
   100→      result.errors.push(`Player not in watchlist: ${opts.playerName}`)
   101→      result.durationMs = Date.now() - startedAt
   102→      return result
   103→    }
   104→  } else {
   105→    const max = opts.maxPlayers ?? 5
   106→    const offset = opts.offset ?? 0
   107→    batch = TRACKED_PLAYERS.slice(offset, offset + max)
   108→  }
   109→
   110→  for (const player of batch) {
   111→    result.playersScanned++
   112→    try {
   113→      // ── STALENESS GUARD ──────────────────────────────────────────────────
   114→      // Before spending xAI/LLM budget on discovery, check whether the player
   115→      // has ALREADY moved. If so, resolve their sagas and skip discovery.
   116→      const movedCheck = await checkPlayerAlreadyMoved(player)
   117→      if (movedCheck.alreadyMoved) {
   118→        console.log(
   119→          `[transfer-pulse/discovery] ${player.name} already moved to ${movedCheck.actualClub} ` +
   120→            `(watchlist had ${player.fromClubName}) — resolving sagas and skipping discovery`,
   121→        )
   122→        const resolved = await resolvePlayerSagas(
   123→          player,
   124→          movedCheck.actualClub ?? 'unknown',
   125→          movedCheck.confidence,
   126→        )
   127→        result.sagasUpdated += resolved
   128→        result.skipped += 1
   129→        continue
   130→      }
   131→
   132→      const outcome = await discoverForPlayer(player)
   133→      result.sagasCreated += outcome.sagasCreated
   134→      result.sagasUpdated += outcome.sagasUpdated
   135→      result.sourcesAdded += outcome.sourcesAdded
   136→      result.skipped += outcome.skipped
   137→      if (outcome.error) result.errors.push(`${player.name}: ${outcome.error}`)
   138→    } catch (err) {
   139→      result.errors.push(`${player.name}: ${String(err).slice(0, 200)}`)
   140→    }
   141→  }
   142→
   143→  result.durationMs = Date.now() - startedAt
   144→  return result
   145→}

   147→// ── Per-player discovery ─────────────────────────────────────────────────────
   148→
   149→async function discoverForPlayer(
   150→  player: TrackedPlayer,
   151→): Promise<{
   152→  sagasCreated: number
   153→  sagasUpdated: number
   154→  sourcesAdded: number
   155→  skipped: number
   156→  error?: string
   157→}> {
   158→  const out = { sagasCreated: 0, sagasUpdated: 0, sourcesAdded: 0, skipped: 0 }
   159→
   160→  // 1. Search X for posts about this player's transfer
   161→  const query =
   162→    `Find real X (Twitter) posts by football transfer journalists reporting ` +
   163→    `a possible transfer of ${player.name} (currently at ${player.fromClubName}) ` +
   164→    `during the summer 2026 window. Look especially for posts from Fabrizio ` +
   165→    `Romano, David Ornstein, Florian Plettenberg, Matteo Moretto, and other ` +
   166→    `known transfer journalists. Return posts that name a destination club.`
   167→  const today = new Date().toISOString().slice(0, 10)
   168→  const fromDate = new Date(Date.now() - 60 * 24 * 60 * 60 * 1000)
   169→    .toISOString()
   170→    .slice(0, 10)
   171→
   172→  const search = await searchXPostsGeneric({
   173→    query,
   174→    fromDate,
   175→    toDate: today,
   176→  })
   177→
   178→  // ── Tier 1 post collection ─────────────────────────────────────────────
   179→  // Primary: xAI x_search (if XAI_API_KEY configured).
   180→  // Fallback: Z.ai web_search with site:x.com queries (works in sandbox
   181→  //   without an explicit API key — the SDK auto-initializes).
   182→  // We MERGE both sources and dedupe by URL so the widest coverage wins.
   183→  let tier1Posts: XPost[] = []
   184→  let primaryError: string | undefined
   185→
   186→  if (search.error) {
   187→    primaryError = search.error
   188→    console.log(
   189→      `[transfer-pulse/discovery] xAI search unavailable for ${player.name}: ${search.error.slice(0, 100)}`,
   190→    )
   191→  } else {
   192→    // Filter to Tier 1 journalists AND enforce freshness (≤60 days).
   193→    // The Snowflake ID decoder gives us the real post date from the URL,
   194→    // so this works even if xAI doesn't return a postedAt timestamp.
   195→    const MAX_TIER1_AGE_DAYS = 60
   196→    tier1Posts = search.posts.filter((p) => {
   197→      if (!TIER1_HANDLES.has(p.handle.toLowerCase())) return false
   198→      // Resolve the post date: postedAt field, or Snowflake decode fallback
   199→      let date: Date | null = null
   200→      if (p.postedAt) {
   201→        const d = new Date(p.postedAt)
   202→        if (!isNaN(d.getTime())) date = d
   203→      }
   204→      if (!date) {
   205→        const statusId = extractStatusId(p.url)
   206→        if (statusId) date = decodeSnowflakeDate(statusId)
   207→      }
   208→      if (!date) return false // no date = reject (anti-hallucination)
   209→      const ageMs = Date.now() - date.getTime()
   210→      return ageMs >= 0 && ageMs <= MAX_TIER1_AGE_DAYS * 24 * 3600 * 1000
   211→    })
   212→    console.log(
   213→      `[transfer-pulse/discovery] xAI: ${search.posts.length} posts, ${tier1Posts.length} fresh Tier 1 for ${player.name}`,
   214→    )
   215→  }
   216→
   217→  // If xAI gave us nothing (key missing OR 0 Tier 1 posts), try Z.ai fallback
   218→  if (tier1Posts.length === 0) {
   219→    console.log(`[transfer-pulse/discovery] trying Z.ai fallback for ${player.name}`)
   220→    // maxAgeDays=60: only accept Tier 1 reports from the last 60 days.
   221→    // This prevents last year's tweets from appearing as "current news."
   222→    const zaiResult = await fetchTier1PostsViaZai(player, { maxAgeDays: 60 })
   223→    if (zaiResult.error) {
   224→      console.log(
   225→        `[transfer-pulse/discovery] Z.ai fallback error: ${zaiResult.error.slice(0, 100)}`,
   226→      )
   227→    } else if (zaiResult.posts.length > 0) {
   228→      // Merge + dedupe by URL
   229→      const seen = new Set(tier1Posts.map((p) => p.url))
   230→      for (const p of zaiResult.posts) {
   231→        if (!seen.has(p.url)) {
   232→          tier1Posts.push(p)
   233→          seen.add(p.url)
   234→        }
   235→      }
   236→      console.log(
   237→        `[transfer-pulse/discovery] Z.ai fallback: +${zaiResult.posts.length} fresh Tier 1 posts for ${player.name}`,
   238→      )
   239→    }
   240→  }
   241→
   242→  if (tier1Posts.length === 0) {
   243→    // No Tier 1 anchor from either source → no saga created (anti-hallucination)
   244→    out.skipped = 1
   245→    if (primaryError) out.error = primaryError
   246→    return out
   247→  }
   248→
   249→  // ── ENTITY-RESOLUTION GATE (added 2026-07-22) ──────────────────────────────
   250→  // Verify ONCE per player that the Tier 1 posts are actually about THIS
   251→  // tracked player (current club = fromClubName). Without this gate, posts
   252→  // about the Atalanta MF "Ederson" can be misattributed to the Man City GK
   253→  // "Ederson", fabricating a "Man City → Atalanta" saga that doesn't exist.
   254→  //
   255→  // We pass ALL the Tier 1 posts in one LLM call (concatenated) so we only
   256→  // make one verification request per player, not one per post — important
   257→  // for staying under Z.ai's rate limit. If verification fails, ALL posts
   258→  // for this player are rejected.
   259→  //
   260→  // Skip the gate if the player already has any existing saga in the DB —
   261→  // we trust prior verification and only need to add new Tier 1 sources.
   262→  const hasExistingSaga = await db.transferSaga.findFirst({
   263→    where: { playerName: player.name },
   264→    select: { id: true },
   265→  })
   266→  if (!hasExistingSaga) {
   267→    const combinedPostText = tier1Posts
   268→      .slice(0, 5)
   269→      .map((p, i) => `POST ${i + 1} (@${p.handle}):\n${p.text.slice(0, 400)}`)
   270→      .join('\n\n')
   271→    const verification = await verifyPlayerCurrentClub(
   272→      combinedPostText,
   273→      player.name,
   274→      player.fromClubName,
   275→    )
   276→    if (!verification.verified) {
   277→      console.log(
   278→        `[transfer-pulse/discovery] rejecting ALL Tier 1 posts for ${player.name}: ` +
   279→          `posts are about "${verification.actualClub ?? 'unknown'}", not "${player.fromClubName}" — ` +
   280→          `likely a same-name-different-player confusion (e.g. Ederson GK vs Ederson MF)`,
   281→      )
   282→      out.skipped = 1
   283→      return out
   284→    }
   285→  }
   286→
   287→  // 3. For each Tier 1 post, extract destination club + upsert saga + source
   288→  for (const post of tier1Posts) {
   289→    const source = getTier1Source(post.handle)
   290→    if (!source) continue
   291→
   292→    const extracted = await extractTransferFields(post.text, player.name)
   293→    if (!extracted || !extracted.toClubName || !extracted.toClubCode) {
   294→      // Could not confidently determine destination → skip this post
   295→      continue
   296→    }
   297→
   298→    const toClubCode = extracted.toClubCode.toUpperCase()
   299→    const toClubName = extracted.toClubName
   300→    const reportedAt = post.postedAt ? safeParseDate(post.postedAt) : new Date()
   301→
   302→    // ── SAME-CLUB GUARD (added 2026-07-21) ─────────────────────────────────
   303→    // If the LLM extracted a destination that matches the player's CURRENT club,
   304→    // this is a contract renewal / "stay" post, NOT a transfer. Reject it.
   305→    // This fixes the "Bruno Fernandes Man United → Manchester United" and
   306→    // "Marcus Rashford Man United → Manchester United" bad sagas where Romano
   307→    // posts about contract talks were misclassified as same-club transfers.
   308→    if (
   309→      toClubCode.toUpperCase() === player.fromClubCode.toUpperCase() ||
   310→      toClubName.toLowerCase().includes(player.fromClubName.toLowerCase()) ||
   311→      player.fromClubName.toLowerCase().includes(toClubName.toLowerCase())
   312→    ) {
   313→      console.log(
   314→        `[transfer-pulse/discovery] rejecting same-club extraction for ${player.name}: ` +
   315→          `"${toClubName}" matches current club "${player.fromClubName}" — likely a contract-renewal post, not a transfer`,
   316→      )
   317→      continue
   318→    }
   319→
   320→    // ── ENTITY-NAME-OVERLAP GUARD (added 2026-07-26) ───────────────────────
   321→    // Reject posts where the text mentions a DIFFERENT player whose name
   322→    // CONTAINS the tracked player's name as a substring — e.g. a post about
   323→    // "Álvaro Rodríguez" should NOT create a saga for "Rodri" (Man City MF),
   324→    // even though "Rodríguez" contains "Rodri". This class of false-match
   325→    // created the bad "Rodri → Bournemouth" saga from a Romano post that was
   326→    // actually about Álvaro Rodríguez (Elche → Bournemouth).
   327→    //
   328→    // Heuristic: if the post text contains a longer name that starts with or
   329→    // contains the tracked player's name AND the tracked player's EXACT full
   330→    // name does NOT appear as a standalone token in the post, reject it.
   331→    const entityOverlap = hasEntityNameOverlap(post.text, player.name)
   332→    if (entityOverlap) {
   333→      console.log(
   334→        `[transfer-pulse/discovery] rejecting entity-name-overlap for ${player.name}: ` +
   335→          `post appears to be about "${entityOverlap}" (a different player whose name contains "${player.name}")`,
   336→      )
   337→      continue
   338→    }
   339→
   340→    // Upsert the saga (one per player + destination club)
   341→    const existing = await db.transferSaga.findUnique({
   342→      where: {
   343→        playerName_toClubCode: {
   344→          playerName: player.name,
   345→          toClubCode,
   346→        },
   347→      },
   348→    })
   349→
   350→    let sagaId: string
   351→    if (existing) {
   352→      // Update metadata; keep status unless the journalist confirmed completion
   353→      const newStatus =
   354→        extracted.isCompleted && existing.status === 'active'
   355→          ? 'completed'
   356→          : existing.status
   357→      await db.transferSaga.update({
   358→        where: { id: existing.id },
   359→        data: {
   360→          fromClubCode: player.fromClubCode,
   361→          fromClubName: player.fromClubName,
   362→          toClubName,
   363→          feeReported: extracted.fee || existing.feeReported,
   364→          playerNationCode: player.nationCode,
   365→          status: newStatus,
   366→          resolvedAt:
   367→            newStatus === 'completed' && !existing.resolvedAt
   368→              ? reportedAt
   369→              : existing.resolvedAt,
   370→          lastUpdatedAt: new Date(),
   371→          tier1Count: await countTier1Sources(existing.id),
   372→        },
   373→      })
   374→      sagaId = existing.id
   375→      out.sagasUpdated++
   376→    } else {
   377→      const created = await db.transferSaga.create({
   378→        data: {
   379→          playerName: player.name,
   380→          playerNationCode: player.nationCode,
   381→          fromClubCode: player.fromClubCode,
   382→          fromClubName: player.fromClubName,
   383→          toClubCode,
   384→          toClubName,
   385→          status: extracted.isCompleted ? 'completed' : 'active',
   386→          feeReported: extracted.fee || '',
   387→          firstReportedAt: reportedAt,
   388→          lastUpdatedAt: new Date(),
   389→          resolvedAt: extracted.isCompleted ? reportedAt : null,
   390→        },
   391→      })
   392→      sagaId = created.id
   393→      out.sagasCreated++
   394→    }
   395→
   396→    // Upsert the Tier 1 source (unique on URL — never double-counted)
   397→    const existingSource = await db.transferSource.findUnique({
   398→      where: { url: post.url },
   399→    })
   400→    if (!existingSource) {
   401→      await db.transferSource.create({
   402→        data: {
   403→          sagaId,
   404→          journalistName: source.name,
   405→          // Store the handle WITHOUT the leading '@' to match existing DB
   406→          // rows and the UI render convention (`@{s.journalistHandle}` in
   407→          // TransferSagaDetail.tsx). Tier1Source.handle carries the '@' for
   408→          // display, so we strip it here.
   409→          journalistHandle: source.handle.replace(/^@/, ''),
   410→          tier: 1,
   411→          url: post.url,
   412→          headline: extracted.headline.slice(0, 280),
   413→          outlet: source.outlet,
   414→          reportedAt,
   415→        },
   416→      })
   417→      out.sourcesAdded++
   418→      // Recompute tier1Count for the saga
   419→      await db.transferSaga.update({
   420→        where: { id: sagaId },
   421→        data: { tier1Count: await countTier1Sources(sagaId) },
   422→      })
   423→    }
   424→  }
   425→
   426→  return out
   427→}

   435→// ── LLM extraction of transfer fields ────────────────────────────────────────
   436→
   437→/**
   438→ * Ask the AI facade (Grok → Cerebras → Groq → Z.ai) to extract structured
   439→ * transfer metadata from a Tier 1 journalist's X post text. Returns null if
   440→ * the destination cannot be confidently determined — we never guess.
   441→ */
   442→async function extractTransferFields(
   443→  postText: string,
   444→  playerName: string,
   445→): Promise<ExtractedTransfer | null> {
   446→  const systemPrompt =
   447→    `You extract structured transfer-rumor metadata from a football journalist's ` +
   448→    `X post. The player is ${playerName}. Read the post text and return a JSON ` +
   449→    `object with these fields:\n` +
   450→    `  "toClubName": string | null   — the destination club's full name, or null if not stated\n` +
   451→    `  "toClubCode": string | null   — a 3-4 letter uppercase code for the destination (e.g. "LIV", "RMA", "SAU"), or null\n` +
   452→    `  "fee": string | null          — the reported transfer fee as a short string (e.g. "£150m", "€80m", "free transfer"), or null\n` +
   453→    `  "headline": string            — a one-line neutral summary of what the journalist reported (max 140 chars)\n` +
   454→    `  "isCompleted": boolean        — true if the journalist says the deal is DONE/confirmed/signed, false if it's still a rumor\n\n` +
   455→    `RULES:\n` +
   456→    `- If the post does NOT clearly name a destination club, return toClubName=null and toClubCode=null.\n` +
   457→    `- Do NOT invent a club. If unsure, return null.\n` +
   458→    `- Output ONLY the JSON object, no commentary.`
   459→
   460→  const result = await ai.chat(
   461→    [
   462→      { role: 'system', content: systemPrompt },
   463→      { role: 'user', content: postText.slice(0, 1000) },
   464→    ],
   465→    { temperature: 0.1, maxTokens: 600, json: true },
   466→  )
   467→
   468→  if (!result.ok || !result.content) {
   469→    console.warn(
   470→      `[transfer-pulse/discovery] extractTransferFields: ai.chat failed (${result.provider}): ${result.error?.slice(0, 120)}`,
   471→    )
   472→    return null
   473→  }
   474→
   475→  return parseExtractedTransfer(result.content)
   476→}
   477→
   478→function parseExtractedTransfer(raw: string): ExtractedTransfer | null {
   479→  let cleaned = raw.trim()
   480→  cleaned = cleaned.replace(/^```(?:json)?\s*/i, '').replace(/\s*```\s*$/i, '')
   481→  const start = cleaned.indexOf('{')
   482→  const end = cleaned.lastIndexOf('}')
   483→  if (start === -1 || end === -1 || end <= start) return null
   484→  try {
   485→    const obj = JSON.parse(cleaned.slice(start, end + 1))
   486→    const toClubName =
   487→      typeof obj.toClubName === 'string' && obj.toClubName.trim()
   488→        ? obj.toClubName.trim()
   489→        : null
   490→    const toClubCode =
   491→      typeof obj.toClubCode === 'string' && obj.toClubCode.trim()
   492→        ? obj.toClubCode.trim().toUpperCase().slice(0, 4)
   493→        : null
   494→    const fee =
   495→      typeof obj.fee === 'string' && obj.fee.trim() ? obj.fee.trim() : null
   496→    const headline =
   497→      typeof obj.headline === 'string' && obj.headline.trim()
   498→        ? obj.headline.trim().slice(0, 280)
   499→        : ''
   500→    const isCompleted = Boolean(obj.isCompleted)
   501→    // Require BOTH name and code to be confident about the destination
   502→    if (!toClubName || !toClubCode) return null
   503→    return { toClubName, toClubCode, fee, headline, isCompleted }
   504→  } catch {
   505→    return null
   506→  }
   507→}

   509→// ── Entity-resolution gate ───────────────────────────────────────────────────
   510→
   511→/**
   512→ * ENTITY RESOLUTION — verify that a Tier 1 post is actually about THIS
   513→ * tracked player (the one currently at `expectedFromClub`), not a same-name
   514→ * different-player.
   515→ *
   516→ * BACKGROUND:
   517→ *   In July 2026, Romano posted reports about "Ederson" being linked with
   518→ *   Manchester United. Our tracked-players list had "Ederson" as the Man City
   519→ *   GK. The discovery pipeline extracted "Manchester United" as the
   520→ *   destination and fabricated a "Man City → Manchester United" saga — but
   521→ *   Romano was actually reporting on the Atalanta MF Ederson (also Brazilian,
   522→ *   also "Ederson"). The same-name confusion produced a fabricated saga.
   523→ *
   524→ * GATE:
   525→ *   Before creating a new saga, we ask the LLM: "Given this post and this
   526→ *   player's expected current club, is the post actually about THIS player?"
   527→ *   The LLM returns {verified: bool, actualClub: string | null}. If
   528→ *   verified=false, the post is rejected and no saga is created.
   529→ *
   530→ * FALLBACK:
   531→ *   If the LLM call fails entirely, we FAIL OPEN (verified=true) to avoid
   532→ *   blocking all discovery — but log a warning. The same-club guard and
   533→ *   Tier 1 handle gate still apply as secondary defenses.
   534→ */
   535→async function verifyPlayerCurrentClub(
   536→  postText: string,
   537→  playerName: string,
   538→  expectedFromClub: string,
   539→): Promise<{ verified: boolean; actualClub: string | null }> {
   540→  const systemPrompt =
   541→    `You are an entity-resolution gate for a football transfer pipeline.\n` +
   542→    `A Tier 1 journalist's X post is being considered for a saga about ${playerName} ` +
   543→    `(expected current club: ${expectedFromClub}).\n\n` +
   544→    `Read the post and answer ONE question: is this post about ${playerName} ` +
   545→    `(the footballer who currently plays for ${expectedFromClub}), or is it about ` +
   546→    `a DIFFERENT footballer who happens to share the name?\n\n` +
   547→    `IMPORTANT CONTEXT:\n` +
   548→    `- Transfer rumor posts naturally mention the DESTINATION club (where the player ` +
   549→    `  might move TO). The destination is NOT the player's current club. So a post ` +
   550→    `  saying "${playerName} to Al-Hilal?" is consistent with the player being at ` +
   551→    `  ${expectedFromClub} — Al-Hilal is the rumored destination, not the current club.\n` +
   552→    `- This gate exists ONLY to catch same-name confusion (e.g. two different Brazilian ` +
   553→    `  players both named "Ederson" — one a GK at Man City, the other a MF at Atalanta).\n` +
   554→    `- For unique names like "Mohamed Salah", "Kylian Mbappé", "Erling Haaland", ` +
   555→    `  "Lamine Yamal", "Vinícius Júnior", "Jude Bellingham" — there is only ONE ` +
   556→    `  famous footballer by that name. The gate should ALMOST ALWAYS return verified=true.\n\n` +
   557→    `Return a JSON object with these fields:\n` +
   558→    `  "verified": boolean    — true if the post is about ${playerName} (the footballer at ${expectedFromClub})\n` +
   559→    `  "actualClub": string | null — only set if verified=false AND you can identify the other player's actual club\n` +
   560→    `  "reason": string       — a 1-sentence explanation\n\n` +
   561→    `RULES:\n` +
   562→    `- Default to verified=true UNLESS the post explicitly mentions a DIFFERENT club as ` +
   563→    `  the player's current team (e.g. "Ederson from Atalanta" when we expected Man City).\n` +
   564→    `- Mentioning a destination club (rumored move) does NOT count as a different current club.\n` +
   565→    `- If unsure, return verified=true (be permissive — other gates handle same-club and bad extractions).\n` +
   566→    `- Output ONLY the JSON object, no commentary.`
   567→
   568→  const result = await ai.chat(
   569→    [
   570→      { role: 'system', content: systemPrompt },
   571→      { role: 'user', content: postText.slice(0, 2000) },
   572→    ],
   573→    { temperature: 0.1, maxTokens: 400, json: true },
   574→  )
   575→
   576→  if (!result.ok || !result.content) {
   577→    console.warn(
   578→      `[transfer-pulse/discovery] verifyPlayerCurrentClub: ai.chat failed (${result.provider}), failing OPEN:`,
   579→      result.error?.slice(0, 100),
   580→    )
   581→    return { verified: true, actualClub: null }
   582→  }
   583→
   584→  // Parse the JSON response
   585→  let cleaned = result.content.trim()
   586→  cleaned = cleaned.replace(/^```(?:json)?\s*/i, '').replace(/\s*```\s*$/i, '')
   587→  const start = cleaned.indexOf('{')
   588→  const end = cleaned.lastIndexOf('}')
   589→  if (start === -1 || end === -1 || end <= start) {
   590→    // Unparseable → fail open (other gates still apply)
   591→    return { verified: true, actualClub: null }
   592→  }
   593→  try {
   594→    const obj = JSON.parse(cleaned.slice(start, end + 1))
   595→    const verified = obj.verified === true
   596→    const actualClub =
   597→      typeof obj.actualClub === 'string' && obj.actualClub.trim()
   598→        ? obj.actualClub.trim()
   599→        : null
   600→    return { verified, actualClub }
   601→  } catch {
   602→    return { verified: true, actualClub: null }
   603→  }
   604→}

   606→// ── Staleness guard: "has this player already moved?" ────────────────────────
   607→
   608→/**
   609→ * SYSTEMIC STALENESS GUARD — checks whether a tracked player has ALREADY
   610→ * completed a transfer AWAY from their watchlist `fromClubName`.
   611→ *
   612→ * BACKGROUND:
   613→ *   This guard was added 2026-07-26 after the "Florian Wirtz → Man City" bug.
   614→ *   Wirtz completed his move to Liverpool in summer 2025, but the watchlist
   615→ *   still had him at Leverkusen. The discovery pipeline kept finding OLD
   616→ *   pre-move Tier 1 posts (Plettenberg/Falk, July 21-22) linking him to Man
   617→ *   City — speculation that predated his actual move. The saga was created
   618→ *   with the WRONG destination and marked "active" even though the player
   619→ *   had already moved to a different club. The Arnold bug was the same
   620→ *   class (Alexander-Arnold → Real Madrid completed in 2025, still surfaced).
   621→ *
   622→ * GATE:
   623→ *   Before running discovery for a player, we ask the AI: "Has {player}
   624→ *   already completed a transfer away from {fromClubName}? If so, to which
   625→ *   club?" The AI returns {alreadyMoved: bool, actualClub: string|null,
   626→ *   confidence: 'high'|'medium'|'low'}.
   627→ *
   628→ * If `alreadyMoved=true` with high/medium confidence, the caller resolves the
   629→ * player's active sagas (see `resolvePlayerSagas`) and SKIPS discovery. This
   630→ * prevents stale pre-move rumors from being re-surfaced as "active" news.
   631→ *
   632→ * FALLBACK:
   633→ *   If the LLM call fails entirely, we FAIL OPEN (alreadyMoved=false) to
   634→ *   avoid blocking all discovery. The entity-resolution gate, same-club
   635→ *   guard, and 60-day freshness filter still apply as secondary defenses.
   636→ *   We only RESOLVE sagas on a positive, confident "already moved" answer.
   637→ *
   638→ * COST:
   639→ *   One LLM call per player per discovery run. Cheaper than the xAI x_search
   640→ *   call it replaces when the player has already moved (skips xAI entirely).
   641→ */
   642→async function checkPlayerAlreadyMoved(player: TrackedPlayer): Promise<{
   643→  alreadyMoved: boolean
   644→  actualClub: string | null
   645→  confidence: 'high' | 'medium' | 'low'
   646→}> {
   647→  // ── PRIMARY: web_search verification (added 2026-07-26) ────────────────
   648→  // The LLM's training data lags reality — Isak moved to Liverpool on 1 Sep
   649→  // 2025 but the LLM still thinks "Newcastle" in some prompts. Web search
   650→  // returns same-day articles so it's always fresher. We ask it what club
   651→  // the player CURRENTLY plays for and compare to the watchlist's fromClub.
   652→  try {
   653→    const web = await verifyPlayerCurrentClubViaWeb(player.name, player.fromClubName)
   654→    if (web.confidence !== 'low' && web.actualClub) {
   655→      const matches = clubsMatch(web.actualClub, player.fromClubName)
   656→      console.log(
   657→        `[transfer-pulse/discovery] checkPlayerAlreadyMoved: ${player.name} ` +
   658→          `web-verified=${web.actualClub} (conf=${web.confidence}) matches watchlist ${player.fromClubName}=${matches}`,
   659→      )
   660→      if (!matches) {
   661→        // Web says he's at a DIFFERENT club → already moved.
   662→        return {
   663→          alreadyMoved: true,
   664→          actualClub: web.actualClub,
   665→          confidence: web.confidence,
   666→        }
   667→      }
   668→      // Web confirms he's still at the watchlist club → not moved.
   669→      // Skip the LLM call entirely (saves budget; web is more authoritative).
   670→      return { alreadyMoved: false, actualClub: null, confidence: web.confidence }
   671→    }
   672→    // Low-confidence web → fall through to LLM check below.
   673→  } catch (err) {
   674→    console.warn(
   675→      `[transfer-pulse/discovery] checkPlayerAlreadyMoved web-check failed for ${player.name}, falling back to LLM: ${String(err).slice(0, 100)}`,
   676→    )
   677→  }
   678→
   679→  // ── FALLBACK: LLM-only check (preserved from prior version) ────────────
   680→  const systemPrompt =
   681→    `You are a football transfer fact-checker. Answer ONE question with ` +
   682→    `high precision: has the footballer ${player.name} ALREADY COMPLETED a ` +
   683→    `transfer AWAY from ${player.fromClubName} (his watchlist club) to a ` +
   684→    `DIFFERENT club — as of ${new Date().toISOString().slice(0, 10)}?\n\n` +
   685→    `Context:\n` +
   686→    `- The watchlist says ${player.name} is currently at ${player.fromClubName}.\n` +
   687→    `- If he is STILL at ${player.fromClubName}, answer alreadyMoved=false.\n` +
   688→    `- If he has COMPLETED a permanent transfer to another club (signed, ` +
   689→    `announced, presented — not just rumored), answer alreadyMoved=true and ` +
   690→    `name the actual destination club in actualClub.\n` +
   691→    `- Loan moves DO count as "moved" only if the loan is long-term (≥1 season) ` +
   692→    `and the player is no longer training/playing for ${player.fromClubName}.\n` +
   693→    `- A mere RUMOR of interest (even from Tier 1 journalists) does NOT count ` +
   694→    `as "moved" — only a COMPLETED transfer counts.\n\n` +
   695→    `Return a JSON object:\n` +
   696→    `  "alreadyMoved": boolean\n` +
   697→    `  "actualClub": string | null   — the destination club's full name, or null if not moved\n` +
   698→    `  "confidence": "high" | "medium" | "low"   — how sure you are\n` +
   699→    `  "reason": string              — 1-sentence explanation\n\n` +
   700→    `RULES:\n` +
   701→    `- Be CONSERVATIVE. If you are not sure the transfer is COMPLETED, answer ` +
   702→    `alreadyMoved=false. Resolving a saga prematurely hides real ongoing rumors.\n` +
   703→    `- Output ONLY the JSON object, no commentary.`
   704→
   705→  const result = await ai.chat(
   706→    [
   707→      { role: 'system', content: systemPrompt },
   708→      { role: 'user', content: `${player.name} — current club per watchlist: ${player.fromClubName}` },
   709→    ],
   710→    { temperature: 0.1, maxTokens: 400, json: true },
   711→  )
   712→
   713→  if (!result.ok || !result.content) {
   714→    console.warn(
   715→      `[transfer-pulse/discovery] checkPlayerAlreadyMoved: ai.chat failed (${result.provider}), failing OPEN:`,
   716→      result.error?.slice(0, 100),
   717→    )
   718→    return { alreadyMoved: false, actualClub: null, confidence: 'low' }
   719→  }
   720→
   721→  let cleaned = result.content.trim()
   722→  cleaned = cleaned.replace(/^```(?:json)?\s*/i, '').replace(/\s*```\s*$/i, '')
   723→  const start = cleaned.indexOf('{')
   724→  const end = cleaned.lastIndexOf('}')
   725→  if (start === -1 || end === -1 || end <= start) {
   726→    return { alreadyMoved: false, actualClub: null, confidence: 'low' }
   727→  }
   728→  try {
   729→    const obj = JSON.parse(cleaned.slice(start, end + 1))
   730→    const alreadyMoved = obj.alreadyMoved === true
   731→    const actualClub =
   732→      typeof obj.actualClub === 'string' && obj.actualClub.trim()
   733→        ? obj.actualClub.trim()
   734→        : null
   735→    const confidenceRaw = String(obj.confidence).toLowerCase().trim()
   736→    const confidence: 'high' | 'medium' | 'low' =
   737→      confidenceRaw === 'high' ? 'high' : confidenceRaw === 'medium' ? 'medium' : 'low'
   738→    // Only trust high/medium confidence "alreadyMoved=true" answers. Low
   739→    // confidence = fail open (keep the saga active, let discovery run).
   740→    if (alreadyMoved && confidence === 'low') {
   741→      console.log(
   742→        `[transfer-pulse/discovery] ${player.name}: LLM says moved but low confidence — failing open (keeping active)`,
   743→      )
   744→      return { alreadyMoved: false, actualClub: null, confidence: 'low' }
   745→    }
   746→    return { alreadyMoved, actualClub, confidence }
   747→  } catch {
   748→    return { alreadyMoved: false, actualClub: null, confidence: 'low' }
   749→  }

   752→/**
   753→ * Resolve all of a player's active sagas after determining they've already
   754→ * moved. For each active saga:
   755→ *   - If the saga's `toClubName` matches the player's ACTUAL new club →
   756→ *     mark `completed` (the rumor was correct and came true).
   757→ *   - If the saga's `toClubName` DIFFERS from the actual new club →
   758→ *     mark `debunked` (the rumor was wrong; he went elsewhere).
   759→ *
   760→ * Resolved sagas keep all their sources/posts/timeline (audit trail
   761→ * preserved per the anti-hallucination contract). Only the `status` and
   762→ * `resolvedAt` fields change.
   763→ *
   764→ * @returns the number of sagas whose status was updated.
   765→ */
   766→async function resolvePlayerSagas(
   767→  player: TrackedPlayer,
   768→  actualClub: string,
   769→  confidence: 'high' | 'medium' | 'low',
   770→): Promise<number> {
   771→  // Only resolve on confident answers (the checkPlayerAlreadyMoved gate
   772→  // already filters low confidence, but double-guard here for safety).
   773→  if (confidence === 'low') return 0
   774→
   775→  const activeSagas = await db.transferSaga.findMany({
   776→    where: { playerName: player.name, status: 'active' },
   777→  })
   778→  if (activeSagas.length === 0) return 0
   779→
   780→  const actualLower = actualClub.toLowerCase().trim()
   781→  let updated = 0
   782→  for (const saga of activeSagas) {
   783→    const sagaToLower = saga.toClubName.toLowerCase().trim()
   784→    // Match if either string contains the other (handles "Manchester City"
   785→    // vs "Man City" naming differences).
   786→    const isMatch =
   787→      actualLower.includes(sagaToLower) || sagaToLower.includes(actualLower)
   788→    const newStatus: 'completed' | 'debunked' = isMatch ? 'completed' : 'debunked'
   789→    await db.transferSaga.update({
   790→      where: { id: saga.id },
   791→      data: {
   792→        status: newStatus,
   793→        resolvedAt: new Date(),
   794→        lastUpdatedAt: new Date(),
   795→      },
   796→    })
   797→    console.log(
   798→      `[transfer-pulse/discovery] resolved ${player.name} → ${saga.toClubName} ` +
   799→        `as [${newStatus}] (actual club: ${actualClub})`,
   800→    )
   801→    updated++
   802→  }
   803→  return updated
   804→}

   814→/**
   815→ * ENTITY-NAME-OVERLAP DETECTOR — catches substring false-matches between
   816→ * tracked players and other footballers whose names contain the tracked
   817→ * name. Returns the longer (different) name if a collision is detected, or
   818→ * null if the post is safely about the tracked player.
   819→ *
   820→ * EXAMPLES (tracked player = "Rodri"):
   821→ *   • post mentions "Álvaro Rodríguez" → returns "Álvaro Rodríguez" (REJECT)
   822→ *   • post mentions "Rodri" standalone  → returns null (ACCEPT)
   823→ *   • post mentions "Rodri Hernández"   → returns null (ACCEPT — full name
   824→ *     contains "Rodri" but the standalone short name also appears, so it's
   825→ *     plausibly the same player)
   826→ *
   827→ * The detector is deliberately conservative: it only fires when (a) the
   828→ * tracked player's EXACT name does NOT appear as a standalone word in the
   829→ * post, AND (b) the post contains a longer name that starts with or contains
   830→ * the tracked name as a substring. This avoids false rejections of posts
   831→ * that genuinely mention the tracked player alongside others.
   832→ *
   833→ * Known overlap pairs this guards against:
   834→ *   "Rodri" ⊂ "Rodríguez", "Rodri Hernández"
   835→ *   "Pedri" ⊂ "Pedrinho"
   836→ *   "Gavi" ⊂ "Gavilán"
   837→ *   (single-name players are most at risk — that's why the entity-resolution
   838→ *   gate + this heuristic both exist)
   839→ */
   840→function hasEntityNameOverlap(postText: string, playerName: string): string | null {
   841→  const text = postText
   842→  const playerLower = playerName.toLowerCase()
   843→  // Check if the tracked player's EXACT name appears as a standalone token.
   844→  // Word-boundary regex: \b ensures "Rodri" doesn't match inside "Rodríguez".
   845→  const exactRe = new RegExp(`\\b${escapeRegex(playerLower)}\\b`, 'i')
   846→  if (exactRe.test(text)) {
   847→    // The exact name appears standalone — accept the post.
   848→    return null
   849→  }
   850→  // The exact name does NOT appear standalone. Now check whether a LONGER
   851→  // name containing the tracked name appears (e.g. "Álvaro Rodríguez").
   852→  // We look for patterns like "<Word> <tracked-name>..." or "<tracked-name><suffix>".
   853→  // Specifically: any capitalized word sequence where one token starts with
   854→  // or contains the tracked name as a substring.
   855→  //
   856→  // To keep this cheap and false-positive-free, we only flag the specific
   857→  // known-dangerous pattern: a longer surname that STARTS WITH the tracked
   858→  // name and continues with more letters (no word boundary). This catches
   859→  // "Rodríguez" from "Rodri", "Pedrinho" from "Pedri", etc.
   860→  const suffixRe = new RegExp(`\\b[A-Z][a-z]+\\s+${escapeRegex(playerLower)}[a-záéíóúñ]+`, 'i')
   861→  // Also catch the pattern where a full name is given and the tracked name
   862→  // is a prefix of the surname: "Álvaro Rodríguez" → "Rodri" + "guez"
   863→  const prefixRe = new RegExp(`\\b[A-Z][a-záéíóúñ]+\\s+${escapeRegex(playerLower)}[a-záéíóúñ]*`, 'i')
   864→  const match = text.match(prefixRe) || text.match(suffixRe)
   865→  if (match) {
   866→    return match[0]
   867→  }
   868→  return null
   869→}

   871→/** Escape special regex characters in a string so it can be used in a RegExp. */
   872→function escapeRegex(s: string): string {
   873→  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   874→}
