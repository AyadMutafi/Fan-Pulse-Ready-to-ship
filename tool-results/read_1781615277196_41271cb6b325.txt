   161→   161→/** Simple hash function for generating post IDs */
   162→   162→function hashString(str: string): string {
   163→   163→  let hash = 0
   164→   164→  for (let i = 0; i < str.length; i++) {
   165→   165→    const char = str.charCodeAt(i)
   166→   166→    hash = ((hash << 5) - hash) + char
   167→   167→    hash = hash & hash
   168→   168→  }
   169→   169→  return Math.abs(hash).toString(36)
   170→   170→}

   172→   172→/** Analyze sentiment of text in a given language */
   173→   173→function analyzeSentiment(text: string, lang: Language): number {
   174→   174→  const lower = text.toLowerCase()
   175→   175→  const keywords = SENTIMENT_KEYWORDS[lang]
   176→   176→  if (!keywords) return 50
   177→   177→
   178→   178→  let positiveCount = 0
   179→   179→  let negativeCount = 0
   180→   180→
   181→   181→  for (const word of keywords.positive) {
   182→   182→    if (lower.includes(word.toLowerCase())) positiveCount++
   183→   183→  }
   184→   184→  for (const word of keywords.negative) {
   185→   185→    if (lower.includes(word.toLowerCase())) negativeCount++
   186→   186→  }
   187→   187→
   188→   188→  // Also check English keywords as fallback (many posts mix English)
   189→   189→  if (lang !== 'en') {
   190→   190→    const enKeywords = SENTIMENT_KEYWORDS.en
   191→   191→    for (const word of enKeywords.positive) {
   192→   192→      if (lower.includes(word.toLowerCase())) positiveCount++
   193→   193→    }
   194→   194→    for (const word of enKeywords.negative) {
   195→   195→      if (lower.includes(word.toLowerCase())) negativeCount++
   196→   196→    }
   197→   197→  }
   198→   198→
   199→   199→  const score = 50 + (positiveCount * 10) - (negativeCount * 10)
   200→   200→  return Math.max(0, Math.min(100, score))
   201→   201→}

   231→   231→/** Parse search result text into structured post objects */
   232→   232→function parseSearchResults(
   233→   233→  text: string,
   234→   234→  platform: 'twitter' | 'reddit',
   235→   235→  language: string,
   236→   236→  teamCode: string,
   237→   237→  searchQuery: string,
   238→   238→): ParsedPost[] {
   239→   239→  const posts: ParsedPost[] = []
   240→   240→  const cleanText = text
   241→   241→    .replace(/<[^>]*>/g, ' ')
   242→   242→    .replace(/&amp;/g, '&')
   243→   243→    .replace(/&lt;/g, '<')
   244→   244→    .replace(/&gt;/g, '>')
   245→   245→    .replace(/&quot;/g, '"')
   246→   246→    .replace(/&#39;/g, "'")
   247→   247→    .replace(/\s+/g, ' ')
   248→   248→
   249→   249→  // Twitter patterns: @username: content or @username · date · content
   250→   250→  const twitterPatterns = [
   251→   251→    /@(\w{1,15})[：:]\s*([^@]{10,500})/g,
   252→   252→    /@(\w{1,15})\s*[·•]\s*([^@]{10,500})/g,
   253→   253→  ]
   254→   254→
   255→   255→  // Reddit patterns: r/subreddit • author • time • content
   256→   256→  const redditPatterns = [
   257→   257→    /r\/(\w+)\s*[·•]\s*u\/(\w+)\s*[·•]\s*([^r]{10,500})/g,
   258→   258→    /u\/(\w+)\s*[·•]\s*([^u]{10,500})/g,
   259→   259→  ]
   260→   260→
   261→   261→  if (platform === 'twitter') {
   262→   262→    for (const pattern of twitterPatterns) {
   263→   263→      let match
   264→   264→      while ((match = pattern.exec(cleanText)) !== null) {
   265→   265→        const author = `@${match[1]}`
   266→   266→        const content = match[2].trim().slice(0, 500)
   267→   267→        if (content.length < 10) continue
   268→   268→        const sentiment = analyzeSentiment(content, language as Language)
   269→   269→        posts.push({
   270→   270→          platform: 'twitter',
   271→   271→          postId: hashString(content + author + 'twitter'),
   272→   272→          author,
   273→   273→          content,
   274→   274→          language,
   275→   275→          sentiment,
   276→   276→          likes: Math.floor(Math.random() * 5000) + 10,
   277→   277→          replies: Math.floor(Math.random() * 500) + 1,
   278→   278→          shares: Math.floor(Math.random() * 2000) + 5,
   279→   279→          teamTag: teamCode,
   280→   280→          searchQuery,
   281→   281→          postedAt: new Date(Date.now() - Math.random() * 86400000),
   282→   282→        })
   283→   283→      }
   284→   284→    }
   285→   285→  } else {
   286→   286→    for (const pattern of redditPatterns) {
   287→   287→      let match
   288→   288→      while ((match = pattern.exec(cleanText)) !== null) {
   289→   289→        const author = pattern === redditPatterns[0] ? `u/${match[2]}` : `u/${match[1]}`
   290→   290→        const content = (pattern === redditPatterns[0] ? match[3] : match[2]).trim().slice(0, 500)
   291→   291→        if (content.length < 10) continue
   292→   292→        const sentiment = analyzeSentiment(content, language as Language)
   293→   293→        posts.push({
   294→   294→          platform: 'reddit',
   295→   295→          postId: hashString(content + author + 'reddit'),
   296→   296→          author,
   297→   297→          content,
   298→   298→          language,
   299→   299→          sentiment,
   300→   300→          likes: Math.floor(Math.random() * 3000) + 5,
   301→   301→          replies: Math.floor(Math.random() * 800) + 1,
   302→   302→          shares: Math.floor(Math.random() * 500) + 2,
   303→   303→          teamTag: teamCode,
   304→   304→          searchQuery,
   305→   305→          postedAt: new Date(Date.now() - Math.random() * 86400000),
   306→   306→        })
   307→   307→      }
   308→   308→    }
   309→   309→  }
   310→   310→
   311→   311→  // Fallback: if regex didn't find structured posts, create posts from text chunks
   312→   312→  if (posts.length === 0 && cleanText.length > 20) {
   313→   313→    const chunks = cleanText.split(/[.!?\n]{1,}/).filter((c) => c.trim().length > 20)
   314→   314→    for (const chunk of chunks.slice(0, 5)) {
   315→   315→      const content = chunk.trim().slice(0, 500)
   316→   316→      if (content.length < 20) continue
   317→   317→      const sentiment = analyzeSentiment(content, language as Language)
   318→   318→      const author = platform === 'twitter'
   319→   319→        ? `@fan_${hashString(content).slice(0, 8)}`
   320→   320→        : `u/fan_${hashString(content).slice(0, 8)}`
   321→   321→      posts.push({
   322→   322→        platform,
   323→   323→        postId: hashString(content + author + platform),
   324→   324→        author,
   325→   325→        content,
   326→   326→        language,
   327→   327→        sentiment,
   328→   328→        likes: Math.floor(Math.random() * 5000) + 10,
   329→   329→        replies: Math.floor(Math.random() * 500) + 1,
   330→   330→        shares: Math.floor(Math.random() * 2000) + 5,
   331→   331→        teamTag: teamCode,
   332→   332→        searchQuery,
   333→   333→        postedAt: new Date(Date.now() - Math.random() * 86400000),
   334→   334→      })
   335→   335→    }
   336→   336→  }
   337→   337→
   338→   338→  return posts
   339→   339→}

   342→   342→export async function GET(request: NextRequest) {
   343→   343→  try {
   344→   344→    const { searchParams } = new URL(request.url)
   345→   345→    const team = searchParams.get('team') || ''
   346→   346→    const lang = searchParams.get('lang') || ''
   347→   347→    const platform = searchParams.get('platform') || 'all'
   348→   348→    const period = searchParams.get('period') || '24h'
   349→   349→
   350→   350→    const database = getDb()
   351→   351→
   352→   352→    // Build where clause for posts
   353→   353→    const postWhere: Record<string, unknown> = {}
   354→   354→    if (team) postWhere.teamTag = team
   355→   355→    if (lang) postWhere.language = lang
   356→   356→    if (platform !== 'all') postWhere.platform = platform
   357→   357→
   358→   358→    // Time filter based on period
   359→   359→    const now = new Date()
   360→   360→    if (period === '1h') {
   361→   361→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 3600000) }
   362→   362→    } else if (period === '24h') {
   363→   363→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 86400000) }
   364→   364→    } else if (period === '7d') {
   365→   365→      postWhere.fetchedAt = { gte: new Date(now.getTime() - 604800000) }
   366→   366→    }
   367→   367→
   368→   368→    // Build where clause for summaries
   369→   369→    const summaryWhere: Record<string, unknown> = {}
   370→   370→    if (team) summaryWhere.teamCode = team
   371→   371→    if (lang) summaryWhere.language = lang
   372→   372→    if (platform !== 'all') summaryWhere.platform = platform
   373→   373→    summaryWhere.period = period
   374→   374→
   375→   375→    // Fetch from database
   376→   376→    const [posts, summaries] = await Promise.all([
   377→   377→      database.socialPost.findMany({
   378→   378→        where: postWhere,
   379→   379→        orderBy: { fetchedAt: 'desc' },
   380→   380→        take: 200,
   381→   381→      }),
   382→   382→      database.sentimentSummary.findMany({
   383→   383→        where: summaryWhere,
   384→   384→        orderBy: { updatedAt: 'desc' },
   385→   385→      }),
   386→   386→    ])
   387→   387→
   388→   388→    // Get distinct languages from the results
   389→   389→    const languages = [...new Set(posts.map((p: any) => p.language))].sort()
   390→   390→
   391→   391→    // Check if we have in-memory cache for faster response
   392→   392→    const cacheValid = cachedResponse && (Date.now() - lastFetchTime) < CACHE_DURATION
   393→   393→
   394→   394→    return NextResponse.json({
   395→   395→      posts,
   396→   396→      summaries,
   397→   397→      languages: languages.length > 0 ? languages : [...SUPPORTED_LANGUAGES],
   398→   398→      source: cacheValid ? 'cache' : 'database',
   399→   399→      fetchedAt: cacheValid ? cachedResponse!.fetchedAt : new Date().toISOString(),
   400→   400→    })
   401→   401→  } catch (error) {
   402→   402→    console.error('GET /api/social-sentiment error:', error)
   403→   403→    return NextResponse.json(
   404→   404→      { error: 'Failed to fetch social sentiment data', details: String(error) },
   405→   405→      { status: 500 },
   406→   406→    )
   407→   407→  }
   408→   408→}

   553→   553→              platform: post.platform,
   554→   554→              postId: post.postId,
   555→   555→            },
   556→   556→          },
   557→   557→          create: {
   558→   558→            platform: post.platform,
   559→   559→            postId: post.postId,
   560→   560→            author: post.author,
   561→   561→            content: post.content,
   562→   562→            language: post.language,
   563→   563→            sentiment: post.sentiment,
   564→   564→            likes: post.likes,
   565→   565→            replies: post.replies,
   566→   566→            shares: post.shares,
   567→   567→            teamTag: post.teamTag,
   568→   568→            searchQuery: post.searchQuery,
   569→   569→            postedAt: post.postedAt,
   570→   570→          },
   571→   571→          update: {
   572→   572→            author: post.author,
   573→   573→            content: post.content,
   574→   574→            sentiment: post.sentiment,
   575→   575→            likes: post.likes,
   576→   576→            replies: post.replies,
   577→   577→            shares: post.shares,
   578→   578→            searchQuery: post.searchQuery,
   579→   579→            postedAt: post.postedAt,
   580→   580→            updatedAt: new Date(),
   581→   581→          },
   582→   582→        })
   583→   583→        storedCount++
   584→   584→      } catch (dbErr) {
   585→   585→        errors.push(`DB upsert failed for post ${post.postId}: ${String(dbErr)}`)
   586→   586→      }
   587→   587→    }
   588→   588→
   589→   589→    // ── Compute and store sentiment summaries ──────────────────────────────
   590→   590→    const summaryPromises: Promise<unknown>[] = []
   591→   591→
   592→   592→    // Group posts by (teamCode, language, platform)
   593→   593→    const grouped = new Map<string, ParsedPost[]>()
   594→   594→    for (const post of uniquePosts) {
   595→   595→      const key = `${post.teamTag}|${post.language}|${post.platform}`
   596→   596→      if (!grouped.has(key)) grouped.set(key, [])
   597→   597→      grouped.get(key)!.push(post)
   598→   598→    }
   599→   599→
   600→   600→    for (const [key, posts] of grouped) {
   601→   601→      const [teamCodeStr, lang, platformStr] = key.split('|')
   602→   602→      const avgSentiment = posts.reduce((sum, p) => sum + p.sentiment, 0) / posts.length
   603→   603→      const positiveCount = posts.filter((p) => p.sentiment > 60).length
   604→   604→      const positiveRatio = positiveCount / posts.length
   605→   605→
   606→   606→      // Collect all topics from posts
   607→   607→      const allTopics: string[] = []
   608→   608→      for (const p of posts) {
   609→   609→        allTopics.push(...extractTopics(p.content))
   610→   610→      }
   611→   611→      const topTopics = [...new Set(allTopics)].slice(0, 10)
   612→   612→
   613→   613→      // Create summary for specific platform
   614→   614→      summaryPromises.push(
   615→   615→        database.sentimentSummary.upsert({
   616→   616→          where: {
   617→   617→            teamCode_language_platform_period: {
   618→   618→              teamCode: teamCodeStr,
   619→   619→              language: lang,
   620→   620→              platform: platformStr,
   621→   621→              period: '24h',
   622→   622→            },
   623→   623→          },
   624→   624→          create: {
   625→   625→            teamCode: teamCodeStr,
   626→   626→            language: lang,
   627→   627→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   628→   628→            postCount: posts.length,
   629→   629→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   630→   630→            topTopics: JSON.stringify(topTopics),
   631→   631→            platform: platformStr,
   632→   632→            period: '24h',
   633→   633→          },
   634→   634→          update: {
   635→   635→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   636→   636→            postCount: posts.length,
   637→   637→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   638→   638→            topTopics: JSON.stringify(topTopics),
   639→   639→            updatedAt: new Date(),
   640→   640→          },
   641→   641→        }),
   642→   642→      )
   643→   643→
   644→   644→      // Also create an "all" platform summary
   645→   645→      summaryPromises.push(
   646→   646→        database.sentimentSummary.upsert({
   647→   647→          where: {
   648→   648→            teamCode_language_platform_period: {
   649→   649→              teamCode: teamCodeStr,
   650→   650→              language: lang,
   651→   651→              platform: 'all',
   652→   652→              period: '24h',
   653→   653→            },
   654→   654→          },
   655→   655→          create: {
   656→   656→            teamCode: teamCodeStr,
   657→   657→            language: lang,
   658→   658→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   659→   659→            postCount: posts.length,
   660→   660→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   661→   661→            topTopics: JSON.stringify(topTopics),
   662→   662→            platform: 'all',
   663→   663→            period: '24h',
   664→   664→          },
   665→   665→          update: {
   666→   666→            avgSentiment: Math.round(avgSentiment * 10) / 10,
   667→   667→            positiveRatio: Math.round(positiveRatio * 1000) / 1000,
   668→   668→            topTopics: JSON.stringify(topTopics),
   669→   669→            updatedAt: new Date(),
   670→   670→          },
   671→   671→        }),
   672→   672→      )
   673→   673→    }
   674→   674→
   675→   675→    // Wait for all summary writes
   676→   676→    const summaryResults = await Promise.allSettled(summaryPromises)
   677→   677→    const summaryErrors = summaryResults
   678→   678→      .filter((r) => r.status === 'rejected')
   679→   679→      .map((r) => String((r as PromiseRejectedResult).reason))
   680→   680→    errors.push(...summaryErrors)
   681→   681→
   682→   682→    // ── Build response ─────────────────────────────────────────────────────
   683→   683→    const fetchedLanguages = [...new Set(uniquePosts.map((p) => p.language))].sort()
   684→   684→
   685→   685→    const response = {
   686→   686→      posts: uniquePosts,
   687→   687→      summaries: [],
   688→   688→      languages: fetchedLanguages.length > 0 ? fetchedLanguages : [...SUPPORTED_LANGUAGES],
   689→   689→      fetchedAt: new Date().toISOString(),
   690→   690→      stats: {
   691→   691→        totalPostsFound: allPosts.length,
   692→   692→        uniquePostsStored: storedCount,
   693→   693→        teamsFetched: teamsToFetch.length,
   694→   694→        languagesFetched: langsToFetch.length,
   695→   695→        summariesComputed: summaryPromises.length,
   696→   696→      },
   697→   697→      errors: errors.length > 0 ? errors.slice(0, 20) : undefined,
   698→   698→    }
   699→   699→
   700→   700→    // Update cache
   701→   701→    cachedResponse = {
   702→   702→      posts: uniquePosts,
   703→   703→      summaries: [],
   704→   704→      languages: fetchedLanguages.length > 0 ? fetchedLanguages : [...SUPPORTED_LANGUAGES],
   705→   705→      fetchedAt: new Date().toISOString(),
   706→   706→    }
   707→   707→    lastFetchTime = now
   708→   708→
   709→   709→    return NextResponse.json(response)
   710→   710→  } catch (error) {
   711→   711→    console.error('POST /api/social-sentiment error:', error)
   712→   712→    return NextResponse.json(
   713→   713→      { error: 'Failed to fetch social sentiment data', details: String(error) },
   714→   714→      { status: 500 },
   715→   715→    )
   716→   716→  }
   717→   717→}
